                E.setPos( Pos )

                Pos = E.pos()
                Cur = ""
                return
            }

            //Increase the array position
            X++
            
            // if end of first row on array, proceed to second row
            if(X > 5){
                X = 1
                Y++

                E.setPos( E.toWorld( vec(-7,-35,0) ) )
            }
                                        
            
            // If end of word, move to next word position
            if(Y > 5){
                
                // Work out distance of gap between words
                int Z = 0
                for(int I = 1;25;5){
                    if(Let[I,int]) Z = max(Z,1)
                    if(Let[I+1,int]) Z = max(Z,2)
                    if(Let[I+2,int]) Z = max(Z,3)
                    if(Let[I+3,int]) Z = max(Z,4)
                    if(Let[I+4,int]) Z = max(Z,5)
                }
                
                
                
                // Wouldnt want the previous statement to think space is just 35 Y in distance ( useless space )
                if(Cur == " "){
                    E.setPos(E.toWorld( vec(42, 70 , 0) ) )
                }else{
                    E.setPos(E.toWorld( vec(42, 35 + (Z*8.75) , 0) ) )
                }
                
                // Reset Cur to recheck the buffer( Variable S ) and reset X,Y for future use.
                Cur = ""
                X,Y = 0,0
                
            }
            
            // Is the current binary set to print?
            if(Let[X+(Y*5),int]){
                W["Paint",int] = 1
                W["Paint",int] = 0
            }
            
            //Move right each time
            E.setPos( E.toWorld( vec(0,7,0) ) )        
        }
    }
