N@Us3r's Driver Bot
I@TargetX TargetY PlayerX PlayerY Engage
O@Angle
#Reset cmds if engaged/disengaging
Engage | (~Engage & !Engage) -> concommand('-forward; -moveleft; -moveright; -speed');
#If not engaged stop here
!Engage -> end;
#Create the vectors
T = vector(TargetX, TargetY, 0)
P = vector(PlayerX, PlayerY, 0)
D = vector($PlayerX, $PlayerY, 0)
#Get the angle between the player and the target
Angle = angnorm(vecyaw(D) - vecyaw(vecsub(T, P)))
#Speed limit
abs(Angle) < 5 | veclength(D) <= 30 -> concommand('+forward');
#Avoid circling around the target
vecdistance(T, P) <= 1000 & abs(Angle) >= 90 -> end;
#Turn if needed
Angle <= -5 -> concommand('+moveleft');
Angle >= 5 -> concommand('+moveright');
