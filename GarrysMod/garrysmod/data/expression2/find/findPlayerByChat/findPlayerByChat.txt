@name findPlayerByChat Explanation (Hologram)
@persist Target:entity

runOnTick(1)
runOnChat(1) # This enables chat commands

if (chatClk(owner())) {
    LS = lastSaid():explode(" ") # This basicly seperates string1 and string2. I will explain that later on.
    if ( LS[1,string] == "-target" ) { # This is string1. String1 is the chat command you will use.
    Target = findPlayerByName((LS[2,string])) # This will find the player that you put into string2.
    # If you change explode(" ") to explode("A") it would be -targetABoom or if you did explode("+") it would be -target+Boom get it?
    # So the format is string1(whatever you put in explode)string2
    # In this one string1 is -target and we are exploding it by - and the player's name is string2
    # This is what you should type in chat ====== -target Boom (You can replace "Boom" with any other playername.)
}
}

# Now for the hologram.
holoCreate(1) # Creates the hologram
holoModel(1,"hqicosphere2") # Changes the model of the hologram
holoColor(1,vec(255,150,0)) # Changes the color of the hologram
holoPos(1,Target:pos()+vec(0,0,100)) # This sets the position of the hologram to the player you are targeting's position then adds 100 to the Z vector thingy.
holoParent(1,Target) # This parents the hologram to the player you are targeting.
