@name RoomDetector
@outputs Clk Player:entity Name:string
@persist Holo IgnoreOwner Dynamic TC:vector BC:vector Players:array
interval(5)
if (first()|dupefinished()){
    runOnChat(1)
    rangerHitEntities(1)
    
    IgnoreOwner=1
    Dynamic=0
    Holo=0
    
    X1=rangerOffset(5000,entity():pos(),vec(1,0,0))
    X2=rangerOffset(5000,entity():pos(),vec(-1,0,0))
    XS=X1:distance()+X2:distance()
    
    Y1=rangerOffset(5000,entity():pos(),vec(0,1,0))
    Y2=rangerOffset(5000,entity():pos(),vec(0,-1,0))
    YS=Y1:distance()+Y2:distance()
    
    ZS=ranger(500):distance()
    
    MID=vec(((X1:pos()+X2:pos())/2):x(),((Y1:pos()+Y2:pos())/2):y(),entity():pos():z()+ZS/2)
    if (Holo){
        holoCreate(1,MID)
        holoScaleUnits(1,-vec(XS,YS,ZS))
        holoAlpha(1,80)
    }
    
    TC=MID+vec(XS/2,YS/2,ZS/2)
    BC=MID-vec(XS/2,YS/2,ZS/2)
    #holoCreate(2,TC)
    #holoCreate(3,BC)
}

if (Dynamic){
    X1=rangerOffset(5000,entity():pos(),vec(1,0,0))
    X2=rangerOffset(5000,entity():pos(),vec(-1,0,0))
    XS=X1:distance()+X2:distance()
    
    Y1=rangerOffset(5000,entity():pos(),vec(0,1,0))
    Y2=rangerOffset(5000,entity():pos(),vec(0,-1,0))
    YS=Y1:distance()+Y2:distance()
    
    ZS=ranger(500):distance()
    
    MID=vec(((X1:pos()+X2:pos())/2):x(),((Y1:pos()+Y2:pos())/2):y(),entity():pos():z()+ZS/2)
    if (Holo){
        holoPos(1,MID)
        holoScaleUnits(1,-vec(XS,YS,ZS))
    }
    
    TC=MID+vec(XS/2,YS/2,ZS/2)
    BC=MID-vec(XS/2,YS/2,ZS/2)
}


if (findCanQuery()){
    if (IgnoreOwner){
        findExcludePlayer(owner())
    }
    findIncludeClass("player")
    findInBox(TC,BC)
    if (find()){
        holoColor(1,vec(0,255,0))
        Clk=1
        Player=find()
        Name=find():name()
    } else {
        Clk=0
        Name="N/A"
        Player=noentity()
        if (Holo){
            holoColor(1,vec(255,0,0))
        }
    }
}
