@name FreezeEverything
@persist Pos:array Ang:array Props:array I First
interval(2)
if (first()|dupefinished()){
    print(4,"Wait for the props to lock...")
    Props=entity():getConstraints()
    First=1
    I=0
}

I++
P=Props[I,entity]
if (First){
    Pos:pushVector(P:pos())
    Ang:pushAngle(P:angles())
}

P:setPos(Pos[I,vector])
P:setAng(Ang[I,angle])
P:propFreeze(1)

if (I==Props:count()){
    if (First){
        print(4,"Locked "+I+" props!")
    }
    First=0
    I=0
}
