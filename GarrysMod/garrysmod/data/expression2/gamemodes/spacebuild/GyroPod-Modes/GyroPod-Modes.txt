@name GyroPod-Modes
@inputs Gyro:wirelink Pod:wirelink 
@persist I P M2 Atv 
@model
#--------------------===============[|Created by IronManhood|]===============--------------------
runOnTick(1)
    
if(changed(P)&P) {I=!I} ##Toggles between modes
P = Pod["R", number] ##Sets "P" equal to "R" from the pod controller
M2 = Pod["Mouse2", number] 
Atv = Pod["Active", number]
Gyro["Activate", number] = Pod["Active", number] 
Gyro["Freeze", number] = Pod["Active", number]==0 ##Sets "Freeze" on the gyropod, to activate when "Active" on the pod controller is 0
if(Atv) ##Keeps the lower part of code from constantly running if nobody is in the the vehicle, tho not really a whole lot of change. Just good practice.
{
 Gyro["Forward", number] = Pod["W", number] ##Sets "Forward" on the gyropod, equal to "W" from the pod controller
 Gyro["Back", number] = Pod["S", number] 
 Gyro["MoveUp", number] = Pod["Space", number] 
 Gyro["MoveDown", number] = Pod["Shift", number] 
    if(!I) ##HOVERMODE
       {
        Gyro["Level", number] = Pod["Active", number]
        ,Gyro["MoveLeft", number] = Pod["A", number]
        ,Gyro["MoveRight", number] = Pod["D", number]
       }
    
    if(I) ##FLIGHTMODE
       {
        Gyro["Level", number] = 0
        ,if(M2) 
            {
             Gyro["RollLeft", number] = 0
             ,Gyro["RollRight", number] = 0
             ,Gyro["MoveLeft", number] = Pod["A", number]
             ,Gyro["MoveRight", number] = Pod["D", number]
            }
              else
                {
                 Gyro["MoveLeft", number] = 0 
                 ,Gyro["MoveRight", number] = 0
                 ,Gyro["RollLeft", number] = Pod["A", number]
                 ,Gyro["RollRight", number] = Pod["D", number]
                }      
       }
}
