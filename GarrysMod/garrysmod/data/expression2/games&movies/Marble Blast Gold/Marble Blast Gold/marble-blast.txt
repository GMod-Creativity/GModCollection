@name Marble Blast
@inputs Pod:wirelink Cam:wirelink
@persist E:entity Force:vector Speed InAir
@outputs Hit
runOnTick(1)

if (first())
{
    InAir=0
    Speed=5
    findByModel("models/hunter/misc/sphere025x025.mdl")
    findIncludePlayerProps(owner())
    E=findClosest(entity():pos())
    if (E)
    {
        if (E:getMaterial()!="models/shiny")
        {
            E:setColor(255,255,0,255)
        }
        E:setMaterial("models/shiny")
        E:setMass(50)
    }
    else
    {
        hint("No Ball spawned.",5)
        selfDestruct()
    }
}

Pers=Pod["Entity",entity]:driver()

Cam["Activated",number]=Pod["Active",number]
Cam["Position",vector]=(E:pos())+(Pers:eye()*-100)
Cam["Angle",angle]=Pers:eyeAngles()

if (Pod["W",number])
{
    E:applyForce((Pers:eye()*Speed):setZ(0)*E:mass())
}
if (Pod["S",number])
{
    E:applyForce((Pers:eye()*-Speed):setZ(0)*E:mass())
}
if (Pod["A",number])
{
    E:applyForce(((Pers:eye():rotate(ang(0,90,0)))*Speed):setZ(0)*E:mass())
}
if (Pod["D",number])
{
    E:applyForce(((Pers:eye():rotate(ang(0,-90,0)))*Speed):setZ(0)*E:mass())
}

rangerFilter(E)
rangerIgnoreWorld(0)
Ranger = rangerOffset(E:pos(),E:pos()-vec(0,0,20))

Hit=Ranger:hit()

if (changed(Pod["Space",number]))
{
    if (Pod["Space",number])
    {
        if (Hit|InAir)
        {
            E:applyForce(vec(0,0,300)*E:mass())
        }
    }
}

if (changed(Pod["Active",number]))
{
    E:applyForce(-(E:vel()))
    E:setPos(entity():pos()+vec(0,0,30))
}

if (Pod["Active",number]==0)
{
    Force=((entity():pos()+vec(0,0,30))-(E:pos()))
    E:applyForce((Force+$Force*5)*E:mass())
}
