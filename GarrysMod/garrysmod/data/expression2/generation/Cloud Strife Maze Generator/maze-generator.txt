    function number getLen(XLen, YWidth, ZHeight, ADir)
    {

        while(inrange(vec(XLen,YWidth,ZHeight),vec(),vec(Length,Width,Height)) & Walls[getWallIndex(XLen, YWidth, ZHeight, ADir),number] != 1)
        {
            Walls[getWallIndex(XLen, YWidth, ZHeight, ADir),number] = 1
            Len = Len + 1
            XLen = XLen + MoveDir[ADir,vector]:x()
            YWidth = YWidth + MoveDir[ADir,vector]:y()
            ZHeight = ZHeight + MoveDir[ADir,vector]:z()
            if(!perf())
            {
                Abort = 4
                SavedVec = vec(XLen,YWidth,ZHeight)
                SavedLen = Len
                exit()
            }
        }  
        return (Len * HorizontalSpacing)
    }

    function void progressBar(Percent,BarIndex)
    {
        while(BarIndex>BarCount)
        {
            BarCount=BarCount + 1
            holoCreate(-1-2*BarCount)
            holoColor(-1-2*BarCount,vec())
            holoCreate(-2-2*BarCount)
            holoColor(-2-2*BarCount,vec(randint(200),randint(200),0))

        holoScaleUnits((-2-2*BarIndex),vec(49*Percent,3,4))
    }
