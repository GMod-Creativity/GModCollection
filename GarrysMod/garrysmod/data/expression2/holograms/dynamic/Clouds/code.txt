@name Clouds!
@outputs Height [HeightVec]:vector CL CLG CLC GX GY
@outputs [CloudParent CloudPos]:array MoveCloud CloudsCount
@outputs [GridBase]:vector2
@outputs [GridPosX GridPosY GridInfo]:array MapXF MapXB MapYF MapYB

if (first()) {
    
    GridBase = vec2(4, 3) #Size of grid for clouds (Max is 6x5 or 5x6)
    Height = 0 #Leave 0 if not sure
    CloudsCount = 6 #Clouds spawned
    
    
    Height = Height
    HeightVec = vec(0,0, Height)
    CL++
    CLG++
    CLC++
    #Auto-Height set
    if (Height == 0) {
        RU = rangerOffset(1000000, entity():pos(), vec(0,0, 1))
        Height = RU:position():z() - 2000
        HeightVec = vec(0,0, Height)
    }   
    
    
    #GetMapInfo
    Forward = rangerOffset(10000000, entity():pos() + HeightVec, vec(1, 0, 0))
    Behind = rangerOffset(10000000, entity():pos() + HeightVec, vec(-1, 0, 0))
    Left = rangerOffset(10000000, entity():pos() + HeightVec, vec(0, 1, 0))
    Right = rangerOffset(10000000, entity():pos() + HeightVec, vec(0, -1, 0))
    
    MapXF = Forward:position():x()
    MapXB = Behind:position():x()
    MapYF = Left:position():y()
    MapYB = Right:position():y()
    
    
    for(GridXC = 0, GridBase:x()) {GridPosX[GridXC, number] = 600 * GridXC}
    for(GridYC = 0, GridBase:y()) {GridPosY[GridYC, number] = 600 * GridYC}
        

    timer("CLG", 1)
    
    
    #print("--------------------------")
    #print("Generating clouds")
    
}

if (CLG & clk("CLG")) {
    GX = 0
    GY = 0
    
    if (CLC) {
        SpawnPos = vec(random(MapXF, MapXB), random(MapYF, MapYB), entity():pos():z() + randint(Height - 4000, Height))
        
        holoCreate(CL, SpawnPos)
        holoScale(CL, vec(50, 50, 10))
        holoMaterial(CL, "models/debug/debugwhite")
        holoColor(CL, vec(210, 210, 210))
        #holoShadow(CL, 1)
        holoAng(CL, ang(0,0,0))
        
        
        CloudPos[CL, vector] = SpawnPos
        CloudParent[CL, number] = CL
        
        
        for(GCloud = 1, GridBase:x() * GridBase:y()) {
            RandID = randint(500, 10000)
            Create = randint(1, 3)
            
            GX++
            if (GX >= GridPosX:count()) {
                GY++
                GX = 0
            }
            
            
            GridX = GridPosX[GX, number]
            GridY = GridPosY[GY, number]
            
            
            GBH = GridInfo
            Connect = 1
            
            
            if (Create > 1 & Connect) {
                holoCreate(RandID, CloudPos[CL, vector] + vec(GridX, GridY, 0), vec(50, 50, 10))
                holoMaterial(RandID, "models/debug/debugwhite")
                holoParent(RandID, CL)
                holoColor(RandID, vec(210, 210, 210))
                #holoShadow(RandID, CL)
                holoAng(RandID, ang(0,0,0))
                GridInfo:pushVector(vec(GridX, GridY, 1))
            }
            
            
        }
            
        
    timer("CLG", 1000)
        
        #MoveCloud = 1
    }
    if (CL > CloudsCount) {CLG = 0, MoveCloud = 1}
    CL++ 
}

interval(10)    

if (MoveCloud) {

    for(CloudMove = 1, CloudParent:count()) {
        CloudCPos = CloudPos[CloudMove, vector]
        CloudNPos = CloudCPos + vec(1.5, 0, 0)
        CloudPos[CloudMove, vector] = CloudNPos
    
        holoPos(CloudMove, CloudNPos)    

        if (CloudCPos:x() > MapXF + 500) {
            ZFit = random(Height - 4000, Height) + entity():pos():z()
            CloudPos[CloudMove, vector] = vec(MapXB, randint(MapYB+250, MapYF+-250), ZFit)
        }
    }
}
