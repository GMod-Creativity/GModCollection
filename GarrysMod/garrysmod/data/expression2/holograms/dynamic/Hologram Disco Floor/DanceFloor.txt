@name DanceFloor - By Jamza
@persist PX PY Ho LX LY R1 R2 G1 G2 B1 B2
interval(200)
runOnChat(1)

if(first()) {
    hint("If you need help type '!help' into chat", 10)
    EP = entity():pos()
    entity():setAlpha(0)
    Ho = 0
    R1=1,R2=255,G1=1,G2=255,B1=1,B2=255
    if(owner():lastSaid():sub(1,5)=="!size"){
        SArray=owner():lastSaid():sub(7):explode("x")
        PX = SArray[1,string]:toNumber()
        PY = SArray[2,string]:toNumber()
        if(PX>5) {hint("Warning: Any size above 5 may give unexepected results!",10)}
        if(PY>5) {hint("Warning: Any size above 5 may give unexepected results!",10)}
    } else {
        PX = 2
        PY = 2
    }
    for(X = 1,PX) {
        LX = (60 * X) - 60
        for(Y = 1,PY) {
            Ho = Ho + 1
            holoCreate(Ho)
            holoModel(Ho, "plane")
            holoMaterial(Ho, "models/debug/debugwhite")
            holoScale(Ho, vec(5,5,5))
            LY = (60 * Y) - 60
            holoPos(Ho, EP+vec(LX,LY,0))
        }
    }
}

for(I = 1,Ho) {
    R = random(R1,R2)
    G = random(G1,G2)
    B = random(B1,B2)
    holoColor(I, vec(R,G,B))
}

if(chatClk(owner())){
    if(lastSaid():sub(1,5)=="!size"){holoDeleteAll(), reset()}
    if(lastSaid()=="!help") {
        hint("--Holo Disco Floor by Jamza--", 10)
        hint("-- Commands --", 10)
        hint(" - !help = Displays this menu", 10)
        hint(" - !size [width]x[height] = Change size of dance floor", 10)
        hint(" - !red, !green, !blue, !all = change colour", 10)
    }
    if(lastSaid()=="!red"){R1=100,R2=255,G1=0,G2=0,B1=0,B2=0}
    if(lastSaid()=="!green"){R1=0,R2=0,G1=100,G2=255,B1=0,B2=0}
    if(lastSaid()=="!blue"){R1=0,R2=0,G1=0,G2=0,B1=100,B2=255}
    if(lastSaid()=="!all"){R1=1,R2=255,G1=1,G2=255,B1=1,B2=255}
}
