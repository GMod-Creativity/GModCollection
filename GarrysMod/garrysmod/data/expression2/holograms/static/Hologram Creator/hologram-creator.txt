@name Holo Creator
@persist I Model:string [Scale Color]:vector

runOnTick(1)
runOnChat(1)

if ( first() ) {
holoCreate(0)
Model = "cube"
Color = vec(255,255,255)
Scale = vec(1,1,1)
}
holoModel(0,Model)
holoMaterial(1,"models/debug/debugwhite")
if ( owner():weapon():type() == "weapon_crowbar" ) {
holoAlpha(0,150)
}

else{
    holoAlpha(0,0)
}

holoPos(0,round(owner():aimPos()/12)*12+vec(0,0,Scale:z()*6))
holoColor(0,Color)
holoScale(0,Scale)

if ( changed(owner():keyAttack1()) & owner():keyAttack1() & owner():weapon():type() == "weapon_crowbar" ) {
    I++
    holoCreate(I)
    holoPos(I,holoEntity(0):pos())
    holoAng(I,holoEntity(0):angles())
    holoModel(I,Model)
    holoColor(I,Color)
    holoScale(I,Scale)
    holoMaterial(I,"models/debug/debugwhite")
}   

if ( changed(owner():keyAttack2()) & owner():keyAttack2() ) {
    holoDelete(I)
    I--
}

if ( chatClk(owner()) & owner():lastSaid():explode(" "):string(1) == "/model" ) {
    Model = owner():lastSaid():explode(" "):string(2)
}

if ( chatClk(owner()) & owner():lastSaid():explode(" "):string(1) == "/col" ) {
Color = vec(owner():lastSaid():explode(" "):string(2):toNumber(),owner():lastSaid():explode(" "):string(3):toNumber(),owner():lastSaid():explode(" "):string(4):toNumber())
}

if ( chatClk(owner()) & owner():lastSaid():explode(" "):string(1) == "/scale" ) {
Scale = vec(owner():lastSaid():explode(" "):string(2):toNumber(),owner():lastSaid():explode(" "):string(3):toNumber(),owner():lastSaid():explode(" "):string(4):toNumber())
}
