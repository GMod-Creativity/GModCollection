@persist Dir:array DisableDown MaxDistance MinDistance HullSize HoloFlashI Speed:vector Color:vector FlashColor:vector
if(dupefinished()){reset()}
if(duped()){}else{
if(first()){
    
    
    Color=vec(0,255,0)
    FlashColor=vec(255,0,0)
    
    DisableDown=0
    
    MaxDistance=6000
    MinDistance=1000
    
    HullSize=min(144,250) #don't modify second value or holos will break
    
    rangerFilter(entity():getConstraints())
    rangerPersist(1)
    
    Dir=array(vec(0,0,1),vec(0,0,-1),vec(0,1,0),vec(0,-1,0),vec(1,0,0),vec(-1,0,0))
    local V = vec(0)
    for(I=1,6){ #so you can use foreach=Holos later instead
        holoCreate(I,V,vec(HullSize*0.1,HullSize*0.1,1),Dir[I,vector]:toAngle()+ang(-90,0,0),Color,"plane")
        holoAlpha(I,0)
        holoDisableShading(I,1)
    }
    holoCreate(7,V,vec(HullSize*0.2,HullSize*0.2,1),ang(0,0,0),Color,"hq_stube")
    holoCreate(8,V,vec(HullSize*0.2,HullSize*0.2,1),ang(0,0,0),Color,"hq_stube")
    holoCreate(9,V,vec(HullSize*0.2,HullSize*0.2,1),ang(0,0,90),Color,"hq_stube")
    holoCreate(10,V,vec(HullSize*0.2,HullSize*0.2,1),ang(0,0,90),Color,"hq_stube")
    holoCreate(11,V,vec(HullSize*0.2,HullSize*0.2,1),ang(90,0,0),Color,"hq_stube")
    holoCreate(12,V,vec(HullSize*0.2,HullSize*0.2,1),ang(90,0,0),Color,"hq_stube")
    for(I=7,12){
        holoAlpha(I,0)
        holoDisableShading(I,1)
    }
    function vector not(Vec:vector) {
        X=!Vec[1]
        Y=!Vec[2]
        Z=!Vec[3]
        return vec(X,Y,Z)
    }
    runOnTick(1)
}

HoloFlashI += 0.1
if(HoloFlashI==2){HoloFlashI=0}

Speed = Speed + entity():vel()*0.01 - Speed*0.01

foreach(J,D:vector=Dir){
    if(DisableDown & J==2){continue}
    local R = rangerOffsetHull(MaxDistance,entity():pos(),D,vec(HullSize))
    local Dist = R:distance()
    local InRange = R:hit()
    local InProx = Dist<MinDistance
    if(InRange){
        local A = 255*(MaxDistance-Dist)/MaxDistance
        holoAlpha(J,A)
        holoAlpha(J+6,A)
        local P = R:pos()+Speed*not(D)-D
        holoPos(J,P)
        holoPos(J+6,P)
        if(InProx){
            if(HoloFlashI>1){ holoAlpha(J,0) }
            holoColor(J,FlashColor)
            holoColor(J+6,FlashColor)
        }else{
            holoColor(J,Color)
            holoColor(J+6,Color)
        }
    }else{
        holoAlpha(J,0)
        holoAlpha(J+6,0)
    }
}
}
