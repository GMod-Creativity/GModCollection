@name Graphing Calc
@inputs Egp:wirelink
@outputs
@persist Equation:string X Y Cords:array
@trigger 
if(first()){
    Egp:egpClear()
    Egp:egpResolution(vec2(0,0),vec2(100,100))
    Egp:egpLine(1,vec2(50,100),vec2(50,0))
    Egp:egpLine(2,vec2(0,50),vec2(100,50))
    X = 0
}
function number solve(X){
    Max = Equation:find("^")
    return X^Equation:sub(Max+1, Equation:length()):toNumber()
}
function void draw(CordL:array){
    foreach(K,V:vector2=CordL){
        if(CordL[K-1,vector2] != vec2(0,0)){
            if(3 <= K <= 49){
                Egp:egpLine(K,CordL[K,vector2],CordL[K-1,vector2])
            }
            elseif(K == 50){
                Egp:egpLine(K,CordL[3,vector2],CordL[K,vector2])
            }
            elseif(50 < K <= 100){
                Egp:egpLine(K,CordL[K,vector2],CordL[K-1,vector2])
            }
        }       
    }   
}
runOnChat(1)
if(entity():owner():lastSaid():sub(0,7):lower() == "!graph " & chatClk(owner())){
    Equation = entity():owner():lastSaid():replace("!graph ","")
    if(Equation != ""){
        I=3
        while(I <= 100){
            if(I >= 50){
                if(X >= 50){
                    X = 0
                }
                Y = solve(X)
                X = X - 1
                Cords[I,vector2] = vec2(50+X,50-Y)
                I = I + 1
            }
            else{
            Y = solve(X)
            X = X + 1
            Cords[I,vector2] = vec2(50+X,50-Y)
            I = I + 1
            }
        }
        draw(Cords)
    }
}
