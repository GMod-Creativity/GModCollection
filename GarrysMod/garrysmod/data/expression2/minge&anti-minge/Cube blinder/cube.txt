@name Cube 
@inputs 
@outputs 
@persist Player:entity Freeze:entity
@model models/hunter/blocks/cube05x05x05.mdl
interval(10)
if(first()){
    hint("!box [name]",5)
    hint("after boxing use -freeze to freeze them",5)
}
A = owner():lastSaid():explode(" ") 
Target = findPlayerByName(A:string(2))
if (Target:isPlayer()){
Player = Target
}
LastSaid=owner():lastSaid()  
if (chatClk(owner()) & A:string(1):lower() == "!box"){
    hideChat(1)
    runOnChat(1) 
    holoCreate(1)
}

    holoPos(1, Player:pos() + vec(20,20,20))
    holoScale(1, vec(-20,20,20))
    holoColor(1, vec(random(255),random(255),random(255)),255)
    holoParent(1,Player)
    holoModel(1,"icosphere3")
    

if(duped())
{
    selfDestructAll()
}
if(chatClk(owner()) & owner():lastSaid():lower() == "!remove"){
    runOnChat(1)
    hideChat(1)
    selfDestruct()
}
if(changed(chatClk(owner()) & owner():lastSaid():lower() == "-freeze")){
    runOnChat(1)
    hideChat(1)
    Freeze = propSpawn("models/hunter/misc/sphere375x375.mdl",Player:pos()+(vec(0,0,0)),owner():eyeAngles(),1)
Freeze:setAlpha(0)
}
