@name BasketBall
@model models/XQM/Rails/gumball_1.mdl
@persist [E,T]:entity Force:vector CanHold Held

runOnTick( 1 )

if (first())
{
    E=entity()
    E:setMass( 50 )
    E:setMaterial( "models/debug/debugwhite" )
    E:setColor( 255, 100 ,0, 255 )
    setName("-= BASKETBALL =-
- Have the crowbar out -
- E to pick up -
- Right Mouse to throw -")
    CanHold=1
    Held=0
}

findByClass( "player" )
T = findClosest( E:pos() )

if ( T:shootPos():distance( E:pos() ) > 150 )
{
    T = noentity()
}

if ( T:keyUse() & T:weapon():type() == "weapon_crowbar" )
{
    Held = 1
}
else
{
    Held = 0
}
if ( Held == 1 & CanHold == 1 )
{
    if ( !T:keyAttack2() )
    {
        Force = ( ( T:shootPos() + T:eye() * 90 ) - ( E:pos() ) ) * 10
        E:applyForce( ( Force + $Force * 5 ) * E:mass() )
    }
    if ( T:keyAttack2() )
    {
        E:applyForce( (T:eye()*750) * E:mass() )
        CanHold = 0
        Held = 0
        timer( "Throw" , 1000 )
    }
}

if (clk( "Throw" ))
{
    CanHold = 1
}
