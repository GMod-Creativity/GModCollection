    function void drawStats(){
        E:egpText(250,"",vec2(32,3))
        E:egpText(251,"",vec2(32,23))
        E:egpText(252,"",vec2(135,10))
        E:egpSize(250,20)

    function void colorStats(Col:vector,ID){
        if(ID == 1){
            for(I=1,3){
                E:egpColor(249+I,Player1["Color",vector])
            }
        }else{
            for(I=1,3){
                E:egpColor(252+I,Player2["Color",vector])
            }
        }
    }

    function void reDrawStat(ID){
        if(ID == 1){
            E:egpSetText(250,"Lifes: " + Player1["Lifes",number] + "/" + MaxLifes)
            E:egpSetText(251,"Power: " + Player1["Power",number] + "/" + MaxPower)
            E:egpSetText(252,"Deaths: " + Player1["deaths",number])
        }else{
            E:egpSetText(253,"Lifes: " + Player2["Lifes",number] + "/" + MaxLifes)
            E:egpSetText(254,"Power: " + Player2["Power",number] + "/" + MaxPower)
            E:egpSetText(255,"Deaths: " + Player2["deaths",number])
        }
    }
    function void replaceWord(ID,CurWord:string,NewWord:string,IDmap){
        local Edit = ""
        Edit = Map[IDmap,string]:sub(ID,ID)
        Edit = Edit:replace(CurWord,NewWord)
        Edit = Map[IDmap,string]:sub(1,ID-1)+Edit+Map[IDmap,string]:sub(ID+1,Map[IDmap,string]:length())
        if(Edit:length() > Map[IDmap,string]:length()){
            Edit = Map[IDmap,string]:sub(1,Map[IDmap,string]:length())
        }
        Map[IDmap,string] = Edit
    }

    function number generateMap(){
        if(Num < 225){
            Num++
        }else{
            IsGenerated = 1
            drawStats()
            reDrawStat(1)
            reDrawStat(2)
            colorStats(Player1["Color",vector],1)
            colorStats(Player2["Color",vector],2)
            return 0
        }
        if(X == 15){
            X = 1
            Y++
        }else{
            X++
        }
        local Col = vec(255,255,255)
        Word = Map[1,string]:index(Num)

        if(Word == "0"){
            Col = vec(100,100,100)
        }elseif(Word == "4"){
            if(CurPlayers == 1){
                Col = Player1["Color",vector]
                Player1["PosID",number] = Num
                CurPlayers++
            }elseif(CurPlayers == 2){
                Col = Player2["Color",vector]
                Player2["PosID",number] = Num
                CurPlayers++
            }
        }
        if(Word == "1"){
            if(randint(0,WallsIntensity) != 0 &
            Num != Player1["PosID",number] &
            Num != Player1["PosID",number] + 1 &
            Num != Player1["PosID",number] + 15 &
            Num != 209 &
            Num != 209 - 1 &
            Num != 209 - 15 & WallsIntensity){
                Col = vec(50,50,50)
                replaceWord(Num,"1","2",1)
            }else{
                Col = vec(0,150,0)
            }
        }
        E:egpBox(Num,vec2(X*31+9,Y*31+61),vec2(30,30))
        E:egpColor(Num,Col)
    }

    function void gameOver(ID){
        if(ID == 1){
            #P1["Eject",number] = 1
            Player1["deaths",number] = Player1["deaths",number] + 1
            reDrawStat(1)
            soundPlay(3,0,"player/crit_death" + randint(1,5) + ".wav")
        }else{
            #P2["Eject",number] = 1
            Player2["deaths",number] = Player2["deaths",number] + 1
            reDrawStat(2)
            soundPlay(3,0,"player/crit_death" + randint(1,5) + ".wav")
        }
    }

    function void damage(ID){
        if(ID == 1){
            if(Player1["Lifes",number] < 2){
                gameOver(1)
            }else{
                soundPlay(3,0,"player/crit_received" + randint(1,3) + ".wav")
                Player1["Lifes",number] = Player1["Lifes",number] - 1
                reDrawStat(1)
            }
        }else{
            if(Player2["Lifes",number] < 2){
                gameOver(2)
            }else{
                soundPlay(3,0,"player/crit_received" + randint(1,3) + ".wav")
                Player2["Lifes",number] = Player2["Lifes",number] - 1
                reDrawStat(2)
            }
        }
    }

    function void playerSetPos(ID,X,Y){
        if(X & Y | X == -1 & Y == -1){
            return
        }
        local NewID = 0
        local PosID = 0
        local NewWord = ""

        if(ID == 1){
            NewID = Player1["PosID",number] + X + Y*15
            NewWord = Map[1,string]:index(NewID)
            if(NewWord == "0" | NewWord == "2" | NewWord == "3" | NewWord == "4"){
                return
            }
            PosID = Player1["PosID",number]
            replaceWord(Player1["PosID",number],"4","1",1)
            E:egpColor(Player1["PosID",number],vec(0,150,0))
            Player1["PosID",number] = NewID

            if(NewWord == "6"){
                if(Player1["Power",number] < MaxPower){
                    Player1["Power",number] = Player1["Power",number] + 1
                    reDrawStat(1)
                }
                soundPlay(4,0,"garrysmod/save_load" + randint(1,4) + ".wav")
                replaceWord(NewID,Map[1,string]:index(NewID),"1",1)
            }
            if(NewWord == "7"){
                if(Player1["Lifes",number] < MaxLifes){
                    Player1["Lifes",number] = Player1["Lifes",number] + 1
                }
                reDrawStat(1)
                soundPlay(4,0,"garrysmod/save_load" + randint(1,4) + ".wav")
                replaceWord(NewID,Map[1,string]:index(NewID),"1",1)
            }
            if(NewWord == "5"){
                damage(1)
            }
            replaceWord(NewID,NewWord,"4",1)
            E:egpColor(NewID,Player1["Color",vector])
        }else{
            NewID = Player2["PosID",number] + X + Y*15
            NewWord = Map[1,string]:index(NewID)
            if(NewWord == "0" | NewWord == "2" | NewWord == "3" | NewWord == "4"){
                return
            }
            PosID = Player2["PosID",number]
            replaceWord(Player2["PosID",number],"4","1",1)
            E:egpColor(Player2["PosID",number],vec(0,150,0))
            Player2["PosID",number] = NewID

            if(NewWord == "6"){
                if(Player2["Power",number] < MaxPower){
                    Player2["Power",number] = Player2["Power",number] + 1
                    reDrawStat(2)
                }
                soundPlay(4,0,"garrysmod/save_load" + randint(1,4) + ".wav")
                replaceWord(NewID,Map[1,string]:index(NewID),"1",1)
                replaceWord(NewID,Map[1,string]:index(NewID),"0",2)
            }
            if(NewWord == "7"){
                if(Player2["Lifes",number] < MaxLifes){
                    Player2["Lifes",number] = Player2["Lifes",number] + 1
                }
                reDrawStat(2)
                soundPlay(4,0,"garrysmod/save_load" + randint(1,4) + ".wav")
                replaceWord(NewID,Map[1,string]:index(NewID),"1",1)
                replaceWord(NewID,Map[1,string]:index(NewID),"0",2)
            }
            if(NewWord == "5"){
                damage(2)
            }
            replaceWord(NewID,NewWord,"4",1)
            E:egpColor(NewID,Player2["Color",vector])
        }

        soundPlay(0,0,"garrysmod/content_download.wav")

        if(Map[1,string]:index(PosID) == "3"){
            for(I=1,BombPos:count()){
                if(BombPos:count() & BombPos[I,number] == PosID){
                    replaceWord(PosID,"4","3",1)
                    E:egpColor(PosID,vec(255,255,0))
                }
            }
        }
    }

    function void createItem(ID){
        local MapString1 = Map[1,string]:index(ID)
        local MapString2 = Map[2,string]:index(ID)
        replaceWord(ID,MapString1,MapString2,1)
        replaceWord(ID,MapString2,"0",2)
        E:egpColor(ID,ItemsCol[MapString2:toNumber(),vector])
    }

    function void removeFireLeft(BombID){
        local PowerL = FireD[FireD:count(),vector4]:x()
        if(!PowerL){
            return
        }
        for(I=1,PowerL){
            local NextID = BombID-I
            local NextWord = Map[1,string]:index(NextID)
            if(NextWord == "0" | NextWord == "3"){
                return
            }
            replaceWord(NextID,NextWord,"1",1)
            E:egpColor(NextID,vec(0,150,0))
            FireT:removeNumber(BombID)
            FireP:removeNumber(BombID)
            if(Map[2,string]:index(NextID) == "6" | Map[2,string]:index(NextID) == "7"){
                createItem(NextID)
            }
        }
    }

    function void checkLeftSide(ID,Power){
        local Disable = 0
        local MaxID = 0
        for(I=1,Power){
            local NextWord = Map[1,string]:index(ID - I)
            if(NextWord == "3"){
                return
            }
            if(NextWord != "0"){
                if(Disable){
                    break
                }
                replaceWord(ID - I,NextWord,"5",1)
                E:egpColor(ID - I,vec(255,150,0))
                if(ID - I == Player1["PosID",number]){
                    damage(1)
                }
                if(ID - I == Player2["PosID",number]){
                    damage(2)
                }
                if(NextWord == "2"){
                    Disable = 1
                    if(randint(1,ItemIntensity) == 1){
                        replaceWord(ID - I,"0",randint(6,7):toString(),2)
                    }
                }
            }else{
                return
            }
            FireD[1,vector4] = vec4(I,0,0,0)
        }
    }

    function void checkUpSide(ID,Power){

        for(I=1,Power){

            FireD[1,vector4] = vec4(FireD[1,vector4]:x(),FireD[1,vector4]:y(),FireD[1,vector4]:z(),I)

    }

    function void explodeBomb(ID,Power){
        FireD:insertVector4(1,vec4(0,0,0,0))
        checkLeftSide(ID,Power)
        checkRightSide(ID,Power)
        checkDownSide(ID,Power)
        checkUpSide(ID,Power)

        FireT:insertNumber(1,curtime() + 0.8)
        FireP:insertNumber(1,ID)

        soundPlay(1,0,"ambient/explosions/explode_7.wav")
        replaceWord(ID,"3","5",1)
        E:egpColor(ID,vec(255,150,0))
        if(ID == Player1["PosID",number]){
            damage(1)
        }
        if(ID == Player2["PosID",number]){
            damage(2)
        }
    }

    function void placeBomb(Pos,Power,ID){
        local Word = Map[1,string]:index(Pos)
        if(Word == "3"){
            return
        }
        replaceWord(Pos,Word,"3",1)
        E:egpColor(Pos,vec(255,255,0))
        BombPower:insertNumber(1,Power)
        BombPos:insertNumber(1,Pos)
        BombOwner:insertNumber(1,ID)
        BombTimer:insertNumber(1,curtime() + BombExplodeTime)
        soundPlay(2,0,"buttons/combine_button1.wav")
    }
}

if(IsGenerated){
    if(P1["Active",number]){
        W1 = P1["W",number]
        S1 = P1["S",number]
        A1 = P1["A",number]
        D1 = P1["D",number]

        Space1 = P1["Space",number]

        if(W1&$W1|S1&$S1|A1&$A1|D1&$D1){
            playerSetPos(1,D1-A1,S1-W1)
        }

        if(Space1&$Space1){
            if(Player1["BombsAmount",number] < Player1["MaxBombs",number]){
                placeBomb(Player1["PosID",number],Player1["Power",number],1)
                Player1["BombsAmount",number] = Player1["BombsAmount",number] + 1
            }
        }
    }

    if(P2["Active",number]){
        W2 = P2["W",number]
        S2 = P2["S",number]
        A2 = P2["A",number]
        D2 = P2["D",number]

        Space2 = P2["Space",number]

        if(W2&$W2|S2&$S2|A2&$A2|D2&$D2){
            playerSetPos(2,D2-A2,S2-W2)
        }

        if(Space2&$Space2){
            if(Player2["BombsAmount",number] < Player2["MaxBombs",number]){
                placeBomb(Player2["PosID",number],Player2["Power",number],1)
                Player2["BombsAmount",number] = Player2["BombsAmount",number] + 1
            }
        }
    }

    if(curtime() >= BombTimer[BombTimer:count(),number] & BombTimer:count()){
        explodeBomb(BombPos[BombTimer:count(),number],BombPower[BombTimer:count(),number])

        if(BombOwner[BombTimer:count(),number] == 1){
            Player1["BombsAmount",number] = Player1["BombsAmount",number] - 1
        }else{
            Player2["BombsAmount",number] = Player2["BombsAmount",number] - 1
        }

        BombOwner():pop()
        BombTimer():pop()
        BombPos():pop()
        BombPower:pop()
    }

    if(curtime() >= FireT[FireT:count(),number] & FireT:count()){
        removeFireLeft(FireP[FireT:count(),number])
        removeFireRight(FireP[FireT:count(),number])
        removeFireUp(FireP[FireT:count(),number])
        removeFireDown(FireP[FireT:count(),number])
        FireD:pop()

        replaceWord(FireP[FireT:count(),number],"5","1",1)
        E:egpColor(FireP[FireT:count(),number],vec(0,150,0))
        E:egpColor(Player1["PosID",number],Player1["Color",vector])
        E:egpColor(Player2["PosID",number],Player2["Color",vector])
        FireT:removeNumber(FireP[FireT:count(),number])
        FireP:removeNumber(FireP[FireT:count(),number])

        FireT:pop()
        FireP:pop()
    }
    if(ops() > MaxOPS){
        MaxOPS = ops()
    }
}else{
    generateMap()
}
