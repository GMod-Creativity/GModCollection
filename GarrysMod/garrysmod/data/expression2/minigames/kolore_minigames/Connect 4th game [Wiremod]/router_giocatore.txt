@name router_giocatore
@inputs Turno:number Sx1:number Dx1:number Crea1:number Sx2:number Dx2:number Crea2:number
@outputs Sx:number Dx:number Crea:number
@persist 
@trigger 
@strict

if(Turno%2 == 0) {
    Sx = Sx1
    Dx = Dx1 
    Crea = Crea1
} else {
    Sx = Sx2
    Dx = Dx2
    Crea = Crea2
}
