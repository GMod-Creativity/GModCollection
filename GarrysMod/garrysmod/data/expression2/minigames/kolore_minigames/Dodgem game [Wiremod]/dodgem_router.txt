@name dodgem_router
@inputs Turn:number 
@inputs D1:number A1:number W1:number S1:number Cancel1:number Confirm1:number 
@inputs D2:number A2:number W2:number S2:number Cancel2:number Confirm2:number 
@outputs Dx:number Sx:number Fw:number Dw:number Cancel:number Confirm:number

@trigger D1 A1 W1 S1 Cancel1 Confirm1 D2 A2 W2 S2 Cancel2 Confirm2 
@persist 

if(Turn % 2 == 1) { #>= 0   #== 1
    #RED
    Dx = A1
    Sx = D1
    Fw = S1
    Dw = W1
    
    Cancel = Cancel1
    Confirm = Confirm1
} else {
    
    #BLU    
    Fw = A2
    Sx = S2
    Dx = W2
    Dw = D2
    Cancel = Cancel2
    Confirm = Confirm2
}
