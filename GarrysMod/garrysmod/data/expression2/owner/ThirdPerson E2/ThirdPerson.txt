@name ThirdPerson Camera
@inputs On Z
@outputs 
@persist 
@trigger 
@model models/hunter/plates/plate025x025.mdl
interval(50)
E=entity()
O=owner()

if(first()){
    #ifdef cameraCreate() #else selfDestruct() 
    #endif
    holoCreate(1)
    holoPos(1,O:toWorld(vec(-50,-25,75)))
    holoAng(1,O:toWorld(ang(10,10,0)))
    holoParent(1,O)
    holoAlpha(1,0)
    
    holoCreate(2)
    holoModel(2,"hqsphere")
    holoScale(2,vec(1,1,1)*0.5)
    
    cameraCreate(1,holoEntity(1):pos())
    cameraAng(1,holoEntity(1):angles())
    cameraParent(1,O)
    
    lightCreate(1)
    lightPos(1,O:aimPos())
    lightBrightness(1,30)
    lightColor(1,vec(200,0,0))
    lightParent(1,holoEntity(2))
}
holoPos(2,O:aimPos())

if(On){
    cameraToggle(1,1)
}else{cameraToggle(1,0)}

if(Z){
    lightToggle(1,1)
}else{lightToggle(1,0)}
