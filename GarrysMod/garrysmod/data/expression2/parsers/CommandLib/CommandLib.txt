@name CommandLib

if(first())
{
    function table readCommand(Command:string)
    { 
        local Command        = Command
        local Size           = Command:length()
        local CommandStruct  = table(
            "command" = "",
            "args"    = table()
        )
        local CommandArgs = table()
        
        local Buffer  = ""
        local Quot    = 0
        local QuotChr = ""
        local I       = 1
        
        for(I = 1, Size)
        {
            local Tok = Command[I]
                if(Quot)
            {
                if(Tok == QuotChr & Command[I-1] != "\\")
                {
                    Quot    = 0
                    QuotChr = ""
                    
                    CommandArgs:pushString(Buffer)
                    Buffer = ""
                }
                else
                {
                    Buffer += Tok
                }
            }
            elseif(Tok == "'" | Tok == "\"" | Tok == " " | Tok == "," | I == Size)
            {
                if(Tok == "'" | Tok == "\"")
                {
                    Quot    = 1
                    QuotChr = Tok
                }
                
                if(Buffer)
                {
                    CommandArgs:pushString(Buffer)
                    Buffer = ""   
                }
            }
            else
            {
                Buffer += Tok    
            }
        }
        
        CommandStruct["command", string] = CommandArgs:removeString(1)
        CommandStruct["args", table]     = CommandArgs
        
        return CommandStruct
    }
    
    
    printTable(readCommand("print \"hi\"0,'hello world'"))
}
