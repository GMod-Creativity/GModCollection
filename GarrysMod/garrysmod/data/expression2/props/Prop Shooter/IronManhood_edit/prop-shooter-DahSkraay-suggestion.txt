@name Prop Shooter
@persist [P E Prop PropE]:entity [MEMORY PropA]:array [Mode Sure SelectNo DoOnce]:number [Props]:string
@persist
@trigger 
runOnTick(1)
runOnChat(1)
#Variables
P = owner()
E = entity()
Ent = P:aimEntity()
LS = P:lastSaid()
LSS = LS:explode(" ")
LSSS = LSS
LSSSS = LSS
#First
if(first()){
    hint("Type /help for instructions",10)
    SelectNo = 1
    Props = "models/props_c17/oildrum001_explosive.mdl"
    hint("The default prop is models/props_c17/oildrum001_explosive.mdl on number 1",10)
    Mode = 1
    Index = 0
    DoOnce = 1
    timer("mode",10)
}
#Array
if(inrange(Mode,1,50) & DoOnce){
    MEMORY[Mode, string] = Props
    DoOnce = 0
}
if(clk("mode")){
    Mode = 0
}
#/propNo command
if(chatClk(owner())){
    if(LSSS:string(1):lower() == "/propno"){
        hideChat(1)
        Mode = LSSS:string(2):toNumber()
        hint("Selected array "+Mode:toString()+" to save to.",10)
}
}
#/prop command
if(chatClk(owner())){
    if(LSSSS:string(1):lower() == "/prop"){
        Props = LSSSS:string(2)
        hint("Saved prop "+Props+" on number "+Mode:toString(),10)
        hideChat(1)
        timer("mode",100)
        DoOnce = 1
    }
}
#/select command
if(chatClk(owner())){
    if(LSS:string(1):lower() == "/select"){
        hideChat(1)
        SelectNo = LSS:string(2):toNumber()
        hint("Prop Selected: "+MEMORY[SelectNo, string],10)
}
}
#/clear command
if(chatClk(owner())){
    if(LS:lower() == "/clear"&!Sure){
        hideChat(1)
        hint("Are you sure you want to clear all saved props? type /clear again",10)
        timer("Sure",100)
}
}
if(clk("Sure")){
    Sure = 1
}
if(Sure){
    if(chatClk(owner())){
        if(LS:lower() == "/clear"){
            hideChat(1)
            Sure = 0
            MEMORY:clear()
            hint("Prop array cleared!",10)
} 
}
}
#/help command
if(chatClk(owner())){
    if(LS:lower() == "/help"){
        hideChat(1)
        print("type /clear to clear your memorized props.")
        print("type /remove to remove all props.")
        print("type /propno [number from 1-50] to select that number to record the prop.")
        print("type /prop [Prop] to record that prop to that number.")
        print("type /select [number from 1-50] to select that number.")
        print("crouch and right click to shoot the selected prop.")
        print("to select a prop, go to prop menu then right click and click copy. After that, in chat,")
        print("type /propno [number from] 1-50 to select a number to save your prop to.")
        print("to find the prop, in the prop menu (Q) right click a prop then select copy.")
        print("then type /prop [CRTL+V] to save that prop to the number set by /ptopno")
        print("then crouch and right click to shoot the prop")
        print("SCROLL UP FOR FULL INSTRUCTIONS.")
}
}
#Shoot the prop
if(changed(P:keyAttack2())&P:keyAttack2()&P:keyDuck()){
    Prop = propSpawn(MEMORY[SelectNo, string], P:pos()+vec(0,0,60), P:eyeAngles(), 0)
    PropA:pushEntity(Prop)
    PropA:count()
    PropE:propGravity(0)
    PropE = PropA:entity(PropA:count())
}
#Apply force
PropE:applyForce(PropE:forward()*10000000)
#Deletion
if(chatClk(owner())){
    if(LS:lower() == "/remove"){
        hideChat(1)
        PropA:propDelete()
        hint("Props Deleted!",10)
}
}
