@name deadmau5 Head - Made by Raubana
@persist Stage Color Teeth

###########
VisibleToOwner=1 #0
InitColor=9 #0-9
AttachToOwner=1 #1 raub ana
ColorSwitchDelay=1 #in minutes
###########



O=owner()

if(first()){
    Stage=0
}


if(Stage==0){
    #inv box
    holoCreate(1)
    holoPos(1,vec(0,-3,-6))
    holoAng(1,ang(20,0,0))
    holoScale(1,vec(0.1,0.1,0.1))
    holoAlpha(1,150)
    
}elseif(Stage==1){
    
    #top head RAUBANA
    holoCreate(2,vec(0,0,0),vec(1.5,1.5,1.5),ang(0,0,0))
    holoModel(2,"hq_dome")
    holoParent(2,1)
    
    #bottom head
    holoCreate(3,vec(0,0,0),vec(1.5,1.5,1.5),ang(130,90,0))
    holoModel(3,"hq_dome")
    holoParent(3,1)
    
    
    #mouth
    holoCreate(4,vec(0,0,0),vec(1.35,1.35,1.35),ang(225,90,0),vec(255,255,255))
    holoModel(4,"hq_dome")
    holoParent(4,1)
    
    #left eye
    holoCreate(5,vec(-3.75,-5.25,3.75),vec(0.45,0.45,0.45),ang(0,0,0),vec(255,255,255))
    holoModel(5,"hq_sphere")#RAUbana
    holoParent(5,1)
    holoMaterial(5,"phoenix_storms/fender_white")
    
    #right eye
    holoCreate(6,vec(3.75,-5.25,3.75),vec(0.45,0.45,0.45),ang(0,0,0),vec(255,255,255))
    holoModel(6,"hq_sphere")
    holoParent(6,1)
    holoMaterial(6,"phoenix_storms/fender_white")
    
    #left ear
    holoCreate(7,vec(12,0,10.5),vec(1.5,1.3,0.1),ang(-30,-85+90,75))#r a u b a n a
    holoModel(7,"hq_rcylinder_thin")
    holoParent(7,1)
    
    #right ear
    holoCreate(8,vec(-12,0,10.5),vec(1.5,1.3,0.1),ang(30,-97+90,75))
    holoModel(8,"hq_rcylinder_thin")
    holoParent(8,1)
    
    #crosses RAUBANA(left eye)
    holoCreate(9,vec(-4.25,-6,4.25),vec(0.3,0.3,0.3),ang(40,-45,-25),vec(255,255,255))
    holoModel(9,"torus2")
    holoParent(9,1)
    holoColor(9,vec(0,0,0))
    
    holoCreate(10,vec(-4.25,-6,4.25),vec(0.3,0.3,0.3),ang(145,-90,-35),vec(255,255,255))
    holoModel(10,"torus2")
    holoParent(10,1)
    holoColor(10,vec(0,0,0))
    
    #crosses (right eye)
    holoCreate(11,vec(4.25,-6,4.25),vec(0.3,0.3,0.3),ang(-40,45,-25),vec(255,255,255))
    holoModel(11,"torus2")
    holoParent(11,1)
    holoColor(11,vec(0,0,0))
    
    holoCreate(12,vec(4.25,-6,4.25),vec(0.3,0.3,0.3),ang(-145,90,-35),vec(255,255,255))
    holoModel(12,"torus2")
    holoParent(12,1)
    holoColor(12,vec(0,0,0))
    
    
    #teeth (top)
    Teeth=8
    for(X=0,Teeth){
        A=180*(X/Teeth)
        Pos=vec(8.1*cos(A),-7*sin(A)-1.5,-1)
        holoCreate(X+1+12)
        holoModel(X+1+12,"cone")
        holoAng(X+1+12,ang(180,260+(100-260)*(X/Teeth),0))
        holoPos(X+1+12,Pos)
        holoScale(X+1+12,vec(0.35,0.01,0.3))
        holoColor(X+1+12,vec(0,0,0))
        holoParent(X+1+12,1)
    }
    #raubana
}elseif(Stage==2){
    if(AttachToOwner==1){
        holoPos(1,O:attachmentPos("mouth"))
        holoAng(1,O:attachmentAng("mouth")+ang(0,90,0))
        holoParentAttachment(1,O,"mouth")
    }else{
        holoPos(1,entity():pos()+vec(0,0,25))
        holoParent(1,entity())
    }
    
}elseif(Stage==3){
    timer("color",10)
    Color=InitColor
    for(X=2,1+12+Teeth){
        holoVisible(X,O,VisibleToOwner)
    }
}

if(Stage!=4){
    interval(100)
    Stage+=1
}



#Changes the color of the head
if(clk("color")){
    timer("color",1000*60*ColorSwitchDelay)#this is how often the color changes
    #raubana
    Ce=vec(255,255,255) #color of the eyes
    Cm=vec(255,255,255) #color of the mouth
    Cr=0 #show the crosses on the eyes
    Gm=0 #the mouth glows
    Ge=1 #the eyes glow
    E=vec(-1,-1,-1) #color of the ears.
    M="models/debug/debugwhite" #the material the head will be
    Me="" #the material the ears will be
    ME="" #the material the EYES will be
    T=0
    
    print(Color:toString())
    if(Color==0){#Grey
        C=vec(127,127,127)
    }elseif(Color==1){#Red
        C=vec(255,0,0)
    }elseif(Color==2){#Green
        C=vec(0,127,0)
    }elseif(Color==3){#Yellow
        E=vec(20,20,20)
        C=vec(240,240,0)
        Cr=1
    }elseif(Color==4){#Blue w/ Sparkles
        C=vec(127,127,255)
        M="phoenix_storms/wire/pcb_blue"
        Cr=1
    }elseif(Color==5){#Black w/ black eyes & read mauth
        C=vec(20,20,20)
        Ce=vec(1,1,1)
        Cm=vec(255,0,0)
        T=1
        Ge=0
        Gm=1
    }elseif(Color==6){#Blue
        C=vec(0,0,255)
    }elseif(Color==7){#Purple
        C=vec(255,0,200)
    }elseif(Color==8){#Disco Head
        C=vec(255,255,255)
        M="debug/env_cubemap_model"
        Me="models/debug/debugwhite"
        Cr=1
        E=vec(20,20,20)
    }elseif(Color==9){#Mirror Head
        C=vec(255,255,255)
        M="debug/env_cubemap_model"
        Me="debug/env_cubemap_model"
        ME="debug/env_cubemap_model"
    }
    #raubana
    #changes the colors
    holoColor(2,C)
    holoColor(3,C)
    holoColor(4,Cm)
    holoColor(5,Ce)
    holoColor(6,Ce)
    if(E:x()==-1){
        holoColor(7,C)
        holoColor(8,C)
    }else{
        holoColor(7,E)
        holoColor(8,E)
    }
    
    #sets the materials
    holoMaterial(2,M)
    holoMaterial(3,M)
    if(Me==""){
        holoMaterial(7,M)
        holoMaterial(8,M)
    }else{
        holoMaterial(7,Me)
        holoMaterial(8,Me)
    }
    
    if(ME==""){
        holoMaterial(5,"models/shiny")
        holoMaterial(6,"models/shiny")
    }else{
        holoMaterial(5,ME)
        holoMaterial(6,ME)
    }
    #sets if the teeth are visible
    for(X=12+1,12+1+Teeth){
        holoAlpha(X,T*255)
    }
    
    #sets if the crosses are visible
    for(X=9,12){
        holoAlpha(X,Cr*255)
    }
    # r a uba ana
    #sets eye glow
    holoDisableShading(5,Ge)
    holoDisableShading(6,Ge)
    
    #sets mouth glow
    holoDisableShading(4,Gm)
    
    Color+=1
    if(Color>9){Color=0}
}
