@name Piano by Someguynamedpie(Multiangular Supercolliding Super piano)
@persist Timeout OldIdx Spawned [O E]:entity
if(first()){
    O=owner()
    E=entity()
    while(holoCanCreate()){
        Spawned++
        holoCreate(Spawned)
        holoPos(Spawned,E:pos()+E:right()*Spawned*-10)
        holoScale(Spawned,vec(1,0.6,0.2))
        runOnTick(1)
        holoParent(Spawned,E)
        holoAng(Spawned,entity():angles())
    }
}

findIncludePlayerProps(O)
findIncludeClass("gmod_wire_hologram")

findInCone(O:shootPos(),O:eye(),500,2.5)
F=find()
holoColor(OldIdx,vec(255,255,255))
Idx=holoIndex(F)
holoColor(Idx,vec(0,0,255))
OldIdx=Idx
if(O:keyUse()&!Timeout){
    E:soundPlay(Idx,0.4,"synth/square.wav")
    soundPitch(Idx,Idx*(200/Spawned))
    timer("Timeout",400)
    Timeout=1
}
if(clk("Timeout")){
    Timeout=0
}
