### Turret Spin ##################################################  ###################################
Angle += Gat
holoAng(13, E:toWorld(ang(-Angle,0,90)))
holoAng(22, E:toWorld(ang(Angle,0,90)))

### Blade Spin ##################################################  ####################################
if (Active) {Spin += 0.17} else {Spin -= 0.07}
Spin = clamp(Spin, 0, 30)
Rotation += Spin
if (Rotation>36000) {Rotation=0}
holoAng(1, E:toWorld(ang(0,Rotation,0)))
holoAng(4, E:toWorld(ang(0,0,Rotation)))

### Pod Controller ##################################################  ################################
W           = Pod["W",          number]
A           = Pod["A",          number]
S           = Pod["S",          number]
D           = Pod["D",          number]
R           = Pod["R",          number]
Shift       = Pod["Shift",      number]
Alt         = Pod["Alt",        number]
Space       = Pod["Space",      number]
Active      = Pod["Active",     number]
M1          = Pod["Mouse1",     number]
M2          = Pod["Mouse2",     number]
Cam:setNumber("Activated", Active)
Driver = Chair:driver()
if (changed(Driver)&Driver!=noentity()) {
    Chair:hintDriver("Hello " + Driver:name() + "! The controls are:", 15000)
    Chair:hintDriver("WASD - Forward/Back and Yaw", 15000)
    Chair:hintDriver("Mouse buttons - Strafe", 15000)
    Chair:hintDriver("Space - Ascend", 15000)
    Chair:hintDriver("Alt - Descend", 15000)
    Chair:hintDriver("Shift - Move Fast", 15000)
    Chair:hintDriver("R - Fire guns", 15000)
}

### Sound ##################################################  #########################################
if (changed(Active)) {
    if (Active) {
        E:soundPlay(0,0,"/vehicles/airboat/fan_blade_idle_loop1.wav")
        E:soundPlay(1,0,"buttons/button17.wav")
        E:soundPlay(4,0,"ambient/machines/fluorescent_hum_1.wav")
        soundVolume(4, 0.5)
    } 
    else        {
        soundPurge()
        E:soundPlay(1,0,"buttons/button16.wav")
    }
}
if (!Active & Spin!=0) {SpinningDown = 1} else {SpinningDown=0}
if (changed(SpinningDown)) {
    if (SpinningDown)   {E:soundPlay(2,0,"/fans/fan4.wav"), soundVolume(2, 0.2)} 
    else                {soundStop(2)}
}
soundPitch(0, Spin + sqrt(E:vel():length()/2))
soundPitch(2, sqrt(Spin)*45)
soundPitch(3, 250)
soundPitch(4, Gat)

if (changed(Gat>25)) {
    if (Gat>25) {
        E:soundPlay(3,0,"/weapons/airboat/airboat_gun_loop2.wav")
        Fire = 1
    } else {
        soundStop(3)
        Fire = 0
    }
}

### Rangers ##################################################  #######################################
rangerFilter(E)
rangerHitWater(1)
RD1 = rangerOffset(E:pos(), E:pos():setZ(-100000))
RD2 = rangerOffset(E:toWorld(vec(0,-30,0)), E:pos():setZ(10000))

### Smoothers ##################################################  #####################################
#Guns
if     (R) {Gat += 0.3}
else       {Gat -= Gat/40}
Gat = clamp(Gat, 0, 30)

#Turning
if     (A) {T += 0.03}
elseif (D) {T -= 0.03}
else       {T -= T/20}
T = clamp(T, -1, 1)
 
#Accelerating
if     (W & !S)                 {Acc += Shift ? 0.1 : 0.05}
elseif (S & !W & Acc>0.7)       {Acc -= Acc/5}
elseif (S & !W)                 {Acc -= 0.04}
else                            {Acc -= Acc/50}
if (!Shift & Acc>3.5)       {Acc -= 0.1} 
Acc = clamp(Acc, -5 , 10)

#Strafing
if     (M1 & !M2)               {Str += Shift ? 0.1 : 0.05}
elseif (M2 & !M1)               {Str -= Shift ? 0.1 : 0.05}
else                            {Str -= Str/30}
if (!Shift & abs(Str)>3.5)  {Str -= Str/25} 
Str = clamp(Str, -10 , 10)

#Height
if     (Space & !Alt)           {Z += Shift ? 0.4 : 0.15}
elseif (Alt & !Space)           {Z -= Shift ? 0.4 : 0.15}
else                            {Z -= Z/15}
if (!Shift & Z>3.5)         {Z -= 0.2} 
Z = clamp(Z, -15 , 15)
if (Active) {H += Z} else {H -= clamp((20-Spin), 0, 20)}
H = clamp(H, RD1:pos():z() + (Active ? 40 : 22), RD2:hit() ? RD2:pos():z()-50 : 10000)

### Position and Angle ##################################################  ############################
if (!E:isUnderWater()) {
    TarPos = E:pos():setZ(H) + E:right():setZ(0):normalized()*Acc*30 + E:forward():setZ(0):normalized()*Str*20
    TarAng = ang(Str*4+T*26, E:angles():yaw() + (3+(E:vel():length()/4000))*T, Acc*1.8)
    
       E:applyForce(((TarPos - E:pos())    * 9  - E:vel())    * E:mass())
    E:applyAngForce(((TarAng - E:angles()) * 40 - E:angVel()) * E:mass()*4)
}
