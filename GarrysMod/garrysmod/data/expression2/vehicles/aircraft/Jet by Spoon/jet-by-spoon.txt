@name Jet by Spoon God
@inputs W:wirelink
@persist [E J P V]:entity [B C G H M O S]
if(tickClk()){
    if(M){
        if(M>0){
            D=V:driver()
            if(D){
                if(O){
                    if(S&W:number("Space")){
                        S-=1
                    }else{
                        if(B==2){
                            Temp=V:angles():roll()/2
                            if(Temp){
                                if(Temp<0){
                                    Temp*=-1
                                }
                                S+=(W:number("Shift")-W:number("Alt"))/Temp
                            }else{
                                S+=(W:number("Shift")-W:number("Alt"))
                            }
                        }else{
                            S+=(W:number("Shift")-W:number("Alt"))*0.5
                        }
                        S=clamp(S,0,150)
                    }
                    SS=sin(S)
                    Temp=clamp(round(S/5),1,30)
                    if(Temp==10){
                        TS1="[---------(+)--------------------] "
                    }elseif(Temp==30){
                        TS1="[---------+-------------------(-)] "
                    }else{
                        TempS="---------+--------------------"
                        TS1="["+TempS:left(Temp-1)+"(-)"+TempS:right(30-Temp)+"] "
                    }
                    TS1+=round(clamp(2*S/3,0,100))+"%"
                    Temp=clamp(round(SS*30),1,30)
                    TS2="["+"-":repeat(Temp-1)+"(-)"+"-":repeat(30-Temp)+"] "+round(clamp(SS*100,0,100))+"%"
                    if(W:number("Space")){
                        SS=2
                    }
                    FP=J:pos()
                    MS=S*500
                    U=V:up()
                    if(S>50){
                        B=1
                        F=V:right()
                        V:printDriver(4,TS1+"\n"+TS2)
                    }else{
                        F=V:right():setZ(-1)
                        if(B){
                            SS=0.75
                            if(B==1){
                                B=2
                                S=0
                            }
                            if(J:vel():length()<5){
                                B=0
                            }
                            V:printDriver(4,TS1+"\n"+TS2+"\n- - - - - - STALL - - - - - -")
                        }else{
                            V:printDriver(4,TS1+"\n"+TS2)
                        }
                    }
                    J:applyForce(F*MS)
                    V:applyForce(F*100)
                    UD=W:number("S")-W:number("W")
                    if(UD){
                        J:applyOffsetForce(U*UD*SS*5000,FP+F*100)
                    }
                    LR=W:number("A")-W:number("D")
                    if(LR){
                        J:applyOffsetForce(U*SS*5000*LR,FP-V:forward()*50)
                        J:applyOffsetForce(U*SS*-5000*LR,FP-V:forward()*-50)
                        #J:applyTorque(vec(LR*MS*30,0,0))
                    }
                }else{
                    O=1
                    V:setAlpha(0)
                }
            }elseif(O){
                S=0
                O=0
                B=0
                V:setAlpha(255)
            }
        }else{
            if(W:entity("Entity")==V){
                M=J:mass()
                J:setMass(1000)
                P:setAlpha(0)
                S=0
                holoDelete(1)
                holoDelete(2)
            }else{
                S+=5
                if(S>360){
                    S-=360
                }
                TN=sin(S)*10
                holoPos(1,P:pos()+vec(0,0,35+TN))
                holoPos(2,V:pos()+vec(0,0,100+TN))
            }
        }
    }elseif(W:entity()==P){
        E:setAlpha(0)
        E:setMass(0.1)
        P:setMass(0.1)
        V:setMass(100)
        if(W:entity("Entity")==V){
            M=J:mass()
            runOnLast(1)
            J:setMass(1000)
            P:setAlpha(0)
            S=0
            holoDelete(1)
            holoDelete(2)
        }else{
            M=-1
            if(concmd("tool_wire_adv_pod")){
                printColor(vec(0,255,0)," (!) ",vec(255,255,255),"Link the Advanced Pod Controller to the seat.")
            }else{
                printColor(vec(0,255,0)," (!) ",vec(255,255,255),"Link the Advanced Pod Controller to the seat using the Advanced Pod Controller tool.")
            }
        }
    }else{
        S+=5
        if(S>360){
            S-=360
        }
        TV=vec(0,0,35+sin(S)*10)
        holoPos(1,P:pos()+TV)
        holoPos(2,E:pos()+TV)
    }
}elseif(first()){
    E=entity()
    J=E:isWeldedTo()
    if(J){
        if(J:isVehicle()){
            printColor(vec(255,0,0)," (!) ",vec(255,255,255),"You must spawn the E2 on the prop model, not the seat.")
            selfDestruct()
        }else{
            R=J:getConstraints()
            N=R:count()
            for(X=1,N){
                E=R[X,entity]
                if(E:isVehicle()&!V){
                    V=E
                }elseif(E:type()=="gmod_wire_adv_pod"){
                    P=E
                }
            }
            if(V){
                if(P){
                    SV=vec(2,2,2)
                    A=ang(180,0,0)
                    Col=vec(0,255,0)
                    E=entity()
                    runOnTick(1)
                    holoCreate(1,SV,SV,A,Col,"hqcone")
                    holoParent(1,P)
                    holoCreate(2,SV,SV,A,Col,"hqcone")
                    holoParent(2,E)
                    C=10
                    if(concmd("tool_wire_adv")){
                        printColor(vec(0,255,0)," (!) ",vec(255,255,255),"Wirelink the E2 to the Advanced Pod Controller.")
                    }else{
                        printColor(vec(0,255,0)," (!) ",vec(255,255,255),"Wirelink the E2 to the Advanced Pod Controller using the Wire Advanced Tool.")
                    }
                }else{
                    concmd("tool_wire_adv_pod")
                    printColor(vec(255,0,0)," (!) ",vec(255,255,255),"You must place an Advanced Pod Controller.")
                    selfDestruct()
                }
            }elseif(P){
                printColor(vec(255,0,0)," (!) ",vec(255,255,255),"You must place a driver seat.")
                selfDestruct()
            }else{
                concmd("tool_wire_adv_pod")
                printColor(vec(255,0,0)," (!) ",vec(255,255,255),"You must place a driver seat and an Advanced Pod Controller.")
                selfDestruct()
            }
        }
    }else{
        printColor(vec(255,0,0)," (!) ",vec(255,255,255),"You must place the E2 on a prop!")
        selfDestruct()
    }
}elseif(last()){
    P:setAlpha(255)
    V:setAlpha(255)
    if(->W){
        J:setMass(M)
    }
}elseif(duped()){
    reset()
}
