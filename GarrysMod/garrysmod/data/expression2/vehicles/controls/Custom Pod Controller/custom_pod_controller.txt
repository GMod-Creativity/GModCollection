@name Custom Pod Controller
@inputs [Chair]:entity
@persist [Driver]:entity [Outputs]:table [Debug]:number
@model models/jaanus/wiretool/wiretool_range.mdl

# ADD or REMOVE outputs for keys here
@outputs Mouse_Left E

# The output name MUST match (case insensitive) the KEYNAME of the KEY being pressed
# If you do not know the KEYNAME, enable debugging (line 17), get in the chair, and press the key


if ( first() | dupefinished() ) {
    # DEBUGGING
    # Enable to print KEYNAMES and other debug info
    
    Debug = 0 # Set to 1 (on) or 0 (off)
    
    #########
    
    if ( !->Chair ) { print("Wire this E2 to a pod!") }
    
    Outputs = table()
    
    foreach ( K, V:string = entity():getWireOutputs() ) {
        Outputs[V:lower(), string] = V
    }
    
    interval(15)
}
elseif ( Driver ) {
    local Key   = keyClkPressed()
    local Press = keyClk(keyClk()) > 0
    
    ioSetOutput(Outputs[Key, string], Press)
    
    if ( Debug ) { print(Key + (Press ? " pressed" : " released")) }
    if ( !Chair:driver() ) {
        # The driver has exited the chair
        if ( Debug ) { print("Driver exited", Driver) }
    
        # Stop listening to his key presses
        runOnKeys(Driver, 0) 
        
        # Set all outputs to 0
        foreach ( K, V:string = Outputs ) {
            ioSetOutput(V, 0)
        }
        
        Driver = noentity()
        
        entity():setColor(vec(255, 160, 0))

        interval(15)
    }
}
else {
    Driver = Chair:driver()

    if ( Driver ) {
        if ( Debug ) { print("Driver entered", Driver) }
        
        entity():setColor(vec(0, 255, 0))
        runOnKeys(Driver, 1)
    }
    else {
        interval(100)
    }
}
