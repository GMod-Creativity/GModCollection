@name Tarif's Protection From Props
@persist Vector:vector [E O Target]:entity Mode:string Radius
@model models/hunter/blocks/cube1x1x1.mdl
runOnChat(1)
runOnTick(1)
 
if (first()) {
E = entity()
E:setMass(50000)
Mode = "follow"
O = owner()
 
holoCreate(1)
holoPos(1,O:pos())
holoParent(1,O)
holoModel(1,"hqsphere2")
holoColor(1,vec(0,229,255),40)
holoScale(1,vec(0,0,0))
}
 
#MODES
 
 
if (Mode == "follow") {
        Vector = O:pos() + vec(0,0,150)
        entity():setColor(50,255,0,255)
    }
   
if (Mode == "shield") {
        if (clk("update")) {
               entity():setColor(238,225,0,255)
                Vector = O:pos() + vec(0,0,150)
                findExcludePlayerProps(O)
                findExcludeClass("player")
                findInSphere(owner():shootPos(),Radius)
                findIncludeClass("prop_physics")
                findIncludeClass("prop_vehicle")
                findSortByDistance(owner():shootPos())
 
                Targets = findToArray()
                Target = Targets[1,entity]
                timer("update",1)
        }
 
if (Target) {
entity():setColor(255,0,0,255)
Vector = Target:pos() + vec(0,0,3)
}
 
 
}
 
#CHAT COMMANDS
 
if (chatClk(O)) {
        if(O:lastSaid() == "-follow") {Mode = "follow"}
}
 
if(O:lastSaid():explode(" ")[1,string] == "-protect"){
Radius = O:lastSaid():explode(" ")[2,string]:toNumber()
        Mode = "shield"  
        timer("update",1)
}
 
if(O:lastSaid() == "-protect off") {
    Mode = "follow"
    holoScale(1,vec(0,0,0))
}
 
if (O:lastSaid() == "-protect sphere") {
holoScale(1,-vec(200,200,200)/2)
Radius = 320
Mode = "shield"
timer("update",1)
}
 
#OTHER STUFF
 
 
#THE APPLYFORCE
 
E:applyForce(((Vector - E:pos())*90 - E:vel())*E:mass())
E:applyAngForce((ang(0,0,0)-E:angles())*E:mass()*5 - E:angVel()*E:mass())
