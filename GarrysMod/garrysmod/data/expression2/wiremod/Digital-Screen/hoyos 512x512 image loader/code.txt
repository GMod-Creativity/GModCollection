@inputs DS:wirelink
@outputs FIR FN:string I1 CNT H:array S:string I2 I3
@outputs CNT2 FN2:string CNT3 DBG CNT4 L:string 
@persist FIR FN:string I1 CNT H:array S:string I2 I3
@persist CNT2 FN2:string CNT3 CNT4 L:string S2:string
@persist
@trigger
@model models/cheeze/wires/cpu2.mdl 
runOnChat(1)
interval(500)
if(first() | duped())
{
    CNT2=0
    CNT3=0
    CNT = 1
}
if(chatClk(owner()) & lastSaid():explode(" "):string(1) == "draw")
{
    CNT2=0
    CNT3=0
    CNT = 1
    FN = lastSaid():explode(" "):string(2)+"_001.txt"
    FN2 = lastSaid():explode(" "):string(2)
    S = "_001.txt"
    if(fileCanLoad())
    {   
        fileLoad(FN)
        print("loading..")
        runOnFile(1)
    }
    else
    {
        print("cant load that file now, plz try again")
    }
}
if(chatClk(owner()) & lastSaid() == "reload")
{
    if(fileCanLoad())
    {   
        fileLoad(FN2+S)
        print("loading..")
        runOnFile(1)
    }
    else
    {
        print("cant load that file now, plz try again")
    }
}
    
I1 = fileLoaded(FN2+S)
if(changed(I1) & !FIR & !first() & !duped())
{
    FIR++
    print("the file "+FN+" is completely loaded")
}
if(CNT >= 10)
{
    S = "_0"+CNT:toString()+".txt"
}
if(I1 & CNT4<512*512)
{
    timer("1",0)
    timer("2",0)
    timer("3",0)
    timer("4",0)
    timer("5",0)
    timer("6",0)
    timer("8",0)
    timer("7",0)
    CNT2+=3
    CNT3+=3
    CNT4++
    S2 = L+fileRead(FN2+S)
    DS[CNT4] = (S2:sub(CNT3-2,CNT3)+"000"):toNumber()
    DBG = (S2:sub(CNT3-2,CNT3)+"000"):toNumber()
    if(CNT3>(fileRead(FN):length()-fileRead(FN):length()%3))
    {
        CNT3 = 1
        print((CNT4/(512*512)*100):toString()+"%")
        L = fileRead(FN):sub(fileRead(FN):length()-fileRead(FN):length()%3,fileRead(FN):length())
        CNT++
        I1 = fileLoaded(FN2+S)
        if(fileCanLoad())
        {
            if(CNT >= 10)
            {
                S = "_0"+CNT:toString()+".txt"
            }
            else
            {
                S = "_00"+CNT:toString()+".txt"
            }   
            print("loading..")
            fileLoad(FN2+S)
        }
    }
}
