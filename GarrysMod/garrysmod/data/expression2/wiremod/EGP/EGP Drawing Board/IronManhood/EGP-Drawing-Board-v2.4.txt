    function pointer()
    {
            EGP:egpUnParent(19)
            EGP:egpRemove(19)
            EGP:egpLineStrip(18,vec2(0,6),vec2(6,0),vec2(-6,0),vec2(0,-6),vec2(0,6))
            EGP:egpColor(18,vec(255,0,0))
            EGP:egpParentToCursor(18)
    }
    function setCrosshair()
    {
        if(ToolMenu == 1)
        {
            pointer()
        }
        elseif(ToolMenu == 0)
        {
            EGP:egpUnParent(18)
            EGP:egpRemove(18)
            if(ToolSel[1, number] == 1)
            {
                if(Outline == 1 & Rounded == 0) {EGP:egpBoxOutline(19,vec2(),vec2(Width,Height))}
                elseif(Outline == 1 & Rounded == 1) {EGP:egpRoundedBoxOutline(19,vec2(),vec2(Width,Height))}
                elseif(Outline == 0 & Rounded == 1) {EGP:egpRoundedBox(19,vec2(),vec2(Width,Height))}
                elseif(Outline == 0 & Rounded == 0) {EGP:egpBox(19,vec2(),vec2(Width,Height))}
                EGP:egpRadius(19,Radius)
            }
            elseif(ToolSel[2, number] == 1)
            {
                if(Outline == 0) {EGP:egpCircle(19,vec2(),vec2(Width,Height))}
                elseif(Outline == 1) {EGP:egpCircleOutline(19,vec2(),vec2(Width,Height))}
                EGP:egpFidelity(19,Fidelity)
                EGP:egpAngle(19, Angle)
            }
            elseif(ToolSel[3, number] == 1)
            {
            }
            elseif(ToolSel[4, number] == 1)
            {
                EGP:egpText(19,"" + SP + "/" + StripCount,vec2(0,-10))
                EGP:egpAlign(19,2,1)
            }
            elseif(ToolSel[5, number] == 1)
            {
            }
            elseif(ToolSel[6, number] == 1)
            {
                EGP:egpText(19,"" + PP + "/" + PolyCount,vec2(0,-10))
                EGP:egpAlign(19,2,1)
            }
            elseif(ToolSel[7, number] == 1)
            {
                if(Outline == 1) {EGP:egpWedgeOutline(19, vec2(), vec2(Height,Width))}
                else{EGP:egpWedge(19, vec2(), vec2(Height,Width))}
                EGP:egpFidelity(19, Fidelity)
                EGP:egpSize(19, Angle)
            }
            elseif(ToolSel[8, number] == 1)
            {
                EGP:egpText(19,"" + TextString,vec2(0,0))
                EGP:egpAlign(19,Align,1)
                EGP:egpSize(19, TextSize)
            }
            EGP:egpColor(19,vec4(R,G,B,A-75))
            if(ToolSel[4, number] == 1 | ToolSel[6, number] == 1)
            {
            EGP:egpColor(19,vec4(0,0,0,255))
            }
            EGP:egpParentToCursor(19)
        }
    }

    function modeSet()
    {
        if(ToolSel[1, number] == 0 & ToolSel[2, number] == 0 & ToolSel[3, number] == 0 & ToolSel[4, number] == 0 & ToolSel[5, number] == 0 & ToolSel[6, number] == 0 & ToolSel[7, number] == 0 & ToolSel[8, number] == 0) {Rounded = 0, Outline = 0, Mode = "None"}
       if(ToolSel[1, number] == 1)
        {
            if(Outline == 1 & Rounded == 0) {Mode = "Box Outline"}
            elseif(Outline == 1 & Rounded == 1) {Mode = "Rounded Box Outline"}
            elseif(Outline == 0 & Rounded == 1) {Mode = "Solid Rounded Box"}
            elseif(Outline == 0 & Rounded == 0) {Mode = "Solid Box"}
        }
        elseif(ToolSel[2, number] == 1)
        {
            if(Outline == 0) {Mode = "Solid Circle"}
            elseif(Outline == 1) {Mode = "Circle Outline"}
        }
        elseif(ToolSel[3, number] == 1)
        {
            Mode = "Line"
        }
        elseif(ToolSel[4, number] == 1)
        {
            Mode = "Strip"
        }
        elseif(ToolSel[5, number] == 1)
        {
            if(Outline == 0) {Mode = "Solid Triangle"}
            elseif(Outline == 1) {Mode = "Triangle Outline"}
        }
        elseif(ToolSel[6, number] == 1)
        {
            if(Outline == 0) {Mode = "Solid Poly"}
            elseif(Outline == 1) {Mode = "Poly Outline"}
        }
        elseif(ToolSel[7, number] == 1)
        {
            if(Outline == 0) {Mode = "Solid Wedge"}
            elseif(Outline == 1) {Mode = "Wedge Outline"}
        }
        elseif(ToolSel[8, number] == 1)
        {
            Mode = "Text"
        }
        updateText()
    }
    modeSet()

    {
        Color = vec4(R,G,B,A)
    }

    {
        EGP:egpColor(I,Color)
    }
    function breakLine()
    {
        TempVar = Points[200, vector2]
        Points = array()
        Points[1, vector2] = TempVar
        layerPush()
        updateText()
        AI = 1
    }
    function draw()
    {
        if(!Switch)
        {
            layerPush()
            updateText()
        }
        DrawTool[AI, vector2] = Cursor
        if(AI < 200) {AI++}
        else{breakLine()}
        EGP:egpLineStrip(I,DrawTool)
        EGP:egpColor(I,vec4(R,G,B,A))
        EGP:egpSize(I,1)
        Switch = 1
    }
    function resetDrawingTool()
    {
        Switch = 0
        DrawTool = array()
        AI = 1
    }
