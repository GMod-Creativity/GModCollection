@name data v time plotter
@inputs EGP:wirelink Data Min Max TimeStep Res
@outputs Mem:array
@persist Timer S Res
@trigger none
interval(1)
if (first() | duped()) {EGP:egpClear()}
Timer++
    if (Timer == TimeStep & ->TimeStep) {
        Size = (Data - Min) / (Max - Min)
        S = 512 * (1 - Size)
        Mem:unshiftNumber(S)
        Mem:remove(Res + 1)
        for (I = 0, Res) {
            Point1 = vec2(512 - (I * 512 / Res), Mem[I, number])
            Point2 = vec2(512 - ((I + 1) * 512 / Res), Mem[I + 1, number])
            EGP:egpLine(I, Point1, Point2)
        }
        Timer = 0
}
