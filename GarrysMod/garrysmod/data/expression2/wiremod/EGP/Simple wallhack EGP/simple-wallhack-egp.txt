@name Not an EGP Wallhack
@inputs EGP:wirelink

interval(100)

if (first()|changed(numPlayers()))
{
    EGP:egpClear()
    for (I=1,numPlayers())
    {
        II=I+numPlayers()
        Pers=players()[I,entity]
        EGP:egp3DTracker(I,Pers:shootPos())
        EGP:egpText(II,"",vec2())
        EGP:egpColor(II,teamColor(Pers:team()))
        EGP:egpParent(II,I)
    }
}

for (I=1,numPlayers())
{
    II=I+numPlayers()
    Pers=players()[I,entity]
    EGP:egpPos(I,Pers:shootPos())
    EGP:egpSetText(II,Pers:name()+": "+floor(Pers:pos():distance(owner():pos())))
    EGP:egpAlpha(II,(Pers:pos():distance(owner():pos())>10)*255)
}
