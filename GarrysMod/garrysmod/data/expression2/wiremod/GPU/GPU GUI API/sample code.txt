@name GGA Sample Code
@inputs ID
@persist [GUI Update]:gtable
runOnTick(1)

if(first() | dupefinished())
{
    GUI = gTable("GGA_GUI_" + ID)
    Update = gTable("GGA_Update_" + ID)
    timer("MainForm", 10)
}

if(GUI["ClickEvent", string])
{
    Trigger = GUI["ClickEvent", string]
    GUI["ClickEvent", string] = ""
    
    if(Trigger == "Button.Play")
    {
        soundPlay(1, 0, "ambient/music/cubanmusic1.wav")
    }
    elseif(Trigger == "Button.Stop")
    {
        soundStop(1)
    }
}

if(clk("MainForm"))
{
    FormR = array()
    FormR:pushString("BGColor/BGColor/ R:255, G:255, B:255, A:255")
    FormR:pushString("Font/Font/ FontID:2, Fontsize:35")
    FormR:pushString("Line/Separator/ StartX:0, StartY:96, EndX:512, EndY:96, R:0, G:0, B:0, A:255")
    FormR:pushString("Grid/Title/ StartX:0, StartY:0, EndX:512, EndY:96, Rows:1, Columns:1")
    FormR:pushString("Grid/Buttons/ StartX:0, StartY:96, EndX:512, EndY:512, Rows:2, Columns:1, SpacingX:50, SpacingY:50")
    FormR:pushString("Label/Title/ Parent:Grid.Title, Label:Cuban Music Player, Row:1, Column:1, R:255, G:128, B:0, A:255")
    FormR:pushString("Button/Play/ Parent:Grid.Buttons, Row:1, Column:1, R:50, G:255, B:50, A:255, Label:Play")
    FormR:pushString("Button/Stop/ Parent:Grid.Buttons, Row:2, Column:1, R:255, G:50, B:50, A:255, Label:Stop")
    
    GUI["LoadForm", number] = 1
    GUI["Form", array] = FormR
}

if(GUI["FormLoaded", number])
{
    #Here you must initialize lists
    #E2 sets "FormLoaded" to 1 when the form is ready and visible on GPU
    GUI["FormLoaded", number] = 0 #Or you'll have infinite loop
}
