mov #frame,#63488;
mov #select,#63489;
mov #regCursor,1; // Enable cursor
cmp #frame,1; je Intro;
cmp #frame,2; je MainMenu;
cmp #frame,10; je MusicMenu;

    cmp #frame,10.1; je HL1Menu;
    cmp #frame,10.2; je HL2Menu;
    cmp #frame,10.3; je EP2Menu;

//=============================================================================

    dsetsize 24;
    mov #pos.x,208; mov #pos.y,274;
    dwrite pos,intro2;
    dsetsize 32;
    mov #pos.x,130; mov #pos.y,468;
    dwrite pos,intro3;
jmp Exit;

MainMenu: // Frame = 2

    call SetupText; dwrite pos,about;

    call SetupText; dwrite pos,custom;

jmp Exit;

// Layout Subroutines
//=============================================================================
DrawFrame:
    call SetDrawColor_Grey;
    mov #pos.x,0; mov #pos.y,0;
    mov #size.x,512; mov #size.y,40;
    drectwh pos,size; // Draw underline
    call SetDrawColor_White;
    mov #pos.x,8; mov #pos.y,0;
    dsetsize 40;
    dsetfont 2;
    mov #init_offset,0;
ret;

SetupText:
    cmp #init_offset,0;
    ce InitialOffset; // Initial offset for text position
    add #pos.y,32;
ret;

DrawReturn:
    mov #pos.x,8;
    mov #pos.y,476;
    call SetDrawColor_Red;
    dsetsize 32;
    dwrite pos,return;
ret;

DrawNext:
    mov #pos.x,440;
    mov #pos.y,476;
    call SetDrawColor_Green;
    dsetsize 32;
    dwrite pos,next;
ret;

DrawHold:
    mov #pos.x,440;
    mov #pos.y,476;
    call SetDrawColor_White;
    cmp #63493,1; // Enable hold?
    ce SetDrawColor_Green;
    dsetsize 32;
    dwrite pos,hold;
ret;

DrawPitchBar:
    dsetsize 32;
    call SetDrawColor_Green;
    mov #pos.x,28;
    mov #pos.y,128;
    dwrite pos,pitch;
    mov #pos.x,128;
    mov #size.x,354;
    mov #size.y,42;
    call SetDrawColor_White;
    drectwh pos,size;
    call SetDrawColor_Green;
    add #pos.x,4;
    add #pos.y,4;
    mov #size.x,#63491;
    mul #size.x,1.385;
    mov #size.y,34;
    drectwh pos,size;
    call SetDrawColor_Black;
    mov #pos.x,268;
    sub #pos.y,4;
    mov #size.x,4;
    mov #size.y,42;
    drectwh pos,size;
ret;

DrawVolumeBar:
    dsetsize 32;
    call SetDrawColor_Green;
    mov #pos.x,12;
    mov #pos.y,216;
    dwrite pos,volume;
    mov #pos.x,128;
    mov #size.x,354;
    mov #size.y,42;
    call SetDrawColor_White;
    drectwh pos,size;
    call SetDrawColor_Green;
    add #pos.x,4;
    add #pos.y,4;
    mov #size.x,#63492;
    mul #size.x,346;
    mov #size.y,34;
    drectwh pos,size;
ret;

InitialOffset: // Initial offset for text position
    mov #pos.x,8;
    add #pos.y,12;
    mov #init_offset,1;
    call SetDrawColor_White;
    dsetsize 32;
ret;

// Color Set
//=============================================================================
SetDrawColor_White: dcolor white; ret;
SetDrawColor_Red: dcolor red; ret;
SetDrawColor_Green: dcolor green; ret;
SetDrawColor_Blue: dcolor blue; ret;
SetDrawColor_Shade: dcolor shade; ret;
SetDrawColor_Grey: dcolor grey; ret;
SetDrawColor_Black: dcolor black; ret;

// 3D Rendering
//=============================================================================
Render3D:
    dcvxpipe 3; // -1..1 corrdinates
    dvxpipe 5; // Matrix projection
    dclrscr black;
    mlookat mViewMatrix,vLookAt; // View matrix
    mperspective mProjectionMatrix,vPerspective; // Initialize transform
    timer #vRotate.w;
    mrotate mRotateMatrix,vRotate; // Rotate
    mtranslate mTranslateMatrix,vTranslate; // Translate
    mmov mModelMatrix,mTranslateMatrix; // Create model matrix
    mmul mModelMatrix,mRotateMatrix;
    mmov mModelViewMatrix,mViewMatrix; // modelViewMatrix = ViewMatrix * modelMatrix
    mmul mModelViewMatrix,mModelMatrix;
    mload mModelViewMatrix; // Load matrix
    mloadproj mProjectionMatrix;
    dsetlight 0,LightData; //Setup light
    denable 0; //Vertex buffer
    denable 1; //ZSortings
    denable 2; //Lighting
    dvxdata_3f VertexData_Intro,21; // Load up vertex data
    dvxflush; // Flush Data
    dcvxpipe 0; // Disable 3D Features
    dvxpipe 0;
    ddisable 0;
    ddisable 1;
    ddisable 2;
ret;
    
// Exit Frame
//=============================================================================
Exit: dexit;
