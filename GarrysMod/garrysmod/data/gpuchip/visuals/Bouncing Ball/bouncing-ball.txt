BallBounding:
    //ALL of the following should be Ball, not Cursor anymore!
    //V1 is cursor to the first point( Cursor - Point1 )
    //V2 is cursor to the second point( Cursor - Point2 )
    //V3 is Point2 - Point1
    //V4 is Point1 - Point2
    //V5 is Perpendicular to Point2 - Point1
    
    //using dot products, tell where cursor will intersect!
    //No need to normalize, except on perpendicular
    //Point3 will hold where this intersection will occur
        //Aren't comments just delightful?.....
        mov #MinDistance,#Radius;
        mul #MinDistance,1.08;
        
        vmov V1,Ball;
        vsub V1,Point1;
        //done
        
        vmov V2,Ball;
        vsub V2,Point2;
        //done
        
        vmov V3,Point2;
        vsub V3,Point1;
        //done
        
        vmov V4,Point1;
        vsub V4,Point2;
        //done
        
    //time for dot product stuff
    //first, check if Point is even within the two points
    //if V1:dot( V3 ) and V2:dot( V4 ) have the same sign then the point is included
        vdot V1,V3;
        
        vdot V2,V4;
        
        //now to check signs (?)
        fsgn ebx,#V2.x;
        fsgn eax,#V1.x;
        
        cmp eax,ebx;
        jne EndThisCall;
    
    //Now for all the perpendiculars
    // in the ( -y,x ) form
    // then norm-ed
        mov #V5.x,#V3.y;
        mov #V5.y,#V3.x;
        neg #V5.x;
        
        vnorm V5,V5;
        
        //get distance from point from line
        vmov V1,Ball;
        vsub V1,Point1;
        
        vdot V1,V5;
        mov edx,#V1.x;
        //fabs edx,edx;
            fsgn eax,edx;
            cmp eax,-1;
            //jne EndThisCall; //check if above line!
            
        fabs edx,edx;
        //edx holds distance
        cmp edx,#MinDistance;
        jge EndThisCall;
        
    //Now for projected hit point!
    //Cursor - Distance*Perp

        call ResetBallv;

        neg eax;
        vmul V5,eax;  //

EndThisCall:
    ret;

ResetBallv:
    //We have Incident(V1) and SurfNorm(V2)
    //NewI = I - 2*(N.I * N)
    //1 N.I
    //2 N.I*N
    //3 -2(N.I*N)
    //4 I + -2(N.I*N)
    
    vnorm SurfaceNorm,SurfaceNorm;
    vmov V1,D;
    vmov V2,SurfaceNorm;
    //1
    vdot V1,SurfaceNorm;
    //2
    vmul V2,#V1.x;
    //3
    vmul V2,-2;
    //4
    vadd V2,D;

doBallV:
        vmov V1,D;
        vmul V1,0.065;
        vadd Ball,V1;
        
        cmp #Ball.x,512;
        jl  _1;
        
        min #Ball.x,512;
        
        _1:
        
        cmp #Ball.x,0;
        jg _2;
        
        max #Ball.x,0;
        
        _2:
        cmp #Ball.y,512;
        jl  _3;
        
        min #Ball.y,512;
        
        _3:
        cmp #Ball.y,0;
        jg  _4;
        
        max #Ball.y,0;
        
        _4:
ret;

GetSpeed:
        mov eax,#D.x;
        mov ebx,#D.y;
        
        fpwr eax,2;
        fpwr ebx,2;
        
        add eax,ebx;
        
        fpwr eax,0.5;
        
ret;
