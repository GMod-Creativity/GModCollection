@name Clock
@model models/props_trainstation/trainstation_clock001.mdl
@persist I Secs Mins Hours OldSecs Tock

function handle(Scale:vector,Color:vector){
    I++
    holoCreate(I)
    holoPos(I, entity():pos())
    holoAng(I, entity():angles())
    holoModel(I,"hqsphere")
    holoScale(I, Scale)
    holoParent(I, entity())
    holoColor(I,Color)
}

if( first() ){
    handle( vec(0.05,0.05,2),vec(255,50,50) )
    handle( vec(0.2,0.2,1.8),vec() )
    handle( vec(0.3,0.3,1),vec() )
}

function move(Ind,Size,Angle){  
holoAng(Ind,entity():toWorld( ang(0,0,Angle)))
holoPos(Ind,entity():toWorld( vec(0,sin(-Angle)*Size,cos(-Angle)*Size) ))
}

timer("Tick",100)

Secs = time("sec")*6
Mins = time("min")*6
Hours = time("hour")*30

move(1,10,-Secs)
move(2,8,-Mins)
move(3,5,-Hours)

if( Secs!=OldSecs ){
    OldSecs=Secs
    Tock = !Tock
    entity():soundPlay(0,0,"ui/menu_focus.wav")
    soundVolume(0,0.05+Tock*0.05)
    soundPitch(0,100+Tock*100)
}
