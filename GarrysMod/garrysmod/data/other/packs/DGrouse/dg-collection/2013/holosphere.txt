@name Holosphere
@inputs H:wirelink
@persist Time Speed Size Half [Offset Sine]:vector X Y Z

interval(1)

if( first() ){

    Speed = 1
    Size = 70

    H["Color",vector] = vec(0,255,0)
    H["LineBeam",number] = 1
    H["FadeTime",number] = 10
    H["Size",number] = 4
    H["Active",number] = 1
    
    Half = Size/2
}

Time += Speed

Z = sin(Time)
X = cos(Time*10)*cos(Time)
Y = sin(Time*10)*cos(Time)


Offset = H:entity():pos()+vec(0,0,Half+30)
Sine = vec(X,Y,Z)*Half

H["Pos",vector] = Offset+Sine
