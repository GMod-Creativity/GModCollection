@name DGPC2 Router
@model models/props_lab/reciever01a.mdl
@persist DGPCS:array

if( first() ){
    dsSetScope(2)
    dsJoinGroup("DG-WIFI")
    
    function array:addnumber( N ){
        local Found = 0
        foreach( K,V:number = This ){
            if( V == N ){
                Found = 1
                break
            }
        }
        if( !Found ){
            This:pushNumber(N)
        }
    }
    
}

if( dsClk("DGPC_Joined") ){
    local A = dsGetArray()
    local N = A[1,string]
    local C = A[2,number]
    print( "Router: "+N+" connected with DGPC["+C+"]." )
    DGPCS:addnumber(C)
}

if( dsClk("GetConnected") ){
    local E = dsGetEntity()
    dsSendDirect( "UD_Connected",E,DGPCS )
}

if( dsClk("SendArray") ){
    local T = dsGetTable()
    local C = T[1,string]
    local E = entity( T[2,number] )
    local A = T[3,array]
    dsSendDirect( C,E,A )
}

if( dsClk("SendNumber") ){
    local T = dsGetTable()
    local C = T[1,string]
    local E = entity( T[2,number] )
    local N = T[3,number]
    dsSendDirect( C,E,N )
}
