@name D-Ray: Guides
@persist LastRanger:ranger

if( 0 ){
    #include "D-Ray/Core"
}

function removeGuides(){
    hookRemove("Guides")
    for( I=1,5+Bounces ){
        holoDelete(I)
    }
    for( S=1,Samples ){
    for( H=1,Samples ){
        local I = S*10000+H*10
        holoDelete(I)
    }}
}

function runGuides(){
    refreshEntVars()
    for( I=1,4 ){
        local A = (360/4)*I
        local R = rangeGuide( ang(sin(A)+sin(A-90),0,cos(A)+cos(A-90))*FOV/2 )
        holoPos(I,R:position())
    }
    
    LastRanger = rangeGuide( ang() )
    holoPos(5,LastRanger:position())
    
    
    
    for( S=1,Samples ){
    for( H=1,Samples ){
        local I = S*10000+H*10
        local R = LastRanger
        
        local Z = 2/Samples*( S ) - 1
        local T = ( 2*pi() )/Samples*( H )
        
        local M = sqrt(1-Z^2)
        local X = M * cosr( T )
        local Y = M * sinr( T )

        local NR = rangeGuide( R:pos(),vec(X,Y,Z) )
        
        
        holoPos(I,NR:pos())
        
    }}
    #[
    
    for( B=1,Bounces ){
        local I = 5+B
        local R = LastRanger
        local D = LastDirection
        local N = R:hitNormal()
        LastRanger = rangeGuide( R:position(),D-2*(D*N)*N )
        holoPos(I,LastRanger:position())
    }
    ]#
    
    timer("Guides",100)
}

function initGuides(){
    for( I=1,5 ){
        holoCreate(I)
        holoModel(I,"hqsphere")
        holoScale(I,vec(0.6))
        holoColor(I,vec(255,0,0))
    }
    
    for( S=1,Samples ){
    for( H=1,Samples ){
        local I = S*10000+H*10
        holoCreate(I)
        holoModel(I,"hqsphere")
        holoScale(I,vec(0.2))
        holoColor(I,vec(20*H,20*S,255))
    }}
    
    hookAdd("Guides","Guide","runGuides",100)
}
