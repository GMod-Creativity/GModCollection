@name D-Ray: Materials

}

function materialAssign( Material:string,Func:string ){
    local T = table()
    Materials[Material,string] = "mat_"+Func
}

function mat_Nil( R:ranger ){
    shade_Diffuse( R,vec(210,210,210) )
}

function mat_Diffuse( R:ranger ){
    shade_Diffuse( R,R:getColor() )
}

function mat_Mirror( R:ranger ){
    shade_Mirror( R,R:getColor() )
}

function mat_Emission( R:ranger ){
    shade_Emission( R,R:getStrength(),R:getColor() )
}

function mat_Sky( R:ranger ){
    shade_Emission( R,SkyBrightness,SkyColor )
}
