@name D-Ray: Variables
@inputs DS:wirelink

# Performance
@persist Threads StepTime

# Quality
@persist Bounces DiffuseBounces Samples AA AARadius

# Camera
@persist Res FOV

# World
@persist SkyColor:vector SkyBrightness

# Options
@persist IgnorePlayers PixelOrder:string

# Resource
@persist [Materials EntityData]:table MaxRange Filter:array
@persist [Epos Edir]:vector Eang:angle CurrentBounce ThreadCount

# Render
@persist Rays:table CurrentAA CurrentPixel:array
@persist [LastDirection]:vector RayID NextExit
@outputs CX CY

@persist RM_Ranger:ranger RM_Func:string

# Pixel Order
@persist O_C_S O_C_N O_C_D

function refreshEntVars(){
    Epos = entity():pos()
    Edir = entity():up()
    Eang = entity():angles()
}

function refreshFilter(){
    Filter = array( entity() )
    if( IgnorePlayers ){
        Filter = Filter:add( players() )
    }
}

function refreshInits(){
    refreshEntVars()
    refreshFilter()
}

MaxRange = 100000000
