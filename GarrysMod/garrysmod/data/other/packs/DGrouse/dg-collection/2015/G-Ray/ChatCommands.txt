@persist LS:string Commands:table

function addCommand( Name:string,Func:string,Arg ){
    Commands[Name,table] = table()
    Commands[Name,table]["Func",string] = Func
    Commands[Name,table]["Arg",number] = Arg
}

function vector stringToVector( A:array ){
    return vec( A[1,string]:toNumber(),A[2,string]:toNumber(),A[3,string]:toNumber() )
}

function runChat(){
    if( chatClk(owner()) ){ 
        LS=owner():lastSaid() 
        
        foreach(K,V:table = Commands){
            if( LS:left(K:length())==K ){
                if( V["Arg",number] == 1 ){ # String Argument
                
                    V["Func",string]( LS:replace(K+" ","") )
                    
                }elseif( V["Arg",number] == 2 ){ # Vector Argument
                
                    local S = LS:replace(K+" ","")
                    local A = S:explode(" ")
                    V["Func",string]( stringToVector( A ) )
                    
                }elseif( V["Arg",number] == 3 ){ # Array Argument
                
                    local S = LS:replace(K+" ","")
                    local A = S:explode(" ")
                    V["Func",string]( A )
                    
                }else{ # No Argument
                
                    V["Func",string]()
                    
                }
                hideChat(1)
                break
            }
        }
    }
}

# Options
# ===================================================
function setIgnorePlayers( S:string ){
    IgnorePlayers = S:toNumber()
    if( IgnorePlayers ){
        print( "Tracers will now ignore players" )
    }else{
        print( "Tracers will now hit players" )
    }
    refreshFilter()
}
addCommand( "!setignoreplayers","setIgnorePlayers",1 )

function toggleIgnorePlayers(){
    if( IgnorePlayers ){
        setIgnorePlayers( "0" )
    }else{
        setIgnorePlayers( "1" )
    }
}
addCommand( "!toggleignoreplayers","toggleIgnorePlayers",0 )
addCommand( "!ignoreplayers","toggleIgnorePlayers",0 )

function makeLight( V:array ){
    local E = owner():aimEntity()
    E:makeLight( V )
    print( "Entity ["+E:id()+"] is now a light." )
}
addCommand( "!makelight","makeLight",3 )

function cmdstartRender(){
    print("Now rendering "+Res+"x"+Res+" image")
    startRender()
    "removeGuides"()
}
addCommand( "!start","cmdstartRender",0 )

function cmdstopRender(){
    hookRemove("RunSample")
    hookRemove("RunMaterial")
    hookRemove("RunExit")
    "initGuides"()
}
addCommand( "!stop","cmdstopRender",0 )
