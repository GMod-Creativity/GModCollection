@name G-Ray: Guides
@persist LastRanger:ranger

if( 0 ){
    #include "G-Ray/Core"
}

function removeGuides(){
    hookRemove("Guides")
    for( I=1,5+Bounces ){
        holoDelete(I)
    }
}

function runGuides(){
    refreshEntVars()
    for( I=1,4 ){
        local A = (360/4)*I
        local R = rangeGuide( ang(sin(A)+sin(A-90),0,cos(A)+cos(A-90))*FOV/2 )
        holoPos(I,R:position())
    }
    
    LastRanger = rangeGuide( ang() )
    holoPos(5,LastRanger:position())

    timer("Guides",100)
}

function initGuides(){
    for( I=1,5 ){
        holoCreate(I)
        holoModel(I,"hqsphere")
        holoScale(I,vec(0.6))
        holoColor(I,vec(255,0,0))
    }
    hookAdd("Guides","Guide","runGuides",100)
}
