@name G-Ray: Hooks
@persist Hooks:table [LastHook LastIdentifier]:string

function hookAdd(Name:string,Indentifier:string,Func:string){
    if( !Hooks[Name,table] ){ Hooks[Name,table] = table() }
    Hooks[Name,table][Indentifier,string] = Func
}

function hookAdd(Name:string,Indentifier:string,Func:string,Time:number){
    if( !Hooks[Name,table] ){ Hooks[Name,table] = table() }
    Hooks[Name,table][Indentifier,string] = Func
    timer(Name,Time)
}

function hookRemove(Name:string){
    Hooks:remove(Name)
    stoptimer(Name)
}

function hookRemoveIdentifier(Name:string,Indentifier:string){
    Hooks[Name,table]:remove(Indentifier)
}

function runHooks(){
    foreach(Name,ID:table = Hooks){
        foreach(K,V:string = ID){
            if( clk(Name) ){
                LastHook = Name
                LastIdentifier = K
                V()
            }
        }
    }
}

function clearHooks(){
    Hooks:clear()
}
