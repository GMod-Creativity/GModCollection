@name G-Ray: Variables
@inputs DS:wirelink

# Performance
@persist Threads StepTime

# Quality
@persist Bounces AA SoftShadows

# Camera
@persist Res FOV

# World
@persist [SkyColor AtmosColor]:vector SkyBrightness AtmosDensity AtmosRange

# Options
@persist IgnorePlayers PixelOrder:string PercentNotify

# Resource
@persist [Materials EntityData]:table MaxRange Filter:array AARadius
@persist [Epos Edir]:vector Eang:angle CurrentBounce ThreadCount

# Render
@persist [Rays Lights]:table CurrentAA CurrentPixel:array
@persist [LastDirection]:vector RayID NextExit Spheres:table
@outputs CX CY Percent

@persist RM_Ranger:ranger RM_Func:string ExitParent Offsets:array
@persist StartTime PercentJump IsLight PercentTime SS CAtmos

# Pixel Order
@persist O_C_S O_C_N O_C_D

function refreshEntVars(){
    Epos = entity():pos()
    Edir = entity():up()
    Eang = entity():angles()
}

function refreshFilter(){
    Filter = array( entity() )
    if( IgnorePlayers ){
        Filter = Filter:add( players() )
    }
    for(I=1,Lights:count()){
        local Data = Lights[I,table]
        Filter:pushEntity( entity(Data["ID",number]) )
    }
}

function refreshInits(){
    refreshEntVars()
    refreshFilter()
    AARadius = FOV/Res/2
    SS = SoftShadows*6
}

MaxRange = 100000000

Spheres = invert(table("",
    "models/hunter/misc/sphere075x075.mdl"
))

Offsets = array(
    vec(1,0,0),
    vec(-1,0,0),
    vec(0,1,0),
    vec(0,-1,0),
    vec(0,0,1),
    vec(0,0,-1)
)
