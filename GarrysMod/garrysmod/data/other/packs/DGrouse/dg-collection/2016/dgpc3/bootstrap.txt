@name DGPC3-Bootstrap
@persist BOOT_Bar BOOT_Percent BootstrapIncludes

function bootstrapIncludes(){
    if( BootstrapIncludes ){ return }
    BootstrapIncludes = 1
    
    #include "dgpc3/draw"
    #include "dgpc3/elements/loadingbar"
    #include "dgpc3/login"
    #include "dgpc3/hooks"
    #include "dgpc3/programs"
    
    mountPrograms()
}

function bootstrap(){

    bootstrapIncludes()

    clear()
    
    text("DGrouse Industries",256,150,40,2,1,1)
    text("DGPC3",256,200,50,2,1,1)
    text("Your computer is loading!",256,300,30,2,1,1)
    BOOT_Bar = loadingBarInit(256,350,300,30,6,2,1,2,0)
    
    hookAdd("BS-BarUpdate","boot","bootBarUpdate",10)
}

function bootBarUpdate(){
    
    BOOT_Percent = BOOT_Percent+0.02
    loadingBarPercent(BOOT_Bar,BOOT_Percent)
    
    if( BOOT_Percent < 1 ){
        timer("BS-BarUpdate",50)
    }else{
        hookRemove("BS-BarUpdate")
        BOOT_Percent = 0
        initLogin()
    }
}
