@name DGPC3-Clicks
@persist LastUser:entity

#include "dgpc3/inputs"
#include "dgpc3/hooks"

function runUser(){
    if( User ){
        runOnKeys(LastUser,0)
        LastUser = User
        runOnKeys(User,1)
    }
    if( !User ){
        runOnKeys(LastUser,0)
    }
}

function runClick(){
    if( keyClk() ){
        if( User:keyAttack1() ){
            timer("LeftClick",0)
        }
        if( User:keyAttack2() ){
            timer("RightClick",0)
        }
    }
}
