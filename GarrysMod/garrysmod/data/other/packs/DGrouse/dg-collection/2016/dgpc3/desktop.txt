@name DGPC3-Desktop
@persist StartMenuOpen StartMenuWindow DesktopIncludes SubMenuWindow
@persist [StartMenuOptions ControlPanelOptions]:table

function dekstopIncludes(){
    if( DesktopIncludes ){ return }
    DesktopIncludes = 1
    
    #include "dgpc3/lib1"
    #include "dgpc3/lib2"
    #include "dgpc3/programs"
    
    #include "dgpc3/elements/backgrounds"
    #include "dgpc3/elements/cursor"
    #include "dgpc3/elements/button"
    #include "dgpc3/elements/window"
    
    #include "dgpc3/desktop/info"
    #include "dgpc3/desktop/manageusers"
    #include "dgpc3/desktop/settings"
    #include "dgpc3/desktop/gadgets"
    #include "dgpc3/desktop/personalize"
    #include "dgpc3/desktop/network"
}

StartMenuOptions["Programs",table] = table(0,"SubMenu",ProgramsOptions)
StartMenuOptions["Control Panel",table] = table(0,"SubMenu",ControlPanelOptions)
StartMenuOptions["Info",table] = table(0,"Function","dt_openInfo")
StartMenuOptions["Logout",table] = table(0,"Function","toLogin")
StartMenuOptions["Shutdown",table] = table(0,"Function","shutdown")

ControlPanelOptions["Network",table] = table(0,"Function","dt_openNetwork")
ControlPanelOptions["Gadgets",table] = table(0,"Function","dt_openGadgets")
ControlPanelOptions["Manage Users",table] = table(1,"Function","dt_openMangageUsers")
ControlPanelOptions["Personalize",table] = table(0,"Function","dt_openPersonalize")

function toLogin(){
    StartMenuOpen = 0
    clearButtons()
    clearWindows()
    clearHooks()
    "initLogin"()
}

function loadUserDesktop(){
    local T = Users[CurrentUser,table]
    T["ColorSet",string]()
    T["Background",string]()
    T["Background Effects",string]()
    T["TaskBar",string]()
}

function closeStartMenu(){
    StartMenuOpen = 0
    windowClose( StartMenuWindow )
    "closeSubMenu"()
}

function toggleStartMenu(){
    
    StartMenuOpen = !StartMenuOpen
    
    if( StartMenuOpen ){
        StartMenuWindow = createmenu(100,433,200,70,StartMenuOptions,CurrentUser)
        
    }else{
        windowClose( StartMenuWindow )
        "closeSubMenu"()
    }
}

function closeSubMenu(){
    if( !SubMenuWindow ){ return }
    windowClose( SubMenuWindow )
    SubMenuWindow = 0
}

function openSubMenu(){
    if( SubMenuWindow ){ closeSubMenu() }
    local SubMenu = LastIdentifier:sub(5)
    SubMenuOptions = StartMenuOptions[SubMenu,table][3,table]
    SubMenuWindow = createmenu(300,433,200,70,SubMenuOptions,SubMenu)
}

function startButton(){
    circle(20,492,36,36,1)
    buttonAdd(circle(20,492,32,32,11),"StartMenu",20,492,64,64,11,12,"toggleStartMenu",1)
    box(25,478,18,18,1)
    box(9,494,18,18,1)
    box(25,494,18,18,1)
    box(41,494,18,18,1)
    box(25,478,14,14,17)
    box(9,494,14,14,18)
    box(25,494,14,14,19)
    box(41,494,14,14,20)
}

function justPersonalized(){
    if( JustPersonalized ){
        dt_openPersonalize()
        JustPersonalized = 0
    }
}

function desktopInit(){
    dekstopIncludes()
    loadUserDesktop()
    loadGadgets()
    startButton()
    justPersonalized()
    cursor()
}
