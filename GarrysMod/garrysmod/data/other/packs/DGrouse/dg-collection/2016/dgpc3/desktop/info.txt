@name DGPC3-D-Info
@persist Info_Window

function dt_InfoIncludes(){
    #include "dgpc3/elements/cursor"
    #include "dgpc3/elements/button"
    #include "dgpc3/elements/window"
}

function dt_openInfo(){

    "toggleStartMenu"()
    local X = 256
    local Y = 256
    local W = 290
    local H = 120
    Info_Window = windowNew(X,Y,W,H,"dt_closeInfo")
    windowContent("Objects",array(
        text("Info",X-W/2+20,Y-H/2+20,30,1,0,1),
        text("Created by DGrouse",X-W/2+20,Y-H/2+60,20,1,0,1),
        text("Started: 19/10/14",X-W/2+20,Y-H/2+80,20,1,0,1),
        text("Last Updated: 5/1/15",X-W/2+20,Y-H/2+100,20,1,0,1)
    ))
    cursor()
}

function dt_closeInfo(){
    windowClose( Info_Window )
}
