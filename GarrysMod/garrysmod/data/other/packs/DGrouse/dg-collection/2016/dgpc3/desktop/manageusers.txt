@name DGPC3-D-ManageUsers
@persist ManUse_Window RemoveUsersWindow

function dt_ManageUsersIncludes(){
    #include "dgpc3/elements/cursor"
    #include "dgpc3/elements/button"
    #include "dgpc3/elements/window"
    #include "dgpc3/lib1"
    #include "dgpc3/lib2"
}

function dt_openMangageUsers(){
    
    "toggleStartMenu"()
    
    local X = 256
    local Y = 256
    local W = 290
    local H = 140
    ManUse_Window = windowNew(X,Y,W,H,"dt_closeMangageUsers")
    local LX = X-W/2 local RX = X+W/2 local TY = Y-H/2 local BY = Y+H/2
    windowContent("Buttons",array(
        buttonAdd(box(X,TY+60,170,30,13),"CreateNewUser",X,TY+60,170,30,13,14,"dt_MUCreateUser",1),
        buttonAdd(box(X,TY+100,170,30,13),"RemoveUser",X,TY+100,170,30,13,14,"dt_MURemoveUser",1)
    ))
    windowContent("Objects",array(
        text("Manage Users",LX+20,TY+20,30,1,0,1),
        text("Create New User",X,TY+60,20,1,1,1),
        text("Remove User",X,TY+100,20,1,1,1)
    ))
    
    cursor()
}

function dt_closeMangageUsers(){
    windowClose( ManUse_Window )
}

function dt_removeUserWindow(){
    windowClose( RemoveUsersWindow )
    RemoveUsersWindow = 0
}

function dt_removeUser(){
    local U = LastIdentifier:sub(5)
    Users:removeTable(U)
    dt_removeUserWindow()
}

function dt_MURemoveUser(){
    
    dt_closeMangageUsers()
    if( RemoveUsersWindow ){ dt_removeUserWindow() }
    
    local T = userRemoveTbl()
    
    RemoveUsersWindow = createmenu(256,400,250,70,T,"Remove User")
    local C = T:count()*40
    windowContent("Buttons",buttonAdd(box(360,390-C,20,20,15),"dt_removeUser",360,390-C,20,20,15,16,"dt_removeUser",1) )
    
    cursor()
}

function dt_MUCreateUser(){
    dt_closeMangageUsers()
    clear()
    ColorSet = CS_Default
    "bg_bluegrad"()
    "bge_spotted"()
    "createUser"()
}
