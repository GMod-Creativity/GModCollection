@name DGPC3-D-Personalize
@persist Personal_Window Change_Window JustPersonalized

function dt_ManageUsersIncludes(){
    #include "dgpc3/elements/cursor"
    #include "dgpc3/elements/button"
    #include "dgpc3/elements/window"
    #include "dgpc3/elements/backgrounds"
    #include "dgpc3/lib1"
    #include "dgpc3/lib2"
}

function dt_openPersonalize(){
    
    "closeStartMenu"()
    
    local X = 130
    local Y = 256
    local W = 290
    local H = 220
    Personal_Window = windowNew(X,Y,W,H)
    local LX = X-W/2 local RX = X+W/2 local TY = Y-H/2 local BY = Y+H/2
    windowContent("Buttons",array( 
        buttonAdd(box(RX-25,TY+25,20,15,16),"closePersonalize",RX-25,TY+25,20,20,15,16,"dt_closePersonalize",1),
        buttonAdd(box(X,TY+60,230,30,13),"Change Background",X,TY+60,250,30,13,14,"dt_ChangeBackground",1),
        buttonAdd(box(X,TY+100,230,30,13),"Change Effects",X,TY+100,250,30,13,14,"dt_ChangeBackgroundEffect",1),
        buttonAdd(box(X,TY+140,230,30,13),"Change TaskBar",X,TY+140,250,30,13,14,"dt_ChangeTaskBar",1),
        buttonAdd(box(X,TY+180,230,30,13),"Change Colors",X,TY+180,250,30,13,14,"dt_ChangeColorScheme",1)
    ))
    windowContent("Objects",array(
        text("Manage Users",LX+20,TY+20,30,1,0,1),
        text("Change Background",X,TY+60,20,1,1,1),
        text("Change Background Effect",X,TY+100,20,1,1,1),
        text("Change Taskbar",X,TY+140,20,1,1,1),
        text("Change Color Scheme",X,TY+180,20,1,1,1)
    ))
    
    cursor()
}

function dt_closePersonalize(){
    windowClose( Personal_Window )
} 

function dt_closeChange(){
    windowClose( Change_Window )
    Change_Window = 0
}

function dt_ChangeSomething( T:table ){
    if( Change_Window ){ dt_closeChange() }
    
    local Nm = LastIdentifier:sub(8)
    
    Change_Window = createmenu(395,316,240,70,T,Nm,"dt_closeChange")
    
    cursor()
}

function dt_ChangeBackground(){
    dt_ChangeSomething( Backgrounds )
}

function dt_ChangeBackgroundEffect(){
    dt_ChangeSomething( BackgroundEffects )
}

function dt_ChangeTaskBar(){
    dt_ChangeSomething( TaskBars )
}

function dt_ChangeColorScheme(){
    dt_ChangeSomething( ColorSets )
}

function dt_SetPersonal( Name:string,Tab:table,Menu:string ){
    local B = LastIdentifier:sub(5)
    local Func = Tab[B,table][4,string]
    Users[CurrentUser,table][Name,string] = Func
    
    clearAll()
    JustPersonalized = 1
    "desktopInit"()
#    dt_openPersonalize()
#    Menu()
}

function dt_SetBackground(){
    dt_SetPersonal( "Background",Backgrounds,"dt_ChangeBackground" )
}

function dt_SetBackgroundEffect(){
    dt_SetPersonal( "Background Effects",BackgroundEffects,"dt_ChangeBackgroundEffect" )
}

function dt_SetTaskBar(){
    dt_SetPersonal( "TaskBar",TaskBars,"dt_ChangeTaskBar" )
}

function dt_SetColorSet(){
    dt_SetPersonal( "ColorSet",ColorSets,"dt_ChangeColorScheme" )
}
