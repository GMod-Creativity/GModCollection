@name DGPC-E-Backgrounds
@persist Backgrounds:table BackgroundEffects:table TaskBars:table ColorSets:table

#include "dgpc3/draw"

function table userDefaults(){
    local T = table()
    T["Background",string] = "bg_bluegrad"
    T["Background Effects",string] = "bge_spotted"
    T["TaskBar",string] = "tb_steelgrad"
    T["ColorSet",string] = "cs_Default"
    T["Gadgets",table] = table()
    return T
}

function addBackgorund( Name:string,AdminOnly,Func:string ){
    Backgrounds[Name,table] = table(0,"Function","dt_SetBackground",Func)
}

function addBackgroundEffect( Name:string,AdminOnly,Func:string ){
    BackgroundEffects[Name,table] = table(0,"Function","dt_SetBackgroundEffect",Func)
}

function addTaskBar( Name:string,AdminOnly,Func:string ){
    TaskBars[Name,table] = table(0,"Function","dt_SetTaskBar",Func)
}

function addColorSet( Name:string,AdminOnly,Func:string ){
    ColorSets[Name,table] = table(0,"Function","dt_SetColorSet",Func)
}

function none(){}


# BACKGROUNDS
addBackgorund("Blue Sky",0,"bg_bluegrad")
addBackgorund("Club Scene",0,"bg_RVB")
addBackgorund("Green Afternoon",0,"bg_GA")
addBackgorund("GM-Forest",0,"bg_GMForest")
addBackgorund("Box Fort",0,"bg_BoxFort")

function bg_bluegrad(){
    box(256,256,512,512,3)
    box(200,200,1000,1000,4) material("gui/gradient") rotation(-120)
}
function bg_RVB(){
    box(256,256,512,512,23)
    box(200,200,1000,1000,22) material("gui/gradient") rotation(-120)
}
function bg_GA(){
    box(256,256,512,512,25)
    box(200,200,1000,1000,24) material("gui/gradient") rotation(-120)
}
function bg_GMForest(){
    box(256,256,512,512,2) material("010001backgrounds/gm_forest")
}
function bg_BoxFort(){
    local T = array( vec2(0,0),vec2(1,1),vec2(1,0) )
    local Power = 6
    
    box(256,256,512,512,1)
    
    local X = 0 local Y = 0
    local C = 255/Power
    local P = 512/Power
    
    for( I=1,Power^2 ){
        
        box(X*P+P/2,Y*P+P/2,P+2,P+2,1) color(vec(X*C,255-Y*C,Y*C))
        
        X = X+1
        if( X >= Power ){
            X = 0
            Y = Y+1
        } 
    }

}



# BACKGROUND EFFECTS
addBackgroundEffect("None",0,"none")
addBackgroundEffect("Spotted Clouds",0,"bge_spotted")
addBackgroundEffect("Boxed",0,"bge_boxes")
addBackgroundEffect("Strike 45's",0,"bge_lines")
addBackgroundEffect("Strikes",0,"bge_strikes")
addBackgroundEffect("DGPC3",0,"bge_dgpc")

function bge_spotted(){
    for(I=1,15){
        local X = randint(0,512)
        local Y = randint(0,512)
        local S = randint(50,100)
        circle(X,Y,S,S,2) alpha(10)
    }
}
function bge_lines(){
    for(I=1,15){
        local X = randint(0,512)
        local Y = randint(0,512)
        local S = randint(80,350)
        circle(X,Y,2,S,2) alpha(10) rotation( 45 )
    }
}
function bge_strikes(){
    for(I=1,15){
        local X = randint(0,512)
        local Y = randint(0,512)
        local S = randint(80,350)
        local R = randint(30,80)
        circle(X,Y,2,S,2) alpha(10) rotation( R )
    }
}
function bge_boxes(){
    for(I=1,15){
        local X = randint(0,512)
        local Y = randint(0,512)
        local S = randint(100,180)
        box(X,Y,S,S,2) alpha(10)
    }
}
function bge_dgpc(){
    text("DGPC3",256,200,80,2,1,1) alpha(150)
    text("By DGrouse Industries",256,250,30,2,1,1) alpha(100)
}
