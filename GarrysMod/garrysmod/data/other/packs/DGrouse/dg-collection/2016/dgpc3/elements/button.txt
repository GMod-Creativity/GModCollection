@name DGPC3-E-Button
@persist ButtonIndex Buttons_Hooked ActiveButtons:array [Buttons ButtonClicks]:table

#include "dgpc3/hooks"
#include "dgpc3/draw"
#include "dgpc3/math"

function buttonCheck(){
    if( ActiveButtons:count() > 0 ){
        if( !Buttons_Hooked ){
            hookAdd("ButtonHighlight","Highlight","buttonHighlight",0)
            Buttons_Hooked = 1
        }
    }else{
        if( Buttons_Hooked ){
            hookRemove("ButtonHighlight")
            Buttons_Hooked = 0
        }
    }
}

function number buttonAdd(Shape,Name:string,X,Y,W,H,Color,HighlightColor,Func:string,Active){
    
    ButtonIndex = ButtonIndex+10
    
    if( Active ){
        ActiveButtons:pushNumber( ButtonIndex )
        buttonCheck()
    }
    
    Buttons[ButtonIndex,array] = array(Shape,X,Y,W,H,Color,HighlightColor,Name,Func,Active)
    
    return ButtonIndex
}

function number buttonAddBox(Name:string,X,Y,W,H,Color,HighlightColor,Func:string,Active){
    return buttonAdd(box(X,Y,W,H,Color),Name,X,Y,W,H,Color,HighlightColor,Func,Active)
}

function buttonSetActive(Ind,Active){
    local A = Buttons[Ind,array]
    
    if( Active ){
        ActiveButtons:pushNumber(Ind)
    }else{
        foreach(K,V:number = ActiveButtons ){
            if( V == Ind ){
                ActiveButtons:remove(Ind)
                break
            }
        }
    }
    buttonCheck()
    A[7,number] = Active
}

function buttonSetColors(Ind,Color,HLColor){
    Buttons[Ind,array][6,number] = Color
    Buttons[Ind,array][7,number] = HLColor
}

function buttonSetColor(Ind,Color){
    local A = Buttons[Ind,array]
    color(A[1,number],Color)
}

function number cbetween(X,Y,W,H){
    local P = cpos()
    if( between(P:x(),X-W/2,X+W/2) && between(P:y(),Y-H/2,Y+H/2) ){
        return 1
    }
    return 0
}

function addClicks(Name:string,Func:string){
    if( !ButtonClicks[Name,number] ){
        ButtonClicks[Name,number] = 1
        hookAdd("LeftClick",Name,Func)
    }
}

function removeClicks(Name:string,Func:string){
    if( ButtonClicks[Name,number] ){
        ButtonClicks[Name,number] = 0
        hookRemoveIdentifier("LeftClick",Name)
    }
}

function buttonHighlight(){
    foreach(K,V:number = ActiveButtons){
        local A = Buttons[V,array]
        local B = A[1,number]
        local X = A[2,number]
        local Y = A[3,number]
        local W = A[4,number]
        local H = A[5,number]
        local C = A[6,number]
        local HC = A[7,number]
        local Name = A[8,string]
        local Func = A[9,string]
        if( cbetween(X,Y,W,H) ){
            color(B,HC)
            addClicks(Name,Func)
        }else{
            color(B,C)
            removeClicks(Name,Func)
        }
    }
    timer("ButtonHighlight",50)
}

function buttonRemove(Ind){
    local A = Buttons[Ind,array]
    remove( A[1,number] )
    removeClicks(A[8,string],A[9,string])
    foreach(K,V:number = ActiveButtons){
        if( Ind == V ){
            ActiveButtons:remove(K)
        }
    }
    Buttons:remove( Ind )
}

function buttonRemove(A:array){
    foreach(K,V:number = A){
        buttonRemove(V)
    }
}

function buttonRemove(T:table){
    foreach(K,V:number = T){
        buttonRemove(V)
    }
}

function clearButtons(){
    ActiveButtons:clear()
    Buttons:clear()
    ButtonIndex = 0
    buttonCheck()
}
