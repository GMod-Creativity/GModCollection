@name DGPC3-E-Cursor
@persist [MouseShape Mouse_Default Cursor]:array

#include "dgpc3/draw"

Mouse_Default = array(
    vec2(0,0),
    vec2(0,30),
    vec2(9,24),
    vec2(14,35),
    vec2(17,34),
    vec2(12,24),
    vec2(21,24)
)

MouseShape = Mouse_Default:clone()

function cursorShape(A:array){
    MouseShape = A:clone()
}

function cursorRemove(){
    if( Cursor ){
        remove( array( Cursor[1,vector2]:x(),Cursor[1,vector2]:y(),Cursor[2,vector2]:x(),Cursor[2,vector2]:y() ) )
    }
}

function cursor(){
    
    cursorRemove()
    
    local S = polygon(MouseShape,1.2,1,-1,1) alpha(230)
    local C = polygon(MouseShape,1,0,0,2)
    Cursor = array( S,C )

    EGP:egpParent(S:y(),C:y())
    EGP:egpParentToCursor( C:y() )
}
