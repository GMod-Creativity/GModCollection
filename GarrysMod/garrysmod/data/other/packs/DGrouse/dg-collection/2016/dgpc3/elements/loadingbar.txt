@name DGPC-E-LoadingBar
@persist LoadingBars:table LoadingBarIndex

#include "dgpc3/draw"

function number loadingBarInit(X,Y,W,H,S,C1,C2,C3,P){
    LoadingBarIndex = LoadingBarIndex+10
    
    local O = box(X,Y,W,H,C1)
    local B = box(X,Y,W-S,H-S,C2)
    local L = box(ceil( X-W*(1-P)/2 ),Y,W*P-S,H-S,C3)
    
    LoadingBars[LoadingBarIndex,array] = array(O,B,L,X,Y,W,H,S)
    
    return LoadingBarIndex
}

function loadingBarPercent(I,P){
    local A = LoadingBars[I,array]
    local L = A[3,number]
    local X = A[4,number]
    local Y = A[5,number]
    local W = A[6,number]
    local H = A[7,number]
    local S = A[8,number]
    
    EGP:egpSize(L,vec2(W*P-S,H-S))
    EGP:egpPos(L,vec2(ceil( X-W*(1-P)/2 ),Y))
}

function loadingBarRemove(Ind,Bool){
    if( Bool ){
        for(I=1,3){
            EGP:egpRemove(LoadingBars[Ind,array][I,number])
        }
    }
    LoadingBars:remove(Ind)
}

function clearLoadingBars(){
    LoadingBars:clear()
    LoadingBarIndex = 0
}
