@name DGPC3-E-Window
@persist Windows:table WindowsIndex

#include "dgpc3/draw"
#include "dgpc3/elements/button"
#include "dgpc3/elements/textbox"

function number windowNew(X,Y,W,H){
    WindowsIndex = WindowsIndex+10
    local O = roundedbox(X,Y,W,H,1,8)
    local I = roundedbox(X,Y,W-12,H-12,2,6)
    Windows[WindowsIndex,table] = table()
    Windows[WindowsIndex,table]["Objects",array] = array(O,I)
    Windows[WindowsIndex,table]["Buttons",array] = array()
    return WindowsIndex
}

function windowContent(Ind,Type:string,Content){
    Windows[Ind,table][Type,array]:pushNumber( Content )
}

function windowContent(Type:string,Content){
    Windows[WindowsIndex,table][Type,array]:pushNumber( Content )
}

function windowContent(Ind,Type:string,Content:array){
    Windows[Ind,table][Type,array] = Windows[Ind,table][Type,array]:add( Content )
}

function windowContent(Type:string,Content:array){
    Windows[WindowsIndex,table][Type,array] = Windows[WindowsIndex,table][Type,array]:add( Content )
}

function array windowGetContent(Ind,Type:string){
    return Windows[Ind,table][Type,array]
}

function array windowGetContent(Type:string){
    return Windows[WindowsIndex,table][Type,array]
}

function windowClose(Ind){    
    remove(Windows[Ind,table]["Objects",array])
    buttonRemove(Windows[Ind,table]["Buttons",array])
    textboxRemove(Windows[Ind,table]["Textboxes",array])
    Windows:remove(Ind)
}

function clearWindows(){
    Windows:clear()
    WindowsIndex = 0
}

function number windowNew(X,Y,W,H,Close:string){
    local Wind = windowNew(X,Y,W,H)
    local BX = X+W/2-25 local BY = Y-H/2+25 local BS = 20
    windowContent("Buttons",buttonAdd(box(BX,BY,BS,BS,12),"WC:"+Close,BX,BY,BS,BS,12,13,Close,1) )
    return Wind
}
