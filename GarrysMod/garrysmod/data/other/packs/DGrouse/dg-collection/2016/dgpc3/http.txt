@name HTTP
@persist HTTPReturnFunc:string

runOnHTTP(1)

function httpReq( Link:string, Func:string ){
    httpRequest( Link )
    HTTPReturnFunc = Func
}

function runHTTP(){
    if( httpClk() ){
        if( HTTPReturnFunc != "" ){
            HTTPReturnFunc( httpData() )
        }
    }
}

function clearHTTP(){
    HTTPReturnFunc = ""
}
