@name DGPC3-Keyboard
@persist [KeyboardHooks UnKeyboardHooks]:table LastMemory

#include "dgpc3/math"
#include "dgpc3/lib1"
#include "dgpc3/inputs"

function keyboardAdd(Name:string,Func:string,Filter:string){
    KeyboardHooks[Name,array] = array()
    KeyboardHooks[Name,array][1,string] = Func
    KeyboardHooks[Name,array][2,string] = Filter
}

function keyboardRemove(Name:string){
    KeyboardHooks:removeArray(Name)
    print( Name )
}

function unkeyboardAdd(Name:string,Func:string,Filter:string){
    UnKeyboardHooks[Name,array] = array()
    UnKeyboardHooks[Name,array][1,string] = Func
    UnKeyboardHooks[Name,array][2,string] = Filter
}

function unkeyboardRemove(Name:string){
    UnKeyboardHooks:removeArray(Name)
}

function number kFilter_None( Mem ){
    return 1
}

function number kFilter_LetNum( Mem ){
    if( any(Mem,array(127,10,32,9)) ){
        return 1
    }
    elseif( between( Mem,65,90 ) || between( Mem,97,122 ) || between( Mem,48,57 ) ){
        return 1
    }
    else{ return 0 }
}

function number kFilter_LetNumSym( Mem ){
    if( any(Mem,array(127,10,32,9)) ){
        return 1
    }
    elseif( between( Mem,65,90 ) || between( Mem,97,125 ) || between( Mem,33,57 ) ){
        return 1
    }
    else{ return 0 }
}

function number kFilter_LetNumSymShift( Mem ){
    return kFilter_LetNumSym( Mem ) || Mem == 154
}

function keyboardDebug( Mem ){
    print( Mem+" : "+toChar(Mem) )
}

function runKeyboardHooks(){
    if( Memory ){
        LastMemory = Memory
        foreach(K,V:array = KeyboardHooks){
            if( V[2,string]( Memory )[number] ){
                V[1,string]( Memory )
            }
        }
    }
    if( !Memory ){
        foreach(K,V:array = UnKeyboardHooks){
            if( V[2,string]( LastMemory )[number] ){
                V[1,string]( LastMemory )
            }
        }
    }
}
