function clearAll(){
    clearWindows()
    clearLoadingBars()
    clearTextBoxes()
    clearButtons()
    clearHooks()
    clearHTTP()
}

function table privlageFix(Tab:table){
    if( Users[CurrentUser,table]["IsAdmin",number] ){ return Tab }
    
    local T = Tab:clone()
    foreach(K,V:table = Tab ){
        if( V[1,number] ){
            T:remove(K)
        }
    }
    return T
}

function number createmenu(X,PY,W,PH,T:table,Head:string,Close:string){
    
    Table = privlageFix(T)
    
    local C = Table:count()
    local Y = PY-C*20
    local H = PH+C*40
    local Window = 0
    
    if( Close == "" ){
        Window = windowNew(X,Y,W,H)
    }else{
        Window = windowNew(X,Y,W,H,Close)
    }
        windowContent("Objects",text(Head,X,Y-H/2+24,30,1,1,1) )
    
    local SX = X
    local SW = 180
    local SH = 32
    local Count = 0
    
    foreach(K,V:table = Table){
        Count++
        local Type = V[2,string]
        local SY = Y-H/2+25+40*Count
        
        if( Type == "SubMenu" ){
            windowContent("Buttons",buttonAdd(box(SX,SY,SW,SH,13),"Menu"+K,SX,SY,SW,SH,13,14,"openSubMenu",1) )
            
        }elseif( Type == "Function" ){
            local Func = V[3,string]
            windowContent("Buttons",buttonAdd(box(SX,SY,SW,SH,13),"Menu"+K,SX,SY,SW,SH,13,14,Func,1) )
        }
        windowContent("Objects",text(K,SX,SY,25,1,1,1) )
    }
    
    cursor()
    
    return Window
}
function number createmenu(X,PY,W,PH,T:table,Head:string){
    return createmenu(X,PY,W,PH,T,Head,"")
}
