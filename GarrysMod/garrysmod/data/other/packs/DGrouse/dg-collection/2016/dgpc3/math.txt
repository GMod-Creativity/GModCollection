@name DGPC3-Math

function number between(X,Y,Z){
    if( X >= Y && X <= Z ){
        return 1
    } return 0
}

function number any( X,A:array ){
    foreach(K,V:number = A){
        if( X == V ){
            return 1
        }
    }
    return 0
}

function number array:counter(){ # Counts unordered arrays
    local Count = 0
    foreach(K,V:number = This){
        Count++
    }
    return Count
}

function table table:sort(){
    local N = This:count()
    local A = This:values()
    local Tf = This:flip()
    while(N > 2){
        for(I = 1, N){
            if(A[I,number] > A[I+1,number]){
                Temp = A[I,number]
                A[I,number] = A[I+1,number]
                A[I+1,number] = Temp
            }
        }
        N--
    }
   
    local T = array()
    foreach(K,V:number=A){
        T:pushString(Tf[V,string])
    }

    return table( A,T )
}
