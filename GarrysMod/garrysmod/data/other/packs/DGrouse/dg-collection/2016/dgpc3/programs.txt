@name DGPC-Programs
@persist [ProgramsOptions]:table

function programMount( Name:string,InitFunc:string,AdminOnly ){
    ProgramsOptions[Name,table] = table(AdminOnly,"Function",InitFunc)    
}

function mountPrograms(){
    programMount( "MineSweeper","initMineSweeper",0 )
    #include "dgpc3/programs/minesweeper/init"
    
    programMount( "Pong","initPong",0 )
    #include "dgpc3/programs/pong/init"
    
    programMount( "Definer","initDefiner",0 )
    #include "dgpc3/programs/definer/init"
}
