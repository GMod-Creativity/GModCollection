@name Definer - Init

function df_unincludes(){
    #include "dgpc3/elements/textbox"
    #include "dgpc3/elements/window"
    #include "dgpc3/elements/button"
    #include "dgpc3/lib2"
}

function df_includes(){
    #include "dgpc3/programs/definer/page"
}

function closeDefiner(){
    clearAll()
    "desktopInit"()
}

function df_background(){
    clear()
    box(256,256,512,512,3)
    box(256,256,750,750,1) rotation(-45) material("gui/gradient")
    box(256,256,750,750,1) rotation(45) material("gui/gradient")
    box(256,256,750,750,1) rotation(225) material("gui/gradient")
    box(256,256,750,750,1) rotation(135) material("gui/gradient")
    textshadow("Define:",256,2,60,2,1,0,2,1,255)
    box(256,64,300,4,2)
}

function initDefiner(){

    df_includes()
    clearAll()
    "toggleStartMenu"()
    
    ColorSet = array(
        vec(0,0,0),       #1 Black
        vec(255,255,255), #2 White
        vec(200,200,200) #3 BackGround
    )
    
    df_background()
    df_InitPage()
}
