@name MineSweeper - Init

function ms_unincludes(){
    #include "dgpc3/elements/loadingbar"
    #include "dgpc3/elements/textbox"
    #include "dgpc3/elements/window"
    #include "dgpc3/elements/button"
    #include "dgpc3/lib2"
}

function ms_includes(){
    #include "dgpc3/programs/minesweeper/variables"
    #include "dgpc3/programs/minesweeper/menu"
    #include "dgpc3/programs/minesweeper/game"
}

function closeMineSweeper(){
    clearAll()
    "desktopInit"()
}

function ms_background(){
    clear()
    box(256,256,600,512,3) rotation(90) material("gui/gradient")
}

function initMineSweeper(){
    
    ms_includes()
    clearAll()
    "toggleStartMenu"()
    
    ColorSet = array(
        vec(0,0,0),       #1 Black
        vec(255,255,255), #2 White
        vec(0,0,50),      #3 Gradient
        vec(255,50,50),   #4 Mine
        vec(200,200,200), #5 Selected
        # Number Colors
        vec(100,100,255), #6 - 1 Blue
        vec(100,255,100), #7 - 2 Green
        vec(255,100,100), #8 - 3 Red
        vec(50,50,150),   #9 - 4 Blue2
        vec(150,50,50),   #10 - 5 Red2
        vec(50,100,255),  #11 - 6 Aqua
        vec(255,50,200),  #12 - 7 Purple
        vec(0,0,0),       #13 - 8 Black
        # Menu
        vec(90,90,90),    #14 Button
        vec(255,150,0)    #15 Button Highlight
    )
    
    ms_background()
    ms_MainMenu()
    
}
