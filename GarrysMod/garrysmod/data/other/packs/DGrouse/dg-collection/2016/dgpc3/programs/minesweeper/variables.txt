@name MineSweeper - Variables
@persist [MS_Var]:table

function number ms_num(V:string){ return MS_Var[V,number] }
function ms_num(V:string,N){ MS_Var[V,number] = N }

function array ms_arr(V:string){ return MS_Var[V,array] }
function ms_arr(V:string,A:array){ MS_Var[V,array] = A }
function ms_arrAdd(V:string,A:array){ MS_Var[V,array] = MS_Var[V,array]:add(A) }
function ms_arrSet(V:string,Key,Value){ MS_Var[V,array][Key,number] = Value }

function table ms_tbl(V:string){ return MS_Var[V,table] }
function ms_tbl(V:string,T:table){ MS_Var[V,table] = T }
function ms_tblAdd(V:string,T:table){ MS_Var[V,table] = MS_Var[V,table]:add(T) }
function ms_tblSet(V:string,Key:string,Value){ MS_Var[V,table][Key,number] = Value }

function string ms_str(V:string){ return MS_Var[V,string] }
function ms_str(V:string,S:string){ MS_Var[V,string] = S }
