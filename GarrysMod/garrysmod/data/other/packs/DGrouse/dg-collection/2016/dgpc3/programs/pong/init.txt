@name Pong - Init

function pg_unincludes(){
    #include "dgpc3/elements/loadingbar"
    #include "dgpc3/elements/textbox"
    #include "dgpc3/elements/window"
    #include "dgpc3/elements/button"
    #include "dgpc3/elements/cursor"
    #include "dgpc3/clicks"
    #include "dgpc3/lib2"
    #include "dgpc3/math"
}

function pg_includes(){
    #include "dgpc3/programs/pong/variables"
    #include "dgpc3/programs/pong/menu"
    #include "dgpc3/programs/pong/game"
}

function closePong(){
    clearAll()
    "desktopInit"()
}

function pg_star(){
    
    pg_num("Stars",pg_num("Stars")+1)
    
    local X = randint(0,512)
    local Y = randint(0,512)
    local S = randint(1,2)
    local A = randint(10,150)
    box(X,Y,S,S,2) alpha(A)
    
    if( pg_num("Stars") < 100 ){
        timer("PG:Stars",10)
    }else{
        hookRemove("PG:Stars")
        pg_MainMenu()
    }
}

function pg_background(){
    clear()
    box(256,256,600,512,3) rotation(90) material("gui/gradient")
    box(256,456,400,512,4) rotation(90) material("gui/gradient")
    pg_num("Stars",0)
    hookAdd("PG:Stars","stars","pg_star",5)
}

function initPong(){
    
    pg_includes()
    
   "toggleStartMenu"()
    clearAll()
    clear()
    
    ColorSet = array(
        vec(0,0,0),       #1 Black
        vec(255,255,255), #2 White
        vec(40,0,50),     #3 Gradient
        vec(50,0,0),      #4 Gradient 2
        vec(90,90,90),    #5 Button
        vec(255,150,0),   #6 Button Highlight
        
        vec(0,0,255),     #7 Player
        vec(255,0,0),     #8 Opponent
        vec(0,255,0),     #9 Ball
        
        vec(200,130,90), #10 Selected
        vec(255,180,40)  #11 Selected Highlight
    )
    
    if( !pg_tbl("Scores") ){
        pg_tbl("Scores",table())
    }
    
    pg_background()
    
}
