@name Pong - Variables
@persist [PG_Var]:table

function number pg_num(V:string){ return PG_Var[V,number] }
function pg_num(V:string,N){ PG_Var[V,number] = N }

function array pg_arr(V:string){ return PG_Var[V,array] }
function pg_arr(V:string,A:array){ PG_Var[V,array] = A }
function pg_arrAdd(V:string,A:array){ PG_Var[V,array] = PG_Var[V,array]:add(A) }
function pg_arrSet(V:string,Key,Value){ PG_Var[V,array][Key,number] = Value }

function table pg_tbl(V:string){ return PG_Var[V,table] }
function pg_tbl(V:string,T:table){ PG_Var[V,table] = T }
function pg_tblAdd(V:string,T:table){ PG_Var[V,table] = PG_Var[V,table]:add(T) }
function pg_tblSet(V:string,Key:string,Value){ PG_Var[V,table][Key,number] = Value }

function string pg_str(V:string){ return PG_Var[V,string] }
function pg_str(V:string,S:string){ PG_Var[V,string] = S }
