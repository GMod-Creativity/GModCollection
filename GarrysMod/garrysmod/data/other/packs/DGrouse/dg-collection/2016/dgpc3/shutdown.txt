@name DGPC-Shutdown
@persist DGPC_On ShutDownTXT ShutdownWait

#include "dgpc3/bootstrap"
#include "dgpc3/inputs"
#include "dgpc3/draw"
#include "dgpc3/elements/loadingbar"
#include "dgpc3/elements/textbox"
#include "dgpc3/elements/window"

function shutdown(){
    clear()
    DGPC_On = 0
    ShutDownTXT = 0
    StartMenuOpen = 0
    defcolor()
    clearWindows()
    clearLoadingBars()
    clearTextBoxes()
    clearButtons()
    clearHooks()
}

function runPower(){
    if( inputClk() ){
        if( Power ){
            if( !DGPC_On ){
                DGPC_On = 1
                bootstrap()
            }else{
                ShutdownWait = 1
                hookAdd("ShutdownWait","sd","shutdown",3000)
                ShutDownTXT = text("Shutting Down",256,10,30,9,1,0)
            }
        }
        if( !Power ){
            if( ShutdownWait ){
                ShutdownWait = 0
                remove(ShutDownTXT)
                hookRemove("ShutdownWait")
                stoptimer("ShutdownWait")
            }
        }
    }
}
