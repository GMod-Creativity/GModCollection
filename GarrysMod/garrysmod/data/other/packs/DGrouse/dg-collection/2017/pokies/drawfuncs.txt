@name Draw Functions
@persist Index
@persist [ColorSet CS_Default]:array
@persist EGP:wirelink User:entity
@persist CurrentFont:string

CS_Default = array(
    vec(0,0,0),       #1 Black
    vec(255,255,255), #2 White
    vec(90,160,90),   #3 Background Base
    vec(20,40,20),    #4 Background Gradient
    vec(210,190,80),  #5 Outlines
    
    vec(40,40,40),    #6 Spinner Background
    vec(255,255,255), #7 Winning Background
    
    vec(255,100,20),  #8 Selection Box
    vec(150,255,80)   #9 Box Lock
)

function setEGP( Screen:wirelink ){
    EGP = Screen
}

function setUser( Player:entity ){
    User = Player
}

function setColorSet( CS:array ){
    ColorSet = CS:clone()
}

function defcolor(){
    ColorSet = CS_Default:clone()
}

defcolor()

function clear(){
    EGP:egpClear()
    Index = 0
}

function rotation(I,A){
    EGP:egpAngle(I,A)
}

function rotation(A){
    EGP:egpAngle(Index,A)
}

function material(I,M:string){
    EGP:egpMaterial(I,M)
}

function material(M:string){
    EGP:egpMaterial(Index,M)
}

function alpha(I,A){
    EGP:egpAlpha(I,A)
}

function alpha(A){
    EGP:egpAlpha(Index,A)
}

function position(I,X,Y){
    EGP:egpPos(I,vec2(X,Y))
}

function position(X,Y){
    EGP:egpPos(Index,vec2(X,Y))
}

function vector2 position(I){
    return EGP:egpPos(I)
}

function move(I,X,Y){
    local P = position(I)
    position(I,P:x()+X,P:y()+Y)
}

function color( I,C:vector ){
    EGP:egpColor(I,C)
}

function color( C:vector ){
    color(Index,C)
}

function color( I,X ){
    if( X > 0 ){
        EGP:egpColor(I,ColorSet[X,vector])
    }else{
        EGP:egpAlpha(I,0)
    }
}

function number line(X,Y,X2,Y2,Color,Size){
    Index++
    EGP:egpLine(Index,vec2(X,Y),vec2(X2,Y2))
    EGP:egpSize(Index,Size)
    color( Index,Color )
    return Index
}

function number box(X,Y,W,H,Color){
    Index++
    EGP:egpBox(Index,vec2(X,Y),vec2(W,H))
    color( Index,Color )
    return Index
}

function number boxoutline(X,Y,W,H,Color,S){
    Index++
    EGP:egpBoxOutline(Index,vec2(X,Y),vec2(W,H))
    EGP:egpSize(Index,S)
    color( Index,Color )
    return Index
}

function number circle(X,Y,W,H,Color){
    Index++
    EGP:egpCircle(Index,vec2(X,Y),vec2(W,H))
    color( Index,Color )
    return Index
}

function number roundedbox(X,Y,W,H,Color,Radius){
    Index++
    EGP:egpRoundedBox(Index,vec2(X,Y),vec2(W,H))
    EGP:egpRadius(Index,Radius)
    color( Index,Color )
    return Index
}

function number roundedboxoutline(X,Y,W,H,Color,Radius,S){
    Index++
    EGP:egpRoundedBoxOutline(Index,vec2(X,Y),vec2(W,H))
    EGP:egpRadius(Index,Radius)
    EGP:egpSize(Index,S)
    color( Index,Color )
    return Index
}

function number text(Text:string,X,Y,Size,Color,AX,AY){
    Index++
    EGP:egpText(Index,Text,vec2(X,Y))
    EGP:egpSize(Index,Size)
    EGP:egpAlign(Index,AX,AY)
    EGP:egpFont(Index,CurrentFont)
    color( Index,Color )
    return Index
}

function number text(Text:string,X,Y,Size,Color){
    return text(Text,X,Y,Size,Color,1,1)
}

function vector2 textshadow(Text:string,X,Y,Size,Color,AX,AY,Dist,ShadowColor,ShadowAlpha){
    local S = text(Text,X+Dist,Y+Dist,Size,ShadowColor,AX,AY)
    alpha(ShadowAlpha)
    local T = text(Text,X,Y,Size,Color,AX,AY)
    return vec2(T,S)
}

function vector2 textshadow(Text:string,X,Y,Size,Color,Dist,ShadowColor,ShadowAlpha){
    return textshadow(Text,X,Y,Size,Color,1,1,Dist,ShadowColor,ShadowAlpha)
}

function font( Font:string ){
    CurrentFont = Font
}

function text(Ind,Text:string){
    EGP:egpSetText(Ind,Text)
}

function vector2 polygon(Shape:array,Scale,X,Y,Color){
    Index++
    local Controller = box(0,0,Scale,Scale,-1)
    
    local S = array()
    foreach(K,V:vector2=Shape){
        S[K,vector2] = V*Scale
    }
    
    Index++
    EGP:egpPoly(Index,S)
    EGP:egpParent(Index,Controller) 
    
    EGP:egpPos(Controller,vec2(X,Y))

    
    color( Index,Color )
    return vec2(Index,Controller)
}

function vector2 cpos(){
    return EGP:egpCursor(User)
}

function remove(Ind){
    EGP:egpRemove(Ind)
}

function remove(A:array){
    foreach(K,V:number = A){
        EGP:egpRemove(V)
    }
}
