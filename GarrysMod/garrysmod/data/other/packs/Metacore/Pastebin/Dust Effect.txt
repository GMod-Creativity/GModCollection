@name Dust Effect
@inputs Base:entity
@persist Effect:effect
@trigger none
 
if(first() | dupefinished())
{
     
    rangerHitWater(1)
    rangerHitEntities(0)
    rangerPersist(1)
    
    Effect = effect() 

}
 
interval(45)

Ranger = rangerOffset(999999,Base:toWorld(vec(0,0,0)),vec(0,0,-1))

Effect:setOrigin(Ranger:pos()) 
Effect:setScale(130) # 180 is max
Effect:play("ThumperDust")
