@name Podracer Cables and Power Coupling
@inputs [Cable_Left_A Cable_Left_B Cable_Right_A Cable_Right_B Coupling_A Coupling_B]:entity
@persist Cable_Width Coupling_Width
@trigger none

if(first() | dupefinished())
{        
    
    # Place the A blocks on the cockpit.
    # Place the B blocks on the engines.

    Cable_Width     = 0.18
    Coupling_Width  = 0.425
    
    Cable_Material = "models/props_forest/FurnacePipes01"

    holoCreate(1)
    holoParent(1,Cable_Left_A)
    holoModel(1,"cylinder")
    holoColor(1,vec(255))
    holoMaterial(1,Cable_Material)
    Cable_Ang_L   = (Cable_Left_A:pos() - Cable_Left_B:pos()):toAngle() + ang(90,0,0)
    Cable_Pos_L   = (Cable_Left_A:pos() + Cable_Left_B:pos())/2
    Cable_Scale_L = (Cable_Left_A:pos() - Cable_Left_B:pos()):length()/12
    holoPos(1,Cable_Pos_L)
    holoAng(1,Cable_Ang_L)
    holoScale(1,vec(Cable_Width,Cable_Width,Cable_Scale_L))
    
    holoCreate(2)
    holoParent(2,Cable_Right_A)
    holoModel(2,"cylinder")
    holoColor(2,vec(255))
    holoMaterial(2,Cable_Material)
    Cable_Ang_R   = (Cable_Right_A:pos() - Cable_Right_B:pos()):toAngle() + ang(90,0,0)
    Cable_Pos_R   = (Cable_Right_A:pos() + Cable_Right_B:pos())/2
    Cable_Scale_R = (Cable_Right_A:pos() - Cable_Right_B:pos()):length()/12
    holoPos(2,Cable_Pos_R)
    holoAng(2,Cable_Ang_R)
    holoScale(2,vec(Cable_Width,Cable_Width,Cable_Scale_R))
    
    holoCreate(3)
    holoParent(3,Coupling_A)
    holoModel(3,"models/sprops/geometry/sphere_24.mdl")
    holoColor(3,vec(255,188,238))
    holoMaterial(3,"models/alyx/emptool_glow")
    Coupling_Ang   = (Coupling_A:pos() - Coupling_B:pos()):toAngle() + ang(90,0,0)
    Coupling_Pos   = (Coupling_A:pos() + Coupling_B:pos())/2
    Coupling_Scale = (Coupling_A:pos() - Coupling_B:pos()):length()/12
    holoPos(3,Coupling_Pos)
    holoAng(3,Coupling_Ang)
    holoScale(3,vec(0.85,0.85,Coupling_Scale))

}

interval(45)

Cable_Ang_L    = (Cable_Left_A:pos() - Cable_Left_B:pos()):toAngle() + ang(90,0,0)
Cable_Pos_L    = (Cable_Left_A:pos() + Cable_Left_B:pos())/2
Cable_Scale_L  = (Cable_Left_A:pos() - Cable_Left_B:pos()):length()/12
holoPos(1,Cable_Pos_L)
holoAng(1,Cable_Ang_L)
holoScale(1,vec(Cable_Width,Cable_Width,Cable_Scale_L))

Cable_Ang_R    = (Cable_Right_A:pos() - Cable_Right_B:pos()):toAngle() + ang(90,0,0)
Cable_Pos_R    = (Cable_Right_A:pos() + Cable_Right_B:pos())/2
Cable_Scale_R  = (Cable_Right_A:pos() - Cable_Right_B:pos()):length()/12
holoPos(2,Cable_Pos_R)
holoAng(2,Cable_Ang_R)
holoScale(2,vec(Cable_Width,Cable_Width,Cable_Scale_R))

Coupling_Ang    = (Coupling_A:pos() - Coupling_B:pos()):toAngle() + ang(90,0,0)
Coupling_Pos    = (Coupling_A:pos() + Coupling_B:pos())/2
Coupling_Scale  = (Coupling_A:pos() - Coupling_B:pos()):length()/24
holoPos(3,Coupling_Pos)
holoAng(3,Coupling_Ang)
holoScale(3,vec(Coupling_Width,Coupling_Width,Coupling_Scale))
