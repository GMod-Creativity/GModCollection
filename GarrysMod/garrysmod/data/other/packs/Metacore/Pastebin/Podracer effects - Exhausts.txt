@name Podracer effects - Exhausts
@inputs Brake_Button Throttle_Button [Cockpit Exhaust_A Exhaust_B]:entity
@persist Exhaust_Color:vector Throttle_Counter Flame_Width Flame_Length Exhaust_Scale Exhaust_Spin [Thrust_Sound Brake_Sound]:string
@persist Brake_Volume
@trigger none
 
if(first() | dupefinished())
{        
 
    # Instructions
 
    # Place the Exhaust_A and Exhaust_B blocks inside the engine exhausts
 
    # Parent all effect blocks to their respective parts
 
    # Effect scales can be adjusted below
 
    Exhaust_Scale   = 0.375
    Exhaust_Color   = vec(255,229,200)
 
    Thrust_Sound    = "acf_extra/airfx/air_distant.wav"

    Brake_Sound     = "acf_extra/vehiclefx/misc/air_04.wav"
    Brake_Volume    = 0.25
 
    holoCreate(4)
    holoParent(4,Exhaust_A)
    holoModel(4,"models/sprops/geometry/fdisc_48.mdl")
    holoColor(4,Exhaust_Color)
    holoMaterial(4,"models/props_c17/industrialbellbottomon01")
    holoColor(4,vec(0))
    holoPos(4,Exhaust_A:toWorld(vec(0,0,0)))
    holoAng(4,Exhaust_A:toWorld(ang(0,0,0)))
    holoScale(4,vec(Exhaust_Scale,0.25,Exhaust_Scale))
 
    holoCreate(5)
    holoParent(5,Exhaust_B)
    holoModel(5,"models/sprops/geometry/fdisc_48.mdl")
    holoMaterial(5,"models/props_c17/industrialbellbottomon01")
    holoColor(5,vec(0))
    holoPos(5,Exhaust_B:toWorld(vec(0,0,0)))
    holoAng(5,Exhaust_B:toWorld(ang(0,0,0)))
    holoScale(5,vec(Exhaust_Scale,0.25,Exhaust_Scale))
 
    Flame_Width   = Exhaust_Scale*0.85
    Flame_Length  = Exhaust_Scale*1.5
 
    holoCreate(6)
    holoParent(6,Exhaust_A)
    holoModel(6,"models/sprops/misc/cones/size_4/cone_48x144.mdl")
    holoMaterial(6,"models/props_combine/tpballglow")
    holoColor(6,Exhaust_Color)
    holoPos(6,Exhaust_A:toWorld(vec(0,0,0)))
    holoAng(6,Exhaust_A:toWorld(ang(0,0,90)))
    holoScale(6,vec(0))
 
    holoCreate(7)
    holoParent(7,Exhaust_B)
    holoModel(7,"models/sprops/misc/cones/size_4/cone_48x144.mdl")
    holoMaterial(7,"models/props_combine/tpballglow")
    holoColor(7,Exhaust_Color)
    holoPos(7,Exhaust_B:toWorld(vec(0,0,0)))
    holoAng(7,Exhaust_B:toWorld(ang(0,0,90)))
    holoScale(7,vec(0))
 
    Cockpit:soundPlay(1,0,Thrust_Sound)
    soundVolume(1,0)
 
}
 
interval(60)
 
Throttle_Counter = clamp(Throttle_Counter + (Throttle_Button-!Throttle_Button*0.05),0,1)
if(changed(Throttle_Counter)){holoColor(4,Exhaust_Color*Throttle_Counter) holoColor(5,Exhaust_Color*Throttle_Counter) holoScale(6,vec(Flame_Width,Flame_Width,Flame_Length*Throttle_Counter)) holoColor(6,Exhaust_Color*Throttle_Counter) holoScale(7,vec(Flame_Width,Flame_Width,Flame_Length*Throttle_Counter)) holoColor(7,Exhaust_Color*Throttle_Counter) soundVolume(1,Throttle_Counter)}
Exhaust_Spin = Exhaust_Spin + 45
holoAng(6,Exhaust_A:toWorld(ang(Exhaust_Spin,0,90)))
holoAng(7,Exhaust_B:toWorld(ang(Exhaust_Spin,0,90)))
 
if((changed(Brake_Button) & Brake_Button)){Cockpit:soundPlay(2,1,Brake_Sound)  soundVolume(2,Brake_Volume)}
