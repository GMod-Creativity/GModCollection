@name 
@inputs Screen:wirelink On
@outputs 
@persist Form1 Form2 X Y RD:ranger RX RY F
@trigger Reset On
if (first()) {
print("Res: "+RX+"X"+RY)
concmd("ulx rcon wire_expression2_quotasoft 1000000000")
concmd("ulx rcon wire_expression2_quotahard 1000000000")
concmd("ulx rcon wire_expression2_quotatick 1000000000")
}


if (On & ~On) {
timer("A",1000)
RX = Screen[1048572]
RY = Screen[1048573]
}

if (clk("A")) {
F += 1
if (F == 55) {stoptimer("A")} else {timer("A",1000)}
for (Y = (F*10)+1, (F+1)*10) {
for (X = 0, RX) {
Form1 = X/(RX/2)-1
Form2 = Y/(RY/2)-1
RD = ranger(10000,Form1,Form2)
Screen[X+(Y*RX)] = (1000-(RD:distance()*1.14))
}
}
}
