@name 
@inputs 
@outputs 
@persist Ang:angle Array:array Ent:entity
@trigger 
runOnTick(1)

findByClass("prop_physics")
Array = findToArray()

Ent = Array[1,entity]

for (X = 2, Array:count()) {
holoCreate(X)
holoPos(X,(Ent:pos() + Array[X,entity]:pos())/2)
holoScaleUnits(X,(Ent:pos() - Array[X,entity]:pos()) * 1.024)
Ang = ang(0,0,0)
holoAng(X,Ang)
holoColor(X,vec(255,0,0),100)

Ent = Array[X,entity]
}
