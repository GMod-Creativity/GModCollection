@name 
@inputs Screen:wirelink
@outputs 
@persist String:string Table:table Check:string Color:string
@trigger 
String = "<red>Hi<br />"
Table["red",number] = 900
Table["green",number] = 90
Table["blue",number] = 9
Table["yellow",number] = 990
Table["white",number] = 999
Table["black",number] = 1

Array = String:explode("<br />")
Check = String:sub(String:find("<")+1,String:find(">")-1)
if (Check != "br /") {Color = Check} else {Color = Color}

for (X = 0,17) {
Screen:writeString(Array[X+1,string]:replace("<",""):replace(">",""):replace(Check,""),0,X,Table[Color,number])
}
