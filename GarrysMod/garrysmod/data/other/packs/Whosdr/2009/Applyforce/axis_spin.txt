@name 
@inputs Cargo:entity On
@outputs 
@persist X SIN COS G:vector A:vector B:vector C:vector D:vector Weld:entity Pos:vector
@trigger all
if (first()) {Pos = entity():isWeldedTo():pos() + vec(0,0,120)}
interval(10)
X += 1
SIN = sin(X) * 100
COS = cos(X) * 100
Weld = entity():isWeldedTo()

if (On) {Pos = owner():aimPos() + vec(0,0,65)}
Weld:applyForce(G * Weld:mass())
A = Pos
B = vec(0,0,0)
C = Weld:pos()
D = Weld:vel()

G = A + B - C - D * 0.2

Weld:applyAngForce(((-Weld:angVel() + -ang(Weld:angles():pitch(),owner():angles():yaw(),Weld:angles():roll()) + ang(0,0,0))*20)*Weld:mass())
