@name 
@inputs 
@outputs 
@persist X SIN COS G:vector A:vector B:vector C:vector D:vector Weld:entity Pos:vector
@trigger all
if (first()) {Pos = entity():isWeldedTo():pos()}
interval(10)
X += 1
SIN = sin(X) * 100
COS = cos(X) * 100
Weld = entity():isWeldedTo()

Weld:applyForce(G * Weld:mass())
A = Pos
B = vec(0,0,0)
C = Weld:pos()
D = Weld:vel()

G = A + B - C - D * 0.2

Weld:applyAngForce((-Weld:angVel() + -Weld:angles() + ang(SIN,SIN+COS,COS))*50)
