@name 
@inputs 
@outputs 
@persist G:vector A:vector B:vector C:vector D:vector
@persist Ent:entity X
@trigger all
runOnTick(1)

findIncludePlayerProps(owner())
findByClass("sent_ball")


for (X = 0, findToArray():count()) {
Ent = findResult(X)
A = owner():aimPos()
C = Ent:pos()
D = Ent:vel()
G = A - C - D * 0.2
if (owner():aimEntity() != Ent & owner():aimEntity():type() != Ent:type()) {
Ent:applyForce(G * Ent:mass())
}
}
