@name 
@inputs W A S D Bearing Player:entity
@outputs 
@persist Pos:vector Ent:entity Angle:angle
@trigger all
if (first() | duped()) {
Ent = entity():isWeldedTo()
Pos = Ent:pos()
Angle = Ent:angles()
holoCreate(1)
}

runOnTick(1)

if (W) {Pos += Player:driver():eye()*20}
if (S) {Pos -= Player:driver():eye()*20}

#Angle = ang(0,Bearing,0)

if (A) {Angle += ang(0,1,0)}
if (D) {Angle -= ang(0,1,0)}

Ent:applyForce((-Ent:vel() + -Ent:pos() + Pos)*Ent:mass())

Ent:applyAngForce(angnorm(-Ent:angVel() + -Ent:angles() + Angle)*Ent:mass())

if (Ent:isPlayerHolding()) {Pos = Ent:pos() Angle = Ent:angles()}
holoPos(1,Pos)
