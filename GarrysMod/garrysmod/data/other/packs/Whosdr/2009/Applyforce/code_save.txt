@name 
@inputs 
@outputs Name:string
@persist X SIN COS G:vector A:vector B:vector C:vector D:vector Player:entity P:array
@trigger all
interval(10)
runOnChat(1)
chatClk(owner())
if (first()) {X = 1}
if (!clk() & owner():lastSaid():lower() == "/next") {
if (X >= 0 & X < P:count()) {
X += 1
} else {
X = 1
}
findByClass("player")
P = findToArray()
Player = P:entity(X)
}
entity():isWeldedTo():applyForce(G * entity():isWeldedTo():mass())
A = Player:pos()
B = vec(0,0,150*(Player==owner()))
C = entity():pos()
D = entity():vel()

G = A + B - C - D * 0.2
Name = Player:name()
entity():applyAngForce(ang(1000,1000,1000))
