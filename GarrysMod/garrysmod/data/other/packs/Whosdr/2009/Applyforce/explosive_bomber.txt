@name 
@inputs 
@outputs 
@persist X SIN COS G:vector A:vector B:vector C:vector D:vector
@trigger all
interval(10)
X += 1
SIN = sin(X) * 100
COS = cos(X) * 100

entity():applyForce(G)
A = owner():aimPos()
B = vec(0,0,entity():vel():z())
C = entity():pos()
D = entity():vel()

G = A + B - C - D * 0.2
