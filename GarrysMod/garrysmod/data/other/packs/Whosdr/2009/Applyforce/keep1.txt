@name 
@inputs In
@outputs 
@persist X SIN COS G:vector A:vector B:vector C:vector D:vector
@trigger all
interval(10)

SIN = sin(curtime()*100)*(10*In)
COS = cos(curtime()*100)*(10*In)

entity():isWeldedTo():applyForce(G * entity():isWeldedTo():mass())
A = owner():pos()
B = vec(SIN,COS,65)
C = entity():pos()
D = entity():vel()

G = A + B - C - D * 0.2


entity():isWeldedTo():applyAngForce((ang(curtime()*1,curtime()*1,curtime()*1) - entity():angles() - entity():angVel() )/20)
