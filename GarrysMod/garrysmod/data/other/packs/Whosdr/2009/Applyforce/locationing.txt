@name 
@inputs Get Next Prev
@outputs Y Z:string
@persist X Pos:array Loc:vector Y Entity:entity
@trigger all
Entity = entity():isWeldedTo()
if (!clk("A")) {
if (Next & Y < Pos:count()) {Y += 1}
if (Prev & Y > 0) {Y -= 1}
}
timer("A",20)

if (Get & !clk("A")) {X += 1 Pos:setVector(X,owner():pos()+vec(0,0,50))}
Loc = Pos:vector(Y)
if (clk("A")) {
Entity:applyForce((-Entity:vel() + -Entity:pos() + Loc)*Entity:mass())
}
Z = toString(Y)

if (clk("A")) {
Entity:applyAngForce( ( (-Entity:angVel() + -Entity:angles() )*Entity:mass() )*20)
}
