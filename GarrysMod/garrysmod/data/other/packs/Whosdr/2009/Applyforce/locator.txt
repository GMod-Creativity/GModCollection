@name 
@inputs E:entity
@outputs Out:string
@persist G:vector A:vector B:vector C:vector D:vector
@persist Ent:entity X
@trigger all
interval(10)

findByClass("sent_ball")
if (X < findToArray():count()) {X += 1} else {X = 1}
Ent = findResult(X)


A = owner():pos()
B = (E:pos() + -owner():pos())/2


C = Ent:pos()
D = Ent:vel()
G = A + B - C - D * 0.2
Ent:applyForce(G * Ent:mass())


Out = E:name()
