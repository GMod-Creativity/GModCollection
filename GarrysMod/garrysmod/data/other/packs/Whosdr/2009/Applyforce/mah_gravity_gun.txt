@name 
@inputs 
@outputs 
@persist Ent:entity X Toggle Pos:vector
@trigger all
#runOnTick(1)
interval(10)
if (owner():aimEntity():pos() != vec(0,0,0) & owner():aimEntity():owner() == owner()) {Ent = owner():aimEntity()}

if (!Toggle) {

if (owner():keyAttack2()) {
Ent:applyForce((-Ent:vel() + -Ent:pos() + owner():pos() + vec(0,0,75) + (owner():eye()*50))*Ent:mass())
}

if (owner():keyAttack1()) {
Ent:applyForce((-Ent:vel() + -Ent:pos() + owner():pos() + vec(0,0,75) + (owner():eye()*X))*Ent:mass())
X += 10
} else {X = 0}

}

if (owner():keyUse()) {Toggle = !Toggle}

if (Toggle) {
if ($Toggle) {Pos = Ent:pos()}
Ent:applyForce((-Ent:vel() - Ent:pos() + Pos)*Ent:mass())
}
