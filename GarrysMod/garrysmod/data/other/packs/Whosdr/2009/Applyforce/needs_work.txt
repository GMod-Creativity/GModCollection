@name 
@inputs 
@outputs 
@persist Ents:array Ent:entity X Pos:vector Target:entity Attack
@trigger 
if (first()) {
findByClass("prop_physics")
findIncludePlayerProps(owner())
Ents = findToArray()
}
runOnTick(1)
runOnChat(1)

if (owner():lastSaid():explode(" "):string(1) == "!swarm" & chatClk(owner())) {
findByClass("player")
Target = findPlayerByName(owner():lastSaid():explode(" "):string(2))
Attack = 1
}

if (owner():lastSaid() == "!sleep") {
Attack = 0
#Target = noentity()
}


if (X < Ents:count()) {X += 1} else {X = 0}
Ent = Ents[X,entity]

if (Attack) {
Vector = vec(sin(curtime()*1000)*1000,cos(curtime()*1000)*1000,0)
Ent:applyForce((-Ent:pos() + -Ent:vel() + Target:pos() + vec(0,0,70) + Vector)*Ent:mass())
}
