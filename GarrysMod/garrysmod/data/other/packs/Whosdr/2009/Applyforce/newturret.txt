@name 
@inputs 
@outputs Pos:vector Name:string Fire
@persist Ent:entity Entity:entity EE:entity
@trigger 
runOnChat(1)
if (first()) {EE = findPlayerByName("whosdr")}
if (owner():lastSaid():explode(" "):string(1) == "!") {
Ent = findPlayerByName(owner():lastSaid():explode(" "):string(2))
if (Ent != noentity()) {Entity = Ent} else {Ent = Entity}
}

if (EE:lastSaid():explode(" "):string(1) == "!") {
Ent = findPlayerByName(EE:lastSaid():explode(" "):string(2))
if (Ent != noentity()) {Entity = Ent} else {Ent = Entity}
}


if (owner():lastSaid() == "!fire" | EE:lastSaid() == "!fire") {
if (Entity != EE) {Fire = 1} else {Fire = 0}
}

if (owner():lastSaid() == "!hold" | EE:lastSaid() == "!hold") {
Fire = 0
}




Pos = Entity:pos()
Name = Entity:name()
