@name 
@inputs 
@outputs 
@persist Ent:entity Constraints:array Mass X
@trigger 
if (first()) {Ent = entity():isWeldedTo() Mass = Ent:mass()
Constraints = Ent:getConstraints()
for (X = 0,Constraints:count()) {
Mass += Constraints[X,entity]:mass()
}
}
runOnTick(1)

Ent:applyForce( (-Ent:pos() + -Ent:vel() + owner():pos() + vec(0,0,100)) *Mass )
Ent:applyAngForce( (-Ent:angles() + -Ent:angVel() + angnorm(owner():angles())) *Mass)
