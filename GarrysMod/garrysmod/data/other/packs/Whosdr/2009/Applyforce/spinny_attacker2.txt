@name 
@inputs 
@outputs 
@persist Prop:entity Player:entity Mode On
@trigger all
runOnChat(1)
chatClk(owner())
interval(20)
if (first()) {
findByClass("prop_physics")
findClipToModel("models/mechanics/wheels/wheel_spike_48.mdl")
Prop = find()
On = 1
}
if (On) {
if (owner():lastSaid():lower() == "!!" & Mode != 0 & !clk("interval")) {
Target = owner()
Mode = 0
}
if (owner():lastSaid():explode(" "):string(1) == "!" & !clk("interval")) {
Player = findPlayerByName(owner():lastSaid():explode(" "):string(2))
Mode = 1
if (Player:pos() == vec(0,0,0) & Mode == 1) {print("Error: No target.") Mode = 0}
}

if (owner():lastSaid():explode(" "):string(1) == "#" & !clk("interval")) {
Player = findPlayerByName(owner():lastSaid():explode(" "):string(2))
Mode = 2
if (Player:pos() == vec(0,0,0) & Mode == 1) {print("Error: No target.") Mode = 0}
}



if (Mode == 2) {
Prop:applyForce((-Prop:pos() + -Prop:vel() + Player:pos() + vec(0,0,120-(owner():keyAttack2())*70)+Target:vel())*Prop:mass())
Prop:applyAngForce((-Prop:angles() + -Prop:angVel() + ang(0,curtime()*10,0))*Prop:mass())
}
elseif (Mode == 1) {
Prop:applyForce((-Prop:pos() + -Prop:vel() + Player:pos() + vec(0,0,50)+Target:vel())*Prop:mass())
Prop:applyAngForce((-Prop:angles() + -Prop:angVel() + ang(0,curtime()*10,0))*Prop:mass())
}
elseif (Mode == 0) {
Prop:applyForce((-Prop:pos() + -Prop:vel() + owner():pos() + vec(0,0,120)+Target:vel())*Prop:mass())
Prop:applyAngForce((-Prop:angles() + -Prop:angVel() + ang(0,0,0))*Prop:mass())
}
}


if (owner():lastSaid() == "!!!") {On = 0}
if (owner():lastSaid() == "!!!!") {On = 1}
