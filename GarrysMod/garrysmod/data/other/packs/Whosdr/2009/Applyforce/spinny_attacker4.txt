@name 
@inputs 
@outputs 
@persist Prop:entity Player:entity Mode On Pos:vector Cloak
@trigger all
runOnChat(1)
interval(100)
if (first() | duped()) {
findIncludePlayerProps(owner())
findByClass("prop_physics")
findClipToModel("models/props_phx/construct/windows/window_curve360x2.mdl")
Prop = find()
On = 1
Cloak = 1
}

if (owner():lastSaid():lower() == "*" & !clk()) {
Cloak = !Cloak * 255
Prop:setAlpha(Cloak)
}

if (On) {

if (owner():lastSaid():lower() == "!!" & Mode != 0 & !clk()) {
Target = owner()
Mode = 0
}
if (owner():lastSaid():explode(" "):string(1) == "!" & !clk()) {
Player = findPlayerByName(owner():lastSaid():explode(" "):string(2))
Mode = 1
if (Player:pos() == vec(0,0,0)) {print("Error: No target.") Mode = 0}
}

if (owner():lastSaid():explode(" "):string(1) == "#" & !clk()) {
Player = findPlayerByName(owner():lastSaid():explode(" "):string(2))
Mode = 2
if (Player:pos() == vec(0,0,0)) {print("Error: No target.") Mode = 0}
}


if (owner():lastSaid():explode(" "):string(1) == "$" & !clk()) {
Player = findPlayerByName("googlebum")
Mode = 3
}

if (owner():lastSaid():explode(" "):string(1) == "^" & !clk()) {
Mode = 4
}

if (Mode == 4) {
findInSphere(owner():pos(),1000)
findClipToClass("player")
findExcludePlayer("Whosdr")
if (find()) {
Player = find()} else {Player = owner()}
Prop:applyForce((-Prop:pos() + -Prop:vel() + Player:pos() + vec(0,0,60*(1+(Player == owner())))+Player:vel())*Prop:mass())
Prop:applyAngForce((-Prop:angles() + -Prop:angVel() + ang(0,curtime()*(Player != owner()),0))*Prop:mass())
}

if (Mode == 3) {
if (owner():keyAttack2()) {
Prop:applyForce((-Prop:pos() + -Prop:vel() + owner():aimPos() + vec(0,0,50))*Prop:mass())
Prop:applyAngForce((-Prop:angles() + -Prop:angVel() + ang(0,curtime(),0))*Prop:mass())
} else {
Prop:applyForce((-Prop:pos() + -Prop:vel() + owner():pos() + vec(0,0,120) + owner():vel())*Prop:mass())
Prop:applyAngForce((-Prop:angles() + -Prop:angVel() + ang(0,0,0))*Prop:mass())
}
}

if (Mode == 2) {
Prop:applyForce((-Prop:pos() + -Prop:vel() + Player:pos() + vec(0,0,120-(owner():keyAttack2())*70)+Player:vel())*Prop:mass())
Prop:applyAngForce((-Prop:angles() + -Prop:angVel() + ang(0,curtime(),0))*Prop:mass())
}
if (Mode == 1) {
Prop:applyForce((-Prop:pos() + -Prop:vel() + Player:pos() + vec(0,0,50)+Player:vel())*Prop:mass())
Prop:applyAngForce((-Prop:angles() + -Prop:angVel() + ang(0,curtime(),0))*Prop:mass())
}
if (Mode == 0) {
Prop:applyForce((-Prop:pos() + -Prop:vel() + owner():pos() + vec(0,0,150)+owner():vel())*Prop:mass())
Prop:applyAngForce((-Prop:angles() + -Prop:angVel() + ang(0,0,0))*Prop:mass())
}



}


if (owner():lastSaid() == "!!!") {On = 0}
if (owner():lastSaid() == "!!!!") {On = 1}



if (owner():lastSaid():index(1) == "!") {hideChat(1)}
elseif (owner():lastSaid():index(1) == "$") {hideChat(1)}
elseif (owner():lastSaid():index(1) == "^") {hideChat(1)}
elseif (owner():lastSaid():index(1) == "*") {hideChat(1)}
elseif (owner():lastSaid():index(1) == "#") {hideChat(1)}
else {hideChat(0)}
