@name 
@inputs Get Next Prev
@outputs Y Z:string
@persist X Pos:array Loc:vector Y
@trigger all
if (!clk("A")) {
if (Next & Y < Pos:count()) {Y += 1}
if (Prev & Y > 0) {Y -= 1}
}
timer("A",20)

if (Get & !clk("A")) {X += 1 Pos:setVector(X,owner():pos()+vec(0,0,50))}
Loc = Pos:vector(Y)
if (clk("A")) {
entity():applyForce(-entity():vel() + -entity():pos() + Loc)
}
Z = toString(Y)
