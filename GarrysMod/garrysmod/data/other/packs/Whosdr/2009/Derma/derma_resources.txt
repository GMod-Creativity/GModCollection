@name 
@inputs A  MA         #Air, Max Air
@inputs E  ME         #Energy, Max Energy
@inputs C  MC         #Coolant, Max Coolant
@inputs W  MW         #Water, Max Water
@inputs HW MHW        #Heavy Water, Max Heavy Water
@outputs AG EG CG WG HWG 
@persist AP EP CP WP HWP
@trigger 
if (first()) {
dPanel("Holder","Resources",vec2(0,0),vec2(300,125))

dLabel("Air", "Holder", "Air", vec2(10,25))
dLabel("Energy", "Holder", "Energy", vec2(10,45))
dLabel("Coolant", "Holder", "Coolant", vec2(10,65))
dLabel("Water", "Holder", "Water", vec2(10,85))
dLabel("Heavy_Water", "Holder", "Heavy_Water", vec2(10,105))

dButton("AirB","Holder","On",vec2(250,25),vec2(50,20))
dButton("EnergyB","Holder","On",vec2(250,45),vec2(50,20))
dButton("CoolantB","Holder","On",vec2(250,65),vec2(50,20))
dButton("WaterB","Holder","On",vec2(250,85),vec2(50,20))
dButton("HeavyB","Holder","On",vec2(250,105),vec2(50,20))
}


interval(200)

enableMouse(owner():keyAttack2())

AP = (floor(A/MA*100) != 0 ? floor(A/MA*100) : 0)
EP = (floor(E/ME*100) != 0 ? floor(E/ME*100) : 0)
CP = (floor(C/MC*100) != 0 ? floor(C/MC*100) : 0)
WP = (floor(W/MW*100) != 0 ? floor(W/MW*100) : 0)
HWP = (floor(HW/MHW*100) != 0 ? floor(HW/MHW*100) : 0)




dLabel("AirA", "Holder",         A+"/"+MA + " : "+AP+"%", vec2(100,25))
dLabel("EnergyA", "Holder",      E+"/"+ME + " : "+EP+"%", vec2(100,45))
dLabel("CoolantA", "Holder",     C+"/"+MC + " : "+CP+"%", vec2(100,65))
dLabel("WaterA", "Holder",       W+"/"+MW + " : "+WP+"%", vec2(100,85))
dLabel("Heavy_WaterA", "Holder", HW+"/"+MHW + " : "+HWP+"%", vec2(100,105))







dButton("AirB","Holder",(!AG ? "On" : "Off"),vec2(250,25),vec2(50,20))
dButton("EnergyB","Holder",(!CG ? "On" : "Off"),vec2(250,45),vec2(50,20))
dButton("CoolantB","Holder",(!EG ? "On" : "Off"),vec2(250,65),vec2(50,20))
dButton("WaterB","Holder",(!WG ? "On" : "Off"),vec2(250,85),vec2(50,20))
dButton("HeavyB","Holder",(!HWG ? "On" : "Off"),vec2(250,105),vec2(50,20))
