@name 
@inputs A  MA         #Air, Max Air
@inputs E  ME         #Energy, Max Energy
@inputs C  MC         #Coolant, Max Coolant
@inputs W  MW         #Water, Max Water
@inputs HW MHW        #Heavy Water, Max Heavy Water
@outputs AG EG CG WG HWG 
@persist AP EP CP WP HWP
@trigger 
if (first()) {
dPanel("Holder","Resources",vec2(0,0),vec2(300,125))
dPanelSetColor("Holder",100,100,100,200)

dLabel("Air", "Holder", "Air", vec2(10,25))
dLabel("Energy", "Holder", "Energy", vec2(10,45))
dLabel("Coolant", "Holder", "Coolant", vec2(10,65))
dLabel("Water", "Holder", "Water", vec2(10,85))
dLabel("Heavy_Water", "Holder", "Heavy_Water", vec2(10,105))

dButton("AirB","Holder","Off",vec2(250,25),vec2(50,20))
dButton("EnergyB","Holder","Off",vec2(250,45),vec2(50,20))
dButton("CoolantB","Holder","Off",vec2(250,65),vec2(50,20))
dButton("WaterB","Holder","Off",vec2(250,85),vec2(50,20))
dButton("HeavyB","Holder","Off",vec2(250,105),vec2(50,20))

dLabel("AirA", "Holder",         "0/0 : 0%                 ", vec2(100,25))
dLabel("EnergyA", "Holder",      "0/0 : 0%                 ", vec2(100,45))
dLabel("CoolantA", "Holder",     "0/0 : 0%                 ", vec2(100,65))
dLabel("WaterA", "Holder",       "0/0 : 0%                 ", vec2(100,85))
dLabel("Heavy_WaterA", "Holder", "0/0 : 0%                 ", vec2(100,105))


AG = EG = CG = WG = HWG = 0
}


interval(200)

enableMouse(owner():keyAttack2())

AP = (floor(A/MA*100) != 0 ? floor(A/MA*100) : 0)
EP = (floor(E/ME*100) != 0 ? floor(E/ME*100) : 0)
CP = (floor(C/MC*100) != 0 ? floor(C/MC*100) : 0)
WP = (floor(W/MW*100) != 0 ? floor(W/MW*100) : 0)
HWP = (floor(HW/MHW*100) != 0 ? floor(HW/MHW*100) : 0)




dLabelSetText("AirA",         A+"/"+MA + " : "+AP+"%")
dLabelSetText("EnergyA",      E+"/"+ME + " : "+EP+"%")
dLabelSetText("CoolantA",     C+"/"+MC + " : "+CP+"%")
dLabelSetText("WaterA",       W+"/"+MW + " : "+WP+"%")
dLabelSetText("Heavy_WaterA", HW+"/"+MHW + " : "+HWP+"%")







dButtonSetText("AirB",(!AG ? " Off" : " On"))
dButtonSetColor("AirB",!AG*200,AG*200,0,200)

dButtonSetText("EnergyB",(!EG ? " Off" : " On"))
dButtonSetColor("EnergyB",!EG*200,EG*200,0,200)

dButtonSetText("CoolantB",(!CG ? " Off" : " On"))
dButtonSetColor("CoolantB",!CG*200,CG*200,0,200)

dButtonSetText("WaterB",(!WG ? " Off" : " On"))
dButtonSetColor("WaterB",!WG*200,WG*200,0,200)

dButtonSetText("HeavyB",(!HWG ? " Off" : " On"))
dButtonSetColor("HeavyB",!HWG*200,HWG*200,0,200)


if (buttonClk("AirB")) {AG = !AG}
if (buttonClk("EnergyB")) {EG = !EG}
if (buttonClk("CoolantB")) {CG = !CG}
if (buttonClk("WaterB")) {WG = !WG}
if (buttonClk("HeavyB")) {HWG = !HWG}
