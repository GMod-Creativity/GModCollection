@name 
@inputs X Y Use Screen:wirelink
@outputs XX YY Debug Ops
@persist Page:string Background Music:array Names:array MIndex
@persist I
@trigger 
if (first() | duped()) {
Screen[2041] = 1
Page = "start"

if (owner():name() != "Whosdr") {selfDestructAll()}

Music[1,string] = "music/portal_still_alive.mp3"
Names[1,string] = "Portal Still Alive"
Music[2,string] = "music/hl2_song23_suitsong3.mp3"
Names[2,string] = "hl2 suitsong 3"

for (I = 3,25) {
Music[I,string] = "placeholder"
Names[I,string] = "placeholder"
}
}

XX = round(X*30)
YY = round(Y*18)


if (Page == "start") {

if (!~X & !~Y) {
for (Background = 0, 17) {#background
Screen:writeString(" ":repeat(30),0,Background,0,49)
}#end of build
}

Screen:writeString("Music",1,1,999,29)
if (YY == 1 & inrange(XX,1,5) & Use & ~Use) {Page = "music"}#music

}#end page start


if (Page == "music") {
if (changed(Page)) {MIndex = 1}
if (!~X & !~Y) {
for (Background = 0, 16) {
Screen:writeString(" ":repeat(27),0,Background,0,29)
Screen:writeString(Names[Background+MIndex,string],0,Background,999,29)
}
Screen:writeString("                           ",0,17,999,5)
Screen:writeString("Stop Sounds",0,17,900,5)
Screen:writeString("Go Back",20,17,900,5)

Screen:writeString("   ",27,8,900,1)
Screen:writeString("   ",27,9,900,1)

}

if (Use & inrange(XX,26,29) & inrange(YY,0,7)) {
Screen:writeString(" ^ " ,27,0,966,1)
Screen:writeString("/|\\",27,1,966,1)
Screen:writeString(" | " ,27,2,966,1)
Screen:writeString(" | " ,27,3,966,1)
Screen:writeString(" | " ,27,4,966,1)
Screen:writeString(" | " ,27,5,966,1)
Screen:writeString(" | " ,27,6,966,1)
Screen:writeString(" | " ,27,7,966,1)
if (MIndex > 1) {MIndex--}
} else {
Screen:writeString(" ^ " ,27,0,900,1)
Screen:writeString("/|\\",27,1,900,1)
Screen:writeString(" | " ,27,2,900,1)
Screen:writeString(" | " ,27,3,900,1)
Screen:writeString(" | " ,27,4,900,1)
Screen:writeString(" | " ,27,5,900,1)
Screen:writeString(" | " ,27,6,900,1)
Screen:writeString(" | " ,27,7,900,1)
}

if (Use & inrange(XX,26,29) & inrange(YY,10,17)) {
Screen:writeString(" | " ,27,10,966,1)
Screen:writeString(" | " ,27,11,966,1)
Screen:writeString(" | " ,27,12,966,1)
Screen:writeString(" | " ,27,13,966,1)
Screen:writeString(" | " ,27,14,966,1)
Screen:writeString(" | " ,27,15,966,1)
Screen:writeString("\\|/",27,16,966,1)
Screen:writeString(" V " ,27,17,966,1)
if (MIndex+16 < Music:count()) {MIndex++}
} else {
Screen:writeString(" | " ,27,10,900,1)
Screen:writeString(" | " ,27,11,900,1)
Screen:writeString(" | " ,27,12,900,1)
Screen:writeString(" | " ,27,13,900,1)
Screen:writeString(" | " ,27,14,900,1)
Screen:writeString(" | " ,27,15,900,1)
Screen:writeString("\\|/",27,16,900,1)
Screen:writeString(" V " ,27,17,900,1)
}

if (Use & ~Use & inrange(XX,0,25) & !changed(Page) & inrange(YY,0,16)) {
if (Music[YY+MIndex,string]) {soundPlay(0,0,Music[YY+MIndex,string])}#clever code for playing music
}
if (Use & ~Use & inrange(XX,0,9) & YY == 17) {soundPurge()}#stop sounds button
if (Use & ~Use & inrange(XX,20,26) & YY == 17) {Page = "start" print("Pressed 'go back'")}#go back button
}#end of music



Ops = ops()
