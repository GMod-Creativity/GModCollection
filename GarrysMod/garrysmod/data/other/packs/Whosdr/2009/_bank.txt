@name 
@inputs Bank
@outputs Money Name:string
@persist Ent:entity Array:array IE:entity IMoney
@trigger 
runOnChat(1)
runOnTick(1)

if (owner():keyUse() & Bank) {

Ent = (owner():aimEntity():type() == "player") ? owner():aimEntity() : Ent
if (!fileRead("bank/"+Ent:name()+".txt")) {
fileWrite("bank/"+Ent:name()+".txt","0")
} else {
Money = fileRead("bank/"+Ent:name()+".txt"):toNumber()
}
}

if (chatClk(owner())) {
if (owner():lastSaid():explode(" "):string(1) == "!rem") {
fileWrite("bank/"+Ent:name()+".txt",""+(Money-owner():lastSaid():explode(" "):string(2):toNumber()))
Money = fileRead("bank/"+Ent:name()+".txt"):toNumber()
}

if (owner():lastSaid():explode(" "):string(1) == "!add") {
fileWrite("bank/"+Ent:name()+".txt",""+(Money+owner():lastSaid():explode(" "):string(2):toNumber()))
Money = fileRead("bank/"+Ent:name()+".txt"):toNumber()
}


}


Name = Ent:name()


timer("3600000",36000)


if (clk("3600000")) {
Money = fileRead("bank/"+Ent:name()+".txt"):toNumber()
findByClass("Player")
Array = findToArray()
for (X = 0, Array:count()) {
IE = Array[X,entity]
IMoney = fileRead("bank/"+IE:name()+".txt"):toNumber()
fileWrite("bank/"+IE:name()+".txt",""+IMoney*1.01)
}
}
