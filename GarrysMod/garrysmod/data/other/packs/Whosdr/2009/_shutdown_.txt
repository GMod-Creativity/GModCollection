@name 
@inputs 
@outputs 
@persist E:entity Props:array Prop:entity Angles:angle Count Index Aim:vector
@persist Oangles:angle Speed
@trigger 
if (first()) {
E = owner()
}


timer("update",1000)
if (clk("update")) {
findIncludePlayerProps(owner())
findByClass("prop_physics")
Props = findToArray()
Count = Props:count()

for (X = 1, Count) {
Props[X,entity]:setMass(50000)
}


}



runOnTick(1)
Oangles = E:angles()
Speed = curtime()*100

if (owner():isAlive()) {
if (owner():keyAttack2()) {
for (Index = 1,Count) {
Prop = Props[Index,entity]
Vector = owner():aimPos()
Prop:applyForce( ((Vector - Prop:pos() + vec(0,0,0))*20 - Prop:vel())*Prop:mass())
Prop:applyAngForce(((-Prop:angles())*20 - Prop:angVel())*Prop:mass())

}
} elseif (owner():keyAttack1()) {
for (Index = 1,Count) {
Angles = ang(0,Speed + 360/Count*Index,0) -Oangles
Vector = E:toWorld(vec(50 + 20*Count,0,0):rotate(Angles))
Prop = Props[Index,entity]
Prop:applyForce( ((Vector - Prop:pos() + vec(0,0,20))*20 - Prop:vel())*Prop:mass())
Prop:applyAngForce(((-Prop:angles())*20 - Prop:angVel())*Prop:mass())
}
} elseif (owner():isCrouch()) {


for (Index = 1,Count) {
Angles = ang(0,Speed*3,0) - Oangles
Vector = E:toWorld(vec(50 + 40*Index,0,0):rotate(Angles))
Prop = Props[Index,entity]
Prop:applyForce( ((Vector - Prop:pos() + vec(0,0,40))*20 - Prop:vel())*Prop:mass())
Prop:applyAngForce(((-Prop:angles())*20 - Prop:angVel())*Prop:mass())
}


} else {
for (Index = 1,Count) {
Angles = ang(0,Speed + 360/Count*Index,0) -Oangles
Vector = E:toWorld(vec(50 + 20*Count,0,0):rotate(Angles))
Prop = Props[Index,entity]
Prop:applyForce( ((Vector - Prop:pos() + vec(0,0,70))*20 - Prop:vel())*Prop:mass())
Prop:applyAngForce(((-Prop:angles())*20 - Prop:angVel())*Prop:mass())
}


}
}
