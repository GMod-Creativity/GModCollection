@name 
@inputs Alarm
@outputs 
@persist Toggle Array:array Ent:entity X
@trigger 
interval(1000)

findIncludePlayerProps(owner())
findByClass("prop_physics")
Array = findToArray()

while (X < Array:count()) {
X++

Ent = Array[X,entity]
if (Alarm) {Toggle = !Toggle} else {Toggle = 0}

if (Alarm) {
if (Toggle) {Ent:setColor(255,0,0)} else {
    Ent:setColor(255,255,255)
    Ent:soundPlay(0,0,"SGA_midway_selfdestruct.wav")
    soundPitch(0,150)
    }
} else {
soundPurge()
Ent:setColor(255,255,255)
}
}

if (X == Array:count()) {X = 0}
