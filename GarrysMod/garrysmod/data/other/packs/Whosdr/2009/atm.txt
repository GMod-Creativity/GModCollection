@name 
@inputs XX YY Use Player:entity Screen:wirelink
@outputs 
@persist X Y Name:string Cash Page:string
@trigger 
if (first()) {
Page = "login"
}

X = round(XX*30)
Y = round(YY*18)
Name = Player:name()

if (Page == "login" & Player) {
Screen[1048574] = 1
if (fileRead("ATM/"+Name+".txt")) {
Cash = fileRead("ATM/"+Name+".txt"):toNumber()
} else {
fileWrite("ATM/"+Name+".txt","100")
}
Screen:writeString("Your account ballance is $"+Cash,0,0)
}
