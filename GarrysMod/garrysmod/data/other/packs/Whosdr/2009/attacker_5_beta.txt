@name 
@inputs 
@outputs 
@persist Prop:entity Props:array Player:entity Mode On Pos:vector Cloak Vec:vector Ang:angle X
@trigger all
runOnChat(1)
interval(10)
if (first() | duped()) {
findIncludePlayerProps(owner())
findByClass("prop_physics")
findClipToModel("models/props_junk/sawblade001a.mdl")
Props = findToArray()
On = 1
Cloak = 1
}

if (owner():lastSaid():lower() == "*" & !clk()) {
Cloak = !Cloak * 255
Prop:setAlpha(Cloak)
}

if (On) {

if (owner():lastSaid():lower() == "!!" & Mode != 0 & !clk()) {
Target = owner()
Mode = 0
}
if (owner():lastSaid():explode(" "):string(1) == "!" & !clk()) {
Player = findPlayerByName(owner():lastSaid():explode(" "):string(2))
Mode = 1
if (Player:pos() == vec(0,0,0)) {print("Error: No target.") Mode = 0}
}

if (owner():lastSaid():explode(" "):string(1) == "#" & !clk()) {
Player = findPlayerByName(owner():lastSaid():explode(" "):string(2))
Mode = 2
if (Player:pos() == vec(0,0,0)) {print("Error: No target.") Mode = 0}
}


if (owner():lastSaid():explode(" "):string(1) == "$" & !clk()) {
Player = findPlayerByName("googlebum")
Mode = 3
}

if (owner():lastSaid():explode(" "):string(1) == "^" & !clk()) {
Mode = 4
}

if (X < Props:count()) {
X++
if (Mode == 4) {
findInSphere(owner():pos(),1000)
findClipToClass("player")
findExcludePlayer("Whosdr")
if (find()) {Player = find()} else {Player = owner()}

Vec = (-Prop:pos() + -Prop:vel() + Player:pos() + vec(0,0,60*(1+(Player == owner())))+Player:vel())
Ang = (-Prop:angles() + -Prop:angVel() + ang(0,curtime()*(Player != owner()),0))
}

if (Mode == 3) {
if (owner():keyAttack2()) {
Vec = (-Prop:pos() + -Prop:vel() + owner():aimPos() + vec(0,0,50))
Ang = (-Prop:angles() + -Prop:angVel() + ang(0,curtime(),0))
} else {
}
}

if (Mode == 2) {

Vec = (-Prop:pos() + -Prop:vel() + Player:pos() + vec(0,0,120-(owner():keyAttack2())*70)+Player:vel())
Ang = (-Prop:angles() + -Prop:angVel() + ang(0,curtime(),0))
}
if (Mode == 1) {
Vec = (-Prop:pos() + -Prop:vel() + Player:pos() + vec(0,0,50)+Player:vel())
Ang = (-Prop:angles() + -Prop:angVel() + ang(0,curtime(),0))
}
if (Mode == 0) {
Vec = (-Prop:pos()) + -Prop:vel() + owner():pos() + vec(0,0,150)+owner():vel()
Ang = (-Prop:angles() + -Prop:angVel() + ang(0,0,0))
}


Prop = Props[X,entity]
Prop:applyForce(Vec*Prop:mass())
Prop:applyAngForce(Ang*Prop:mass())


}
} else {X = 0}


if (owner():lastSaid() == "!!!") {On = 0}
if (owner():lastSaid() == "!!!!") {On = 1}



if (owner():lastSaid():index(1) == "!") {hideChat(1)}
elseif (owner():lastSaid():index(1) == "$") {hideChat(1)}
elseif (owner():lastSaid():index(1) == "^") {hideChat(1)}
