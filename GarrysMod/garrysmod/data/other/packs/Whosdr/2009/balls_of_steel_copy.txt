@name 
@inputs 
@outputs 
@persist E:entity Props:array Prop:entity Angles:angle Count Index
@persist Oangles:angle Speed
@trigger 
if (first()) {
E = owner()
}


timer("update",1000)
if (clk("update")) {
findIncludePlayerProps(owner())
findByClass("prop_physics")
Props = findToArray()
Count = Props:count()

}



runOnTick(1)
Oangles = E:angles()
Speed = curtime()*100

if (owner():keyAttack2()) {
for (Index = 1,Count) {
Prop = Props[Index,entity]
Vector = owner():aimPos()
Prop:applyForce( ((Vector - Prop:pos() + vec(0,0,0))*20 - Prop:vel())*Prop:mass())
Prop:applyAngForce(((-Prop:angles())*20 - Prop:angVel())*Prop:mass())

}
} else {


for (Index = 1,Count) {
Angles = ang(0,Speed + 360/Count*Index,0) -Oangles
Vector = E:toWorld(vec(50 + 20*Count,0,0):rotate(Angles))
Prop = Props[Index,entity]
Prop:setMass(50000)
Prop:applyForce( ((Vector - Prop:pos() + vec(0,0,70))*20 - Prop:vel())*Prop:mass())
Prop:applyAngForce(((-Prop:angles())*20 - Prop:angVel())*Prop:mass())
}




}
