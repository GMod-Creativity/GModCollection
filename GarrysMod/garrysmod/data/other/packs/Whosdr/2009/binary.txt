@name 
@inputs String:string
@outputs Add:string Array:array Bin:string
@persist Numb X 
@persist L1 L2 L3 L4 L5 L6 L7 L8
@trigger 
#if (first()) {String = "AlmarM"}

Bin = ""
for (X = 1,String:length()) {
Numb = toByte(String:index(X))

if (Numb >= 128) {L1 = 1, Numb -= 128} else {L1 = 0}
if (Numb >= 64 ) {L2 = 1, Numb -= 64}  else {L2 = 0}
if (Numb >= 32 ) {L3 = 1, Numb -= 32}  else {L3 = 0}
if (Numb >= 16 ) {L4 = 1, Numb -= 16}  else {L4 = 0}
if (Numb >= 8  ) {L5 = 1, Numb -= 8}   else {L5 = 0}
if (Numb >= 4  ) {L6 = 1, Numb -= 4}   else {L6 = 0}
if (Numb >= 2  ) {L7 = 1, Numb -= 2}   else {L7 = 0}
if (Numb >= 1  ) {L8 = 1, Numb -= 1}   else {L8 = 0}

Add = L1 + "" + L2 + "" + L3 + "" + L4 + "" + L5 + "" + L6 + "" + L7 + "" + L8 + " "
Array[X,string] = Add
}

for (X = 1,Array:count()) {
#print(Array[X,string])
Bin += Array[X,string]
}
