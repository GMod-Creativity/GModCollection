@name 
@inputs Screen:wirelink
@outputs 
@persist PosX PosY SizeX SizeY SpeedX SpeedY DispX DispY X Y
@trigger 
if (first()) {
    PosX = 256
    PosY = 256
    SizeX = randint(20)
    SizeY = randint(20)
    SpeedX = randint(-10,10)
    SpeedY = randint(-20,20)
if (first()) {Screen[1048574] = 1}
}


interval(10)

if (DispX >= 512 | DispX <= 0) {SpeedX = -SpeedX * 1}
if (DispY >= 512 | DispY <= 0) {SpeedY = -SpeedY * 1}


#Screen[1048574] = 1

PosX += SpeedX
PosY += SpeedY

DispX = PosX + SizeX
DispY = PosY + SizeY

for (X = 1, SizeX) {
for (Y = 1, SizeY) {
Screen[(PosX + X) + ((PosY + Y)*512)] = randint(999999)
}
}
