@name 
@inputs Screen:wirelink
@outputs X Y
@persist RX RY SX SY DX DY X Y I J
@trigger 
if (first() | changed(Screen)) {
RX = 32 #resolution X
RY = 32 #resolution Y

SX = 5 #size X
SY = 5 #size Y

DX = randint(1,5) #direction X
DY = randint(1,5) #direction Y

Screen[1048573] = RX #set res X
Screen[1048572] = RY #set res Y

X = round(RX/2)
Y = round(RY/2)
}
interval(200)

Screen[1048574] = 1  #reset screen

X += DX
Y += DY

for (I = 0, SY-1) {
for (J = 0, SX-1) {
Screen[(X+I)+((Y+J)*RX)] = 99999
}
}


if (X <= 1 | X >= (RX-SX) -1) {DX = -DX}
if (Y <= 1 | Y >= (RY-SY) -1) {DY = -DY}
