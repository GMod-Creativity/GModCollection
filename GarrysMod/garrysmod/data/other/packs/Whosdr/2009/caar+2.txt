@name 
@inputs W S Player:entity
@outputs 
@persist Pos:vector Ent:entity Angle:angle
@trigger all
if (first() | duped()) {
Ent = entity():isWeldedTo()
Pos = Ent:pos()
Angle = Ent:angles()
holoCreate(1)
}

runOnTick(1)

if (W) {Pos += Player:driver():eye()*20}
if (S) {Pos -= Player:driver():eye()*20}

#Angle = ang(0,Bearing,0)

Angle = Player:driver():eyeAngles() + ang(0,-90,0)

Ent:applyForce(( -Ent:vel() + (Pos -Ent:pos())*20 )*Ent:mass())

Ent:applyAngForce(angnorm( (Angle -Ent:angVel())*10  -Ent:angles())*Ent:mass())

if (Ent:isPlayerHolding()) {Pos = Ent:pos() Angle = Ent:angles()}
holoPos(1,Pos)
