@name
@inputs 
@outputs 
@persist Calc:string Split:array Current Op Char:string
@trigger all
runOnChat(1)
if (owner():lastSaid():sub(1,5) == "-calc" & chatClk(owner())) {
Calc = owner():lastSaid():sub(6,1000)
Split = Calc:explode(" ")
Op = 1
hideChat(1)
} else {hideChat(0)}

#op list

#+ 1
#- 2
#X 3
#/ 4
#^ 5
#


for (X = 1,Split:count()) {
Char = Split[X,string]
if (X == 1) {Current = Char:toNumber()} else {

    if (Char == "+") {Op = 1}
elseif (Char == "-") {Op = 2}
elseif (Char == "X") {Op = 3}
elseif (Char == "*") {Op = 3}
elseif (Char == "/") {Op = 4}
elseif (Char == "^") {Op = 5}
elseif (Char:toNumber()) {
        if (Op == 1) {Current = Current + Char:toNumber()}
    elseif (Op == 2) {Current = Current - Char:toNumber()}
    elseif (Op == 3) {Current = Current * Char:toNumber()}
    elseif (Op == 4) {Current = Current / Char:toNumber()}
    elseif (Op == 5) {Current = Current ^ Char:toNumber()}
    } else {print("invalid oporation")}


}
}

concmd("say " + Calc + " = " + Current)
