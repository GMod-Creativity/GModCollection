@name 
@inputs Screen:wirelink On Reset
@outputs 
@persist Form1 Form2 X Y RD:ranger RX RY
@trigger Reset On
if (first()) {
print("Res: "+RX+"X"+RY)
concmd("ulx rcon wire_expression2_quotasoft 1000000000")
concmd("ulx rcon wire_expression2_quotahard 1000000000")
concmd("ulx rcon wire_expression2_quotatick 1000000000")
}

RX = Screen[1048572]
RY = Screen[1048573]

if (On & ~On) {
timer("A",10)
}
 
if (clk("A")) {
    for (Y = 0, 85) {
        for (X = 0, RX) {
            Form1 = X/(RX/2)-1
            Form2 = Y/(RY/2)-1
            RD = ranger(10000,Form1,Form2)
            Screen[X+(Y*RX)] = (1000-(RD:distance()*1.14))
        }
stoptimer("A")
timer("B",10000)
    }
}




if (clk("B")) {
    for (Y = 86, 170) {
        for (X = 0, RX) {
            Form1 = X/(RX/2)-1
            Form2 = Y/(RY/2)-1
            RD = ranger(10000,Form1,Form2)
            Screen[X+(Y*RX)] = (1000-(RD:distance()*1.14))
        }
stoptimer("B")
timer("C",1000)
    }
}


if (clk("C")) {
    for (Y = 171, 256) {
        for (X = 0, RX) {
            Form1 = X/(RX/2)-1
            Form2 = Y/(RY/2)-1
            RD = ranger(10000,Form1,Form2)
            Screen[X+(Y*RX)] = (1000-(RD:distance()*1.14))
        }
stoptimer("C")
    }
}



if (Reset & ~Reset) {
#    Screen[1048574] = 1

for (Y = 0,RY) {
for (X = 0,RX) {    
Screen[X+(Y*RX)] = 0
}
}
}
