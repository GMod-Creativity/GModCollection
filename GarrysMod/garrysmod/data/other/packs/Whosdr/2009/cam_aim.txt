@name 
@inputs 
@outputs Cam:vector Eye:vector On
@persist Zoom:array Ind X
@trigger 
if (first()) {

Zoom[1,number] = 50
Zoom[2,number] = 100
Zoom[3,number] = 200
Zoom[4,number] = 500
Zoom[5,number] = 1000

runOnTick(1)
}


if (owner():weapon():type() == "weapon_crossbow") {
    if (changed(owner():weapon())) {Ind = 0}
    if (owner():keyUse() & changed(owner():keyUse())) {Ind++}
    if (Ind >= 6) {Ind = 0, On = 0}
    if (changed(owner():keyAttack1()) & !owner():keyAttack1()) {Ind = 0,On = 0}
    if (Ind) {On = 1, Cam = owner():pos() + owner():eye()*Zoom[Ind,number], Eye = owner():eye()/Zoom[Ind,number]}
} else {
    if (changed(owner():weapon())) {Ind = 0,On = 0}
}
