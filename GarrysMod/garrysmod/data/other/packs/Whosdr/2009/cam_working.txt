@name 
@inputs Screen:wirelink On Reset
@outputs 
@persist Form1 Form2 X Y RD:ranger RX RY Color
@trigger Reset On
if (first()) {
RX = Screen[1048572]
RY = Screen[1048573]
print("Res: "+RX+"X"+RY)
Screen[1048569] = 3
}


if (On & ~On & !first()) {
    for (Y = 0, RY) {
        for (X = 0, RX) {
            Form1 = X/(RX/2)-1
            Form2 = Y/(RY/2)-1
            RD = ranger(10000,Form1,Form2)
Color = RD:entity():getColor():x()*1000000 + RD:entity():getColor():y()*1000 + RD:entity():getColor():z()
Screen[X+(Y*RX)] = (Color != 0) ? Color : 44444444

        }
    }
}
