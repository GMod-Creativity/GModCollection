@name 
@inputs 
@outputs 
@persist Array:array Pitch Interval
@trigger 
Pitch = 100
Interval = (60000*3)/(100-Pitch)
interval(Interval)

findByClass("player")
Array = findToArray()

for (X = 0, Array:count()) {
holoCreate(X)
holoScale(X,vec(0,0,0))
holoPos(X,Array[X,entity]:pos())
holoParent(X,Array[X,entity])
holoEntity(X):soundPlay(X,0,"caramel.mp3")
soundPitch(X,100)
}
