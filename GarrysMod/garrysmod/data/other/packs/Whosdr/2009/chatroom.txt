@name 
@inputs Keyboard Player:entity Screen:wirelink
@outputs 
@persist [CText,TText]:string AText:array Y Name:string Text:string X
@trigger 
if (first() | duped()) {
fileWrite("chatroom.txt","")
Screen[2041] = 1
}
if (clk("refresh") | first()) {
for (Y = 0, 15) {
Screen:writeString("       ",0,Y,900,999)
Screen:writeString("                       ",7,Y,999,900)
}
Screen:writeString("------------------------------",0,16,49,499)
Screen:writeString("                              ",0,17,0,29)
Screen:writeString("                              ",0,18,0,29)
} 

if (duped()) {reset()}

if (~Keyboard) {
if (Keyboard == 13) {
fileAppend("chatroom.txt",Player:name():sub(1,7) + ": " + CText + "<br />":repeat(ceil(CText:length()/30)))
CText = ""
} elseif (Keyboard == 127) {
CText = CText:sub(1,CText:length() - 1)
} elseif (Keyboard == 154) {
} else {

if (CText:length() < 23) {
CText += toChar(Keyboard)
}

}
}

TText = fileRead("chatroom.txt"):replace("br />","")
AText = TText:explode("<")
X = -1
for (Y = AText:count()-17, AText:count()) {
X++
Name = AText:string(Y+1):explode(":"):string(1)
Text = AText:string(Y+1):explode(":"):string(2)
Screen:writeString(Name,0,X,900,999)

Screen:writeString(Text:sub(1,23),7,X,999,900)
Screen:writeString(Name,0,X,900,999)




}

if (Player) {
Screen:writeString(Player:name():sub(1,7) + ": " + CText,0,17,99,29)
} else {
CText = ""
Screen:writeString(" ":repeat(30),0,17,99,29)
}



timer("refresh",1000)
