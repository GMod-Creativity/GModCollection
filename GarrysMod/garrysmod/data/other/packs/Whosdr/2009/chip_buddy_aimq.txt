@name Chip Buddy Aimq
@inputs NumInPut I 
@persist Weld:entity EE:entity

runOnTick(1)

if (first()) {
EE=entity()
EE:setMass(0.01)
Weld = EE:isWeldedTo()
Weld:setMass(50000)
}

Weld:applyForce((-Weld:pos() - Weld:vel() + owner():aimPos() + vec(0,0,50))*Weld:mass())
#EE:applyForce((vec(0,0,9.015)))
Weld:applyAngForce((-Weld:angles() - Weld:angVel() + ang(0,curtime()*100,0))*Weld:mass())

if(chatClk(owner())){
    if(owner():lastSaid() == "+trail"){
    Weld:setTrails(25,25,25,"trails/tube",vec(255,255,255),100)
    }

    if(owner():lastSaid() == "-trail") {
    Weld:setTrails(0,0,0,"trails/tube",vec(0,0,0),0)
    }

    if(owner():lastSaid() == "-green"){
    Weld:setColor(0,255,0,255)
    }
    
    if(owner():lastSaid() == "-unknown"){
    Weld:setColor(10,15,10,255)
    }
       
    if(owner():lastSaid() == "-red"){
    Weld:setColor(255,0,0,255)
    }
            
    if(owner():lastSaid() == "-blue"){
    Weld:setColor(0,0,255,255)
    }
    
    if(owner():lastSaid() == "-invis"){
    Weld:setColor(0,0,0,0)
    EE:setColor(0,0,0,0)
    }
    
    if(owner():lastSaid() == "-norm"){
    Weld:setColor(255,255,255,255)
    EE:setColor(255,255,255,255)
    }
    
    }
