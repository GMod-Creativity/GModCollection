#######

if (Arg1 == "number") {Number = Arg2:toNumber()}
if (Arg1 == "entity") {Entity = findPlayerByName(Arg2)}
if (Arg1 == "vector") {Vector = vec(Arg2:explode(","):string(1):toNumber(),Arg2:explode(","):string(2):toNumber(),Arg2:explode(","):string(3):toNumber())}
if (Arg1 == "angle") {Angle = ang(Arg2:explode(","):string(1):toNumber(),Arg2:explode(","):string(2):toNumber(),Arg2:explode(","):string(3):toNumber())}

if (Arg1 == "iris") {IrisOpen = !IrisOpen}
if (Arg1 == "close") {Close = 1 timer("close",100)}
if (clk("close")) {Close = 0, stoptimer("close")}

timer("dial",100)
I++
if (I < 7) {Address = toByte(Dialing:index(I))} else {Address = 13, stoptimer("dial"),Dialing = "", I = 0,Arg1 = "" Arg2 = ""}
print(Address+"")
}


}

for (X = 0, 17) {

}

if (K == 127) {Page = "DOS" Screen[2041] = 1}

}


if (Page == "status") {


timer("ping",500)

Screen:writeString("name: "+hostname():sub(1,54),0,0,CC)

Screen:writeString("Date: "+ time("day") + "/" + time("month") + "/" + time("year") + " ":repeat(24 - time("day"):toString():length() - time("month"):toString():length() - time("year"):toString():length()),0,6,CC)
Screen:writeString("Players: "+numPlayers() + "/" + maxPlayers(),0,4,CC)


if (K == 127) {Page = "DOS" Screen[2041] = 1 stoptimer("ping")}

}




if (Page == "music") {
if (K & K != 127) {soundPlay(1,0,"synth/sine.wav") soundPitch(1,K)} else {soundStop(1)}
if (K == 127) {Page = "DOS"}
}



if (Page == "stargate") {

}

Screen:writeString("         *          *         ",0,14,999)
Screen:writeString("          *##****##*          ",0,15,999)
Screen:writeString("                              ",0,16,999)
}

Screen:writeString("##",14,0,ChevColor[7,number]) #chevron 7
Screen:writeString("#",8,2,ChevColor[6,number])#chevron 6
Screen:writeString("#",7,3,ChevColor[6,number])#chevron 6
Screen:writeString("#",21,2,ChevColor[1,number])#chevron 1
Screen:writeString("#",22,3,ChevColor[1,number])#chevron 1

Screen:writeString("#",23,7,ChevColor[2,number])#chevron 2

Screen:writeString("#",22,12,ChevColor[3,number])#chevron 3

Screen:writeString("#",7,12,ChevColor[4,number])#chevron 4

if (K == 127) {Page = "DOS"}

if (!~Chevron & !~K & !~Text) {

Screen:writeString(((Iris) ? "=" : " "):repeat(12),9 ,2,555,29*Open)
