@name 
@inputs 
@outputs 
@persist Array:array Ent:entity X
@trigger 
interval(1000)
findExcludeClass("prop_physics")
findIncludePlayerProps(owner())
findInSphere(entity():pos(),1000000)
Array = findToArray()

for (X = 0, Array:count()) {
Ent = Array[X,entity]
Ent:setMaterial("")

if (Ent:type() == "air_tank") {Ent:setColor(255,255,255)}
elseif (Ent:type() == "air_compressor") {Ent:setColor(255,255,255)}
elseif (Ent:type() == "air_exchanger") {Ent:setColor(255,255,255)}

elseif (Ent:type() == "energy_cell") {Ent:setColor(255,255,0)}
elseif (Ent:type() == "energy_solar") {Ent:setColor(255,255,0)}
elseif (Ent:type() == "energy_solar_sail") {Ent:setColor(255,255,0)}
elseif (Ent:type() == "warp_core") {Ent:setColor(255,255,0)}
elseif (Ent:type() == "energy_wind") {Ent:setColor(255,255,0)}
elseif (Ent:type() == "heat_exchanger") {Ent:setColor(255,255,0)}

elseif (Ent:type() == "coolant_tank") {Ent:setColor(0,150,255)}
elseif (Ent:type() == "coolant_compressor") {Ent:setColor(0,150,255)}


elseif (Ent:type() == "water_tank") {Ent:setColor(0,0,255)}
elseif (Ent:type() == "water_pump") {Ent:setColor(0,0,255)}

elseif (Ent:type() == "hvywater_tank") {Ent:setColor(150,150,150)}
elseif (Ent:type() == "hvywater_electrolyzer") {Ent:setColor(150,150,150)}




elseif (Ent:type() == "Stargate") {Ent:setColor(255,0,0)}
elseif (Ent:type() == "climate_control") {Ent:setColor(200,200,200)}
#elseif (Ent:type() == "Prop_physics") {Ent:setColor(255,255,255)}
elseif (Ent:type() == "res_cache") {Ent:setColor(100,100,100)}

else {Ent:setColor(255,255,255)}

}
