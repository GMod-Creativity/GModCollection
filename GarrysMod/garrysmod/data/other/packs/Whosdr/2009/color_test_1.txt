@name 
@inputs Screen:wirelink
@outputs 
@persist Color:table Colors:string I RX RY X Y
@trigger 
if (first()) {
Color["R",number] = 900 #red
Color["Y",number] = 990 #yellow
Color["B",number] =   9 #blue
Color["C",number] =  99 #cyan
Color["G",number] =  90 #green
Color["P",number] = 909 #purple
Color["O",number] =  94 #orange
Color["D",number] =   0 #dark(black)
Color["W",number] = 999 #white
Screen[1048569] = 1     #RGB mode
Screen[1048574] = 1     #Reset
RX = 10
RY = 10
Screen[1048573] = RX    #Res X
Screen[1048572] = RY    #Res Y

Colors = "WDOPGCBYRW":repeat(10)
}
if (Screen) {

for (Y = 0,RX-1) {
for (X = 0,RY-1) {
I++
Screen[X+(Y*RX)] = Color[Colors:index(I),number]
}
}


}
