@name 
@inputs Screen:wirelink Resource:wirelink
@outputs 
@persist A E C
@persist MA ME MC
@persist PA PE PC
@trigger 
timer("update",1000)
A = Resource["Air",number]
E = Resource["Energy",number]
C = Resource["coolant",number]

MA = Resource["Max Air",number]
ME = Resource["Max Energy",number]
MC = Resource["Max Coolant",number]

PA = floor(A/MA*100)
PE = floor(E/ME*100)
PC = floor(C/MC*100)
