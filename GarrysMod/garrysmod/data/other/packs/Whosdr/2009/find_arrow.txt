@name 
@inputs 
@outputs 
@persist Player:entity LS:array Pos:vector Angle:angle Speed
@trigger 
if (first()) {


holoCreate(1)
holoModel(1,"cube")
holoColor(1,vec(255,0,0),255)
holoPos(1,entity():pos() + vec(0,0,150))
holoScale(1,vec(1,2,8))
holoAng(1,ang())

holoCreate(2)
holoModel(2,"pyramid")
holoColor(2,vec(255,0,0),255)
holoPos(2,entity():pos() + vec(0,0,90))
holoScale(2,vec(1,4,2))
holoAng(2,ang(0,0,180))


holoParent(2,1)
Player = owner()

Speed = 1000
}

runOnTick(1)
runOnChat(1)

if (chatClk(owner())) { 
LS = owner():lastSaid():explode(" ")
if (LS[1,string] == "!find") {
Player = findPlayerByName(LS[2,string])
hideChat(1)
} else {
hideChat(0)
}
}

if (Player == noentity()) {Player = owner()}
Pos = Player:pos() + vec(0,0,200+(sin(curtime()*Speed)*30))
Angle = ang(0,curtime()*Speed,0)
holoPos(1,Pos)
holoAng(1,Angle)
