@name Follower
@persist Tar:entity P:entity E:entity
runOnTick(1)
runOnChat(1)

if (first()) {
E = entity()
if (entity():isWeldedTo()) {P = entity():isWeldedTo()} else {P = entity()}
E:setMass(1)
Tar = owner()
}

if (chatClk(owner()) & owner():lastSaid():explode(" "):string(1):lower() == "!") {
Tar = findPlayerByName(owner():lastSaid():explode(" "):string(2)) 
}

if (owner():lastSaid() == "!!") {Tar = owner()}

P:applyForce((-P:pos() - P:vel() + Tar:pos() + vec(0,0,50+ (Tar == owner())*50))*P:mass()) 
P:applyAngForce((-P:angles() - P:angVel() + ang(0,curtime()/10*(Tar != owner()),0))*P:mass())

if (owner():lastSaid():explode(" "):string(1) == "!") {
hideChat(1)
} else {
hideChat(0)
}
