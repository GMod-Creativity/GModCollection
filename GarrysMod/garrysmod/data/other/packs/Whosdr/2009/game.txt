@name 
@inputs Screen:wirelink S D Fire
@outputs Points
@persist X Y Bullet:vector2 B Blocks:array
@persist I   #loops
@trigger 
if (first()) {
X = 15
Y = 30
Bullet = vec2(-1,-1)

for (I = 1, 12) {
Blocks[I,vector2] = vec2(randint(30)+1,randint(20) - 1)
}

}

interval(500)

if (S & X > 0) {X--}
if (D & (X+2) < 31) {X++}

Screen[1048574] = 1 #reset

##############Player####################
Screen[X+(Y*32)] = 999999 #left        #
Screen[X+1+(Y*32)] = 999999 #middle    #
Screen[X+2+(Y*32)] = 999999 #right     #
Screen[X+1+((Y-1)*32)] = 999999 #top   #
########################################


for (I = 1, Blocks:count()) {
Screen[Blocks[I,vector2]:x()+Blocks[I,vector2]:y()*32] = 999999 

if (Bullet == vec2(Blocks[I,vector2]:x(),Blocks[I,vector2]:y()*32)   ) {
Points += 10
Bullet = vec2(-1,-1)
Blocks[I,vector2] = vec2(-1,-1)
}

}

if (Screen[Bullet:x()+(Bullet:y()-1)] != 999999) {
Screen[Bullet:x() + Bullet:y()] = 999999
}

if (Fire & Bullet:y() < 0) {Bullet = vec2(X+1,32*(Y-3))}
Bullet -= vec2(0,32*(Bullet:y() > -1))

if (Screen[Bullet:x()+(Bullet:y()-1)] == 999999) {
print(Bullet:x()+(Bullet:y()*32-1)+"")
Points += 10
Bullet = vec2(-1,-1)
} else {
Screen[Bullet:x() + Bullet:y()] = 999999
}
