@name 
@inputs W S A D Shift Ent:entity Space Active Attack2
@outputs Pos:vector Eye:vector
@persist Player:entity X Array:array On
@trigger all
if (first()) {
Pos = entity():pos()
holoCreate(1)
holoScale(1,vec(1,1,1))
holoMaterial(1,"models/alyx/emptool_glow")
holoModel(1,"icosphere3")
}

Player = Ent:driver()
interval(10)

if (W) {Pos += Player:eye()*(20+(Shift*80))}
if (S) {Pos -= Player:eye()*(20+(Shift*80))}

Eye = Player:eye()

if (A) {Pos += holoEntity(1):right()*(20+(Shift*80))}
if (D) {Pos -= holoEntity(1):right()*(20+(Shift*80))}

findByClass("player")
findExcludePlayer(owner())
Array = findToArray()
if (Space & ~Space) {
if (X < Array:count()) {X += 1} else {X = 1}
Pos = Array:entity(X):pos() + vec(0,0,80)
}

holoPos(1,Pos)


if (!Active) {
Pos = Ent:pos() + vec(0,0,75) + vec(sin(curtime()*100)*10,cos(curtime()*100)*10,0)
}

holoColor(1,vec(255,0,0),50)


if (Attack2 & ~Attack2) {On = !On}

if (On) {
holoScale(1,vec(1,1,1))
} else {
holoScale(1,vec(0,0,0))
}
