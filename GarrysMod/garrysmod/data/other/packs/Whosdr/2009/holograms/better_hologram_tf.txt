@name 
@inputs Next Prev Ent:entity
@outputs String:string
@persist Vec1:vector Vec2:vector Vec3:vector Vec4:vector X Finds:array Find:entity
@trigger all
if (first()) {
holoCreate(1)
holoCreate(2)
holoCreate(3)
holoCreate(4)


holoCreate(5)
holoCreate(6)
holoCreate(7)
holoCreate(8)

holoColor(1,vec(0,255,0))
holoColor(2,vec(255,0,0))
holoColor(3,vec(0,255,0))
holoColor(4,vec(255,0,0))

holoColor(5,vec(0,255,0))
holoColor(6,vec(255,0,0))
holoColor(7,vec(0,255,0))
holoColor(8,vec(255,0,0))


holoScale(1,vec(1,1,10))
holoScale(2,vec(1000,0.5,0.5))
holoScale(3,vec(1,1,10))
holoScale(4,vec(1000,0.5,0.5))

holoScale(5,vec(1,1,10))
holoScale(6,vec(0.5,1000,0.5))
holoScale(7,vec(1,1,10))
holoScale(8,vec(0.5,1000,0.5))

}

runOnTick(1)

Vec1 = Ent:pos() + vec(cos(curtime()*100)*300,sin(curtime()*100)*300,100)
Vec2 = Ent:pos() + vec(-cos(curtime()*100)*300,-sin(curtime()*100)*300,100)

Vec3 = Ent:pos() + vec(sin(curtime()*100)*300,cos(curtime()*100)*300,100)
Vec4 = Ent:pos() + vec(-sin(curtime()*100)*300,-cos(curtime()*100)*300,100)


holoPos(1,Vec1)
holoPos(3,Vec2)

holoPos(5,Vec3)
holoPos(7,Vec4)

holoPos(2,Ent:pos() + vec(0,0,140) )
holoPos(4,Ent:pos() + vec(0,0,50) )

holoPos(6,Ent:pos() + vec(0,0,140) )
holoPos(8,Ent:pos() + vec(0,0,50) )



holoAng(2,ang(0,curtime()*100,0))
holoAng(4,ang(0,curtime()*100,0))

holoAng(6,ang(0,-(curtime())*100,0))
holoAng(8,ang(0,-(curtime())*100,0))


findInSphere(Ent:pos(),300)
findClipToClass("player")
Finds = findToArray()
Find = Finds:entity(X)
if (Next & ~Next & X < Finds:count()) {X += 1} else {X = 1}
if (Prev & Prev & X > 1) {X -= 1} else {X = Finds:count()}
String = Find:name()
