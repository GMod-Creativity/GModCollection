@name 
@inputs Ent:entity
@outputs 
@persist Vec1:vector Vec2:vector Find:array Pos:vector
@trigger all
if (first()) {
holoCreate(1)
holoCreate(2)
holoCreate(3)
holoCreate(4)

Pos = Ent:pos()

holoColor(1,vec(0,255,0))
holoColor(2,vec(255,0,0))
holoColor(3,vec(0,255,0))
holoColor(4,vec(255,0,0))

holoScale(1,vec(1,1,30))
holoScale(2,vec(1000,0.5,0.5))
holoScale(3,vec(1,1,30))
holoScale(4,vec(1000,0.5,0.5))

}
#interval(300)
runOnTick(1)

Vec1 = Pos + vec(cos(curtime()*100)*1000,sin(curtime()*100)*1000,150)
Vec2 = Pos + vec(-cos(curtime()*100)*1000,-sin(curtime()*100)*1000,150)

holoPos(1,Vec1)
holoPos(3,Vec2)

holoPos(2,Pos + vec(0,0,140) )
holoPos(4,Pos + vec(0,0,50) )


holoAng(2,ang(0,curtime()*100,0))
holoAng(4,ang(0,curtime()*100,0))

findInSphere(Pos,2000)
findByClass("player")
Find = findToArray()
