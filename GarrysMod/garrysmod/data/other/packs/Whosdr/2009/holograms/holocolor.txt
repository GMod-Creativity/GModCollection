@name 
@inputs 
@outputs Red Green
@persist Color:vector
@trigger all
if (first()) {
holoCreate(1)
}
interval(20)
holoScale(1,vec(0.2,owner():health() / 25,0.2))
holoPos(1,owner():pos() + vec(0,0,50) + (owner():eye()*50) + (owner():vel()/3))
holoAng(1,owner():angles())
Red = (100-owner():health())*2.5
Green = owner():health()*2.5
Color = vec(Red,Green,0)
holoColor(1,Color)
