@name 
@inputs E ME A MA C MC
@outputs 
@persist 
@trigger all

holoCreate(1)
holoCreate(2)
holoCreate(3)

holoColor(1,vec(255,255,0))
holoColor(2,vec(255,255,255))
holoColor(3,vec(0,100,255))
}

holoScale(1,vec(1,1,(E/ME)*5))
holoPos(1,entity():pos() + vec(0,0,(E/ME*40)+0) + entity():right()*50)
holoParent(1,entity())

holoScale(2,vec(1,1,(A/MA)*5))
holoPos(2,entity():pos() + vec(0,0,(A/MA*40)+0))
holoParent(2,entity())

holoScale(3,vec(1,1,(C/MC)*5))
holoPos(3,entity():pos() + vec(0,0,(C/MC*40)+0) + entity():right()*-50)
