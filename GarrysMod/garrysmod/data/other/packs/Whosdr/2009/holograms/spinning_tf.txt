@name 
@inputs Next Prev
@outputs String:string
@persist Vec1:vector Vec2:vector X Finds:array Find:entity
@trigger all
if (first()) {
holoCreate(1)
holoCreate(2)
holoCreate(3)
holoCreate(4)



holoColor(1,vec(0,255,0))
holoColor(2,vec(255,0,0))
holoColor(3,vec(0,255,0))
holoColor(4,vec(255,0,0))

holoScale(1,vec(1,1,30))
holoScale(2,vec(1000,0.5,0.5))
holoScale(3,vec(1,1,30))
holoScale(4,vec(1000,0.5,0.5))

}

runOnTick(1)

Vec1 = owner():pos() + vec(cos(curtime()*100)*1000,sin(curtime()*100)*1000,150)
Vec2 = owner():pos() + vec(-cos(curtime()*100)*1000,-sin(curtime()*100)*1000,150)

holoPos(1,Vec1)
holoPos(3,Vec2)

holoPos(2,owner():pos() + vec(0,0,140) )
holoPos(4,owner():pos() + vec(0,0,50) )


holoAng(2,ang(0,curtime()*100,0))
holoAng(4,ang(0,curtime()*100,0))

findInSphere(owner():pos(),2000)
findClipToClass("player")
Finds = findToArray()
Find = Finds:entity(X)
if (Next & ~Next & X < Finds:count()) {X += 1} else {X = 1}
if (Prev & Prev & X > 1) {X -= 1} else {X = Finds:count()}
String = Find:name()
