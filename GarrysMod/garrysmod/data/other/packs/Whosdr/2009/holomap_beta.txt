@name 
@inputs 
@outputs 
@persist Array:array
@trigger 
if (first()) {
holoCreate(1000)
holoPos(1000,entity():pos() + vec(0,0,50))    
holoScale(1000,vec(-100,-100,-100))    
holoParent(1000,entity())    
holoModel(1000,"")

findInSphere(entity():pos(),100000)
Array = findToArray()
}
interval(100)

for (X = 0,Array:count()) {
holoCreate(X)
holoScaleUnits(X,Array[X,entity]:boxSize()/5)
holoPos(X,Array[X,entity]:pos()/5 + entity():pos() + vec(0,0,0))
holoColor(X, Array[X,entity]:getColor()     )
}
