@name 
@inputs Set Go Reset
@outputs 
@persist X Pos:array Color:vector
@trigger 
if (Set & ~Set) {
X += 1
holoCreate(X)
holoPos(X,owner():aimPos())
Pos[X,vector] = owner():aimPos()
}

interval(100)

if (Go) {
if (~Go) {X = 0}
X++
holoColor(X,Color)
}

if (X == Pos:count() & Go) {Color = vec(random(0,255),random(0,255),random(0,255)) X = 0}


if (Reset & ~Reset) {
for (X = 1,Pos:count()) {
holoColor(X,vec(255,255,255))
}
X = 0
}
