@name 
@inputs W A S D Ent:entity
@outputs Cam:vector Eye:vector
@persist Pos:vector Speed1 Speed2 E:entity
@trigger 
@model models/hunter/plates/plate075x075.mdl

if(first()){
E = entity()
    holoCreate(1)# 
    holoCreate(2)#
    holoCreate(3)# 
    holoCreate(4)# 
    holoCreate(5)
    holoCreate(6)
    ######## Body ########
    holoPos(1, E:pos() + E:right()*6 + E:forward()*-8)
    holoScale(1, vec(3,1,0.2))
    ######## wings ########
    holoPos(2, E:pos() + E:right()*-15 + E:forward()*10 + E:right()*6 + E:forward()*-8)
    holoScale(2, vec(1,1,0.4))
    holoAng(2, E:angles() + ang(0,0,0))
    ######## wings ########
    holoPos(3, E:pos() + E:right()*15 + E:forward()*10 + E:right()*6 + E:forward()*-8)
    holoScale(3, vec(1,1,0.4))
    holoAng(3, E:angles())
    ######## Ring ########
    holoModel(4, "torus3")
    holoPos(4, E:pos() + E:right()*6 + E:forward()*-8)
    holoScale(4,vec(1.3,1.3,1))
    ######## side1 ########
    holoPos(5, E:pos() - E:right()*5 + E:forward()*10 + E:right()*6 + E:forward()*-8)
    holoScale(5, vec(0.2,2,0.18))
    holoAng(5, E:angles())

    ######## side2 ########
    holoPos(6, E:pos() + E:right()*5 + E:forward()*10 + E:right()*6 + E:forward()*-8)
    holoScale(6, vec(0.2,2,0.18))
    holoAng(6, E:angles())

    holoMaterial(1,"spacebuild/bodyskin")
    holoMaterial(2,"spacebuild/bodyskin")
    holoMaterial(3,"spacebuild/bodyskin")
    holoMaterial(4,"phoenix_storms/futuristictrackramp_1-2")
    holoMaterial(5,"spacebuild/sblight_d") #crossbar1
    holoMaterial(6,"spacebuild/sblight_d") #crossbar2

for (X = 1,6) {
holoParent(X,1)
#holoMaterial(X,"cmats/station_metal_b")
Speed1 = 5
Speed2 = 5
}

holoAng(1,E:angles() + ang(0,180,0))
holoParent(1,E)  
Pos = E:pos()

E:setAlpha(0)
}    

runOnTick(1)

if (W) {
Speed1 += (Speed1 < 80)/5
} else {
Speed1 -= (Speed1>5)/2
}

if (S) {
Speed2 += (Speed2 < 40)/5
} else {
Speed2 -= (Speed2>5)/2
}


Eye = Ent:driver():eye()
if (W) {Pos += Eye*Speed1}
if (S) {Pos -= Eye*Speed2}
if (A) {Pos += -E:right()*(Speed1+Speed2/2)}
if (D) {Pos += E:right()*(Speed1+Speed2)/2}

E:applyForce(( (-E:pos() + Pos)*(Speed1+Speed2)/2 - E:vel())*E:mass())
#E:applyAngForce(angnorm( -E:angles() - E:angVel() + Ent:driver():eyeAngles())*E:mass())

Torque = E:toLocal(rotationVector(quat(Ent:driver():eyeAngles())/quat(E))+E:pos())
E:applyTorque((Torque*100-E:angVelVector()*10)*E:inertia())



Cam = Pos + vec(0,0,20) + E:right()*5 + E:forward()*-30

if (E:isPlayerHolding()) {Pos = E:pos()}
