@persist Main:entity Outer:array Mode:string Prop:entity Angles:angle Vector:vector
@trigger 
if (first()) {

findIncludePlayerProps(owner())
findByClass("gmod_laser")
Outer = findToArray()
Main = Outer[1,entity]
Main:setMass(5000)
Mode = "idle"
#holoCreate(1)
#holoPos(1,Main:pos())
}
runOnTick(1)


if (Mode == "idle") {

Main:applyForce(( (vec(0,0,1000)-Main:pos())*20 -Main:vel())*Main:mass())
Main:applyAngForce(( (ang(0,0,180)-Main:angles())*20 -Main:angVel())*Main:mass())

for (X = 1,8) {
Angles = ang(0,1 + 360/Outer:count()*X,0) -Main:angles()
Vector = Main:toWorld(vec(500 + 20*Outer:count(),0,0):rotate(Angles))
Prop = Outer[X,entity]
Prop:applyForce(( (vec(0,0,1000)-Prop:pos())*20 -Prop:vel())*Prop:mass())

}


}
