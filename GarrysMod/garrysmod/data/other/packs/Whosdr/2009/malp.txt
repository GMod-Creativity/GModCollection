@name 
@inputs Malp:entity Driver:entity W A S D Up Down Shift
@outputs Cam:vector Angle:vector Drag:vector
@persist Pos:vector Posit:vector Vel:vector Ang:angle Speed
@trigger 
if (first() | duped() | ~Malp) {
holoCreate(1)
#holoPos(1,Malp:pos() + vec(0,0,Malp:boxSize():z()/2))
holoPos(1,Malp:pos())
holoAng(1,Malp:angles())
holoScaleUnits(1,Malp:boxSize()*-1.3)
holoParent(1,Malp)
holoColor(1,vec(50,50,255),150)
holoMaterial(1,"phoenix_storms/chrome")
holoModel(1,"hqsphere2")


Pos = Malp:pos()
}
runOnTick(1)

Malp:applyForce(( (Pos -Malp:pos())*20 - Malp:vel())*Malp:mass())
#Malp:applyAngForce(angnorm( -Malp:angles() -Malp:angVel() + Driver:driver():eyeAngles() )*Malp:mass())

Torque = Malp:toLocal(rotationVector(quat(Driver:eyeAngles())/quat(Malp))+Malp:pos())
Malp:applyTorque((Torque*100-Malp:angVelVector()*10)*Malp:inertia())



Posit = Malp:pos()
Vel = $Posit
Speed = (Shift) ? 200 : 10
if (Vel:x() > 20 | Vel:y() > 20 | Vel:z() > 20 | Malp:isPlayerHolding()) {Pos = Malp:pos()}
Pos += (Malp:forward()*(W-S)*Speed) + (Malp:right()*(D-A)*Speed) + vec(0,0,1*(Up-Down)*Speed)

Angle = Malp:forward()
Cam = Malp:pos()
Drag = owner():eye()*1000 + Malp:pos()
