@name 
@inputs 
@outputs 
@persist Array:array Pitch Interval
@trigger 
Pitch = 100
Interval = (60000*3)/(100-Pitch)
interval(Interval)

findIncludePlayer("teabager")
findByClass("player")
Array = findToArray()

for (X = 0, Array:count()) {
holoCreate(X)
holoScale(X,vec(3,3,7))
holoPos(X,Array[X,entity]:pos() + vec(0,0,30))
holoParent(X,Array[X,entity])
holoEntity(X):soundPlay(X,0,"synth/square.wav")
soundPitch(X,200)
holoEntity(X):soundPlay(X*2,0,"synth/saw.wav")
soundPitch(X*2,200)
}
