@name Adv rt
@inputs Up Down One1 Two2 Three3 Four4
@outputs One Two Three Four X
@persist

if (Down == 1 & X > 1) {X -= 1}

if (X == 1) {One = 1 Two = 0 Three = 0 Four = 0}
if (X == 2) {One = 0 Two = 1 Three = 0 Four = 0}
if (X == 3) {One = 0 Two = 0 Three = 1 Four = 0}
if (X == 4) {One = 0 Two = 0 Three = 0 Four = 1}

if (One1 == 1) {X = 1}
if (Two2 == 1) {X = 2}
if (Three3 == 1) {X = 3}
if (Four4 == 1) {X = 4}
