@name 
@inputs 
@outputs Pos:vector Fire Found
@persist Player:entity X Players:array
@trigger 
interval(10)
findByClass("player")
Players = findToArray()

if (X < Players:count() & Found == 0) {X += 1}
if (X >= Players:count() & Found == 0) {X = 1}

Player = Players[X,entity]

if (Player:weapon():model() == "models/weapons/v_crowbar.mdl") {
Found = 1
Fire = Player:keyAttack1()
} else {Found = 0}

if (Found) {
Pos = Player:pos() + vec(0,0,70)
}
