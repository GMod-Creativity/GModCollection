@name 
@inputs 
@outputs 
@persist Ent:entity OldMass Attack2
@trigger all
interval(100)
Attack2 = owner():keyAttack2()
if (owner():aimEntity():pos() != vec(0,0,0) & !owner():keyAttack2() & !$Attack2) {
OldMass = owner():aimEntity():mass()
}

if (owner():keyAttack2()) {
owner():aimEntity():setMass(1)
} else {
owner():aimEntity():setMass(OldMass)
}
