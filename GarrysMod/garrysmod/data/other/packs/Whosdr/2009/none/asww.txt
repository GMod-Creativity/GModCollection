@name 
@inputs 
@outputs A B C D E
@persist S:string String:string X
@trigger 
S = fileRead("lights.txt")
interval(1000)
X += 1
String = S:explode("<br />"):string(X)
A = String:explode(","):string(1):toNumber()
B = String:explode(","):string(2):toNumber()
C = String:explode(","):string(3):toNumber()
D = String:explode(","):string(4):toNumber()
E = String:explode(","):string(5):toNumber()
