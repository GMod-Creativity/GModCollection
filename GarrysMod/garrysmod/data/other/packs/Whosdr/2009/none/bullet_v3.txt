@name 
@inputs On
@outputs Bullet2 Reload2 On2
@persist Bullet Y Z Reload
########################
interval(100)          #
Bullet2 = Bullet + 100 #
Reload2 = Reload + 3   #
########################
if (clk() & On == 1 & Bullet > -100) {Bullet -=1}
if (Bullet >= -99 & On == 1) {On2 = 1}
if (Bullet <= -100) {On2 == 0 Bullet = -100}
if (Bullet >= -99 & On == 0) {On2 = 0}
if (Bullet == -100 & Reload >= -2) {Y = 1}
if (Y == 1 & clk()) {Z +=1}
if (Z == 100) {Bullet = 0 Reload -= 1}
if (Z == 101) {Z = 0 Y = 0}
if (Bullet == -100 & Reload == -3) {Bullet = -100 On2 = 0}
if (Y == 1) {On2 = 0}
