@name 
@inputs 
@outputs R B
@persist On Speed
@trigger all
if (first()) {R = 255 B = 0 On = 0 Speed = 0}
runOnChat(1)
chatClk(owner())
interval(Speed)
if (!clk() & owner():lastSaid():lower() == "/on") {On = 1}
if (!clk() & owner():lastSaid():lower() == "/off") {On = 0}
if (owner():lastSaid():explode(" "):string(1):lower() == "/speed") {Speed = owner():lastSaid():explode(" "):string(2):toNumber()}
if (On & R == 255) {R = 0 B = 255}
elseif (On & R == 0) {R = 255 B = 0}
else {R = 0 B = 0}
