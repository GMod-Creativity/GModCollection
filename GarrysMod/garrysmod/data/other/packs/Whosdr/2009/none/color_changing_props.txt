@name 
@inputs On
@outputs 
@persist X Find:array Entity:entity Color:vector
@trigger all
interval(100)

findByClass("prop_physics")
findIncludePlayerProps(owner())
findSortByDistance(entity():pos()) 
Find = findToArray()
if (X < Find:count()) {X += 1} else {X = 1 Color = vec(random(0,255),random(0,255),random(0,255))}


if (On) {
Find:entity(X):setColor(Color)
Find:entity(X):setMaterial("models/debug/debugwhite")
} else {
Find:entity(X):setColor(255,255,255,255)
Find:entity(X):setMaterial("")
}
