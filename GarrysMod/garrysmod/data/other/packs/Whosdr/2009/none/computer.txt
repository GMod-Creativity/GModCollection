@name 
@inputs E:entity On F:entity
@outputs C
@persist R G B A
@trigger all
interval(100)

if (On) {
C = 50
E:setColor(R,G,B,A)
R = random(0,200)
G = random(0,200)
B = random(0,200)
A = 1
F:setColor(255,255,255,255)
} else {
C = 0
E:setColor(R,G,B,A)
R = 0
G = 0
B = 0
A = 0
F:setColor(0,0,0,255)
}
