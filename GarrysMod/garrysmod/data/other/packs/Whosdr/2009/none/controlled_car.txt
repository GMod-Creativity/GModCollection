@name 
@inputs 
@outputs Forward Left
@persist X
@trigger all
runOnChat(1)
if (owner():lastSaid() == "!on" & X == 0) {X = 1}
if (owner():lastSaid() == "!off" & X == 1) {X = 0}
if (X == 0) {Forward = 0 Left = 0}
if (owner():lastSaid() == "forward" & X == 1) {Forward = 1}
if (owner():lastSaid() == "backward" & X == 1) {Forward = -1}
if (owner():lastSaid() == "left" & X == 1) {Left = -1}
if (owner():lastSaid() == "right" & X == 1) {Left = 1}
if (owner():lastSaid() == "stop" & X == 1) {Left = 0 Forward = 0}
