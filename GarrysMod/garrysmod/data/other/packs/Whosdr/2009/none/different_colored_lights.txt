@name Different colored lights
@inputs On
@outputs Red Green Blue
@persist X
interval(1000)
if (clk() & On == 1) {X +=1}
if (On == 0) {Red = 0 Green = 0 Blue = 0}

if (X == 1) {Red = 255 Green = 0 Blue = 0}
if (X == 2) {Red = 0 Green = 255 Blue = 0}
if (X == 3) {Red = 0 Green = 0 Blue = 255}
if (X == 4) {Red = 255 Green = 255 Blue = 0}
if (X == 5) {Red = 255 Green = 0 Blue =2550}
if (X == 6) {Red = 0 Green = 255 Blue = 255}
if (X == 7) {Red = 255 Green = 255 Blue = 255}
if (X == 8) {X = 0}
