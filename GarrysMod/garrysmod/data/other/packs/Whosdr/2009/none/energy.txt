@name 
@inputs Eplug UP1 UP2 UP3 UP4 UP5
@outputs Energy O1 O2 O3 O4 O5
@persist Energy
@trigger all
interval(1000)
if (first()) {EP = 0}
if (Eplug & Energy < 100) {Energy += 1}
EP = Energy
EP -= (UP1+UP2+UP3+UP4+UP5)
if (EP > -1) {
Energy = EP
O1 = 1
O2 = 1
O3 = 1
O4 = 1
O5 = 1
} else {
O1 = 0
O2 = 0
O3 = 0
O4 = 0
O5 = 0
}
