@name 
@inputs 
@outputs Vec:vector Jump
@persist EVec:vector Dist
@trigger all
runOnChat(1)
chatClk()
interval(10)
EVec = (lastSpoke():pos() + vec(0,0,65)) + (lastSpoke():eye() * Dist)
Jump = 0
if (lastSaid():lower() == "!bring" & !clk()) {
Vec = EVec 
Jump = 1
}

if (lastSaid():explode(" "):string(1) == "!dist" & !clk()) {Dist = lastSaid():explode(" "):string(2):toNumber()}
