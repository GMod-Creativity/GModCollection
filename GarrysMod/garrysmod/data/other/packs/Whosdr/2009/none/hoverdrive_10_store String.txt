@name 
@inputs SX SY SZ Get Set Up Down String:string
@outputs X Y Z Q Coords:string Name:string
@persist X1 X2 X3 X4 X5 X6 X7 X8 X9 X10
@persist Y1 Y2 Y3 Y4 Y5 Y6 Y7 Y8 Y9 Y10
@persist Z1 Z2 Z3 Z4 Z5 Z6 Z7 Z8 Z9 Z10
@persist One:string Two:string Three:string Four:string Five:string Six:string Seven:string Eight:string Nine:string Ten:string
@persist 
@trigger Up Down Get Set
if (Up & Q <= 9) {Q += 1}
if (Down & Q >= 2) {Q -= 1}
if (Q == 1 & Get) {X = X1 Y = Y1 Z = Z1 Name = One}
if (Q == 1 & Set) {X1 = SX Y1 = SY Z1 = SZ One = String}

if (Q == 2 & Get) {X = X2 Y = Y2 Z = Z2 Name = Two}
if (Q == 2 & Set) {X2 = SX Y2 = SY Z2 = SZ Two = String}

if (Q == 3 & Get) {X = X3 Y = Y3 Z = Z3 Name = Three}
if (Q == 3 & Set) {X3 = SX Y3 = SY Z3 = SZ Three = String}

if (Q == 4 & Get) {X = X4 Y = Y4 Z = Z4 Name = Four}
if (Q == 4 & Set) {X4 = SX Y4 = SY Z4 = SZ Four = String}

if (Q == 5 & Get) {X = X5 Y = Y5 Z = Z5 Name = Five}
if (Q == 5 & Set) {X5 = SX Y5 = SY Z5 = SZ Five = String}

if (Q == 6 & Get) {X = X6 Y = Y6 Z = Z6 Name = Six}
if (Q == 6 & Set) {X6 = SX Y6 = SY Z6 = SZ Six = String}

if (Q == 7 & Get) {X = X1 Y = Y1 Z = Z1 Name = Seven}
if (Q == 7 & Set) {X7 = SX Y7 = SY Z7 = SZ Seven = String}

if (Q == 8 & Get) {X = X8 Y = Y8 Z = Z8 Name = Eight}
if (Q == 8 & Set) {X8 = SX Y8 = SY Z8 = SZ Eight = String}

if (Q == 9 & Get) {X = X9 Y = Y9 Z = Z9 Name = Nine}
if (Q == 9 & Set) {X9 = SX Y9 = SY Z9 = SZ Nine = String}

if (Q == 10 & Get) {X = X10 Y = Y10 Z = Z10 Name = Ten}
if (Q == 10 & Set) {X10 = SX Y10 = SY Z10 = SZ Ten = String}

Coords = round(X) + "," + round(Y) + "," + round(Z)
