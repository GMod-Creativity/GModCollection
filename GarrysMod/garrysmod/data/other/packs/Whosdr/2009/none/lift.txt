@name 
@inputs Up Down Set Lift:entity
@outputs Floor
@persist High:array Height Pos:vector Mass Props:array
@trigger all
if (first()) {
Floor = 1
High = array()
High:setNumber(1,0)
High:setNumber(2,50)
High:setNumber(3,100)
High:setNumber(4,150)
High:setNumber(5,200)
High:setNumber(6,250)
High:setNumber(7,300)
High:setNumber(8,350)
High:setNumber(9,400)
High:setNumber(10,450)
Pos = entity():pos()
}

if (Set & ~Set) {
Height = High:number(Floor)
}

if (Up & ~Up & Floor < High:count()) {Floor += 1}
if (Down & ~Down & Floor > 1) {Floor -= 1}

runOnTick(1)

findInSphere(Lift:pos() + vec(0,0,100),100)
Props = findToArray()
Mass = 0
for (X = 0, Props:count()) {
Mass += Props[X,entity]:mass()
}

Lift:applyForce((-Lift:pos() - Lift:vel() + Pos + vec(0,0,Height)):normalized()*50*Mass)
