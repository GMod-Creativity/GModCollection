@name 
@inputs Screen:wirelink
@outputs 
@persist X Y T:string P C K
@trigger all
interval(100)
if (first()) {Screen:setNumber("Reset",1) Screen:setNumber("Reset",0)}
Screen:writeString(T,X,Y,C)
X = random(0,29)
Y = random(0,17)
P = random(97,123)
C = random(111,999)
K = round(random(0,1))
if (K == 1) {T = toChar(P)}
else
{T = toChar(P)}
