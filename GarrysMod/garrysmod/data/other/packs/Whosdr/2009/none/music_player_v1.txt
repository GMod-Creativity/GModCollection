@name Music player v1
@inputs On Up Down
@outputs Name:string
@persist Song
if (On == 1 & Song == 1) {soundPlay(1,0,"music/hl2_song23_suitsong3.mp3") Name = ("Suitsong")}
if (On == 1 & Song == 2) {soundPlay(1,0,"music/portal_still_alive.mp3") Name = ("Portal still alive")}
if (On == 1 & Song == 3) {soundPlay(1,0,"music/hl2_song4.mp3") Name = ("Hl2_song4") }
if (On == 1 & Song == 4) {soundPlay(1,0,"ambient/music/looping_radio_mix.wav") Name = ("Portal radio")}
if (Up == 1 & Song <= 3) {Song +=1}
if (Down == 1 & Song >= 1) {Song -=1}
if (On == 1 & Song == 0) {soundPlay(1,1,"1")}
