@name 
@inputs E:entity Nex
@outputs Pos:vector Name:string Nextt
@persist F1:string F2:string F3:string F4:string F5:string T:string Next
@trigger all
interval(10)
if (first()) {
F1 = "whosdr"
F2 = "err9"
F3 = " "
F4 = " "
F5 = " "
}
if (!Next) {
Pos = E:pos()
Name = E:name()
} else {
Pos = vec(0,0,0)
Name = "No Target"
}
T = E:name():lower()
if (T == F1:lower() | T == F2:lower() | T == F3:lower() | T == F4:lower() | T == F5:lower() | Nex) {Next = 1} else {Next = 0}
if (Next) {Nextt = !Nextt}
