@name 
@inputs Pod:wirelink
@outputs Pos:vector Eye:vector
@persist Player:entity X Array:array W S A D Shift Ent:entity Space
@trigger all
if (first()) {
Pos = entity():pos()
}

W = Pod:number("W")
A = Pod:number("A")
S = Pod:number("S")
D = Pod:number("D")
Shift = Pod:number("Shift")
Space = Pod:number("Space")
Ent = Pod:entity("Entity")

Player = Ent:driver()
interval(10)

if (W) {Pos += Player:eye()*(20+(Shift*80))}
if (S) {Pos -= Player:eye()*(20+(Shift*80))}

Eye = Player:eye()

if (A) {Pos -= Player:forward()*(20+(Shift*80))}
if (D) {Pos += Player:forward()*(20+(Shift*80))}

findByClass("player")
findExcludePlayer(owner())
Array = findToArray()
if (Space & ~Space) {
if (X < Array:count()) {X += 1} else {X = 1}
Pos = Array:entity(X):pos() + vec(0,0,80)
}
