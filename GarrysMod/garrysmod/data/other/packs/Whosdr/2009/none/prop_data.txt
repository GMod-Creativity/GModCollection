@name 
@inputs 
@outputs 
@persist Entity:entity Model:string Size:string   Position:string   Mass 
@persist   Type:string Owner:string Angles:string Properties:string
@trigger 
interval(100)
Entity = owner():aimEntity()
Model = Entity:model():replace(".mdl",""):replace("/","-")
Size = round(Entity:boxSize()):toString()
Position = round(Entity:pos()):toString()
Mass = Entity:mass()
Type = Entity:type()
Owner = Entity:owner():name()
Angles = round(Entity:angles()):toString()
Properties = "Model:  "+Model+"<br />Size:  "+Size+"<br />Position:  "+Position+"<br />Mass:  "+Mass+"<br />Type:  "+Type+"<br />Owner:  "+Owner+"<br />Angles:  "+Angles


if (Entity:pos() != vec(0,0,0)) {
fileWrite("entitys/"+Model+".txt",Properties)
}
