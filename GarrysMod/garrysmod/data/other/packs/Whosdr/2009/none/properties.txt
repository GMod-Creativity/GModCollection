@name 
@inputs 
@outputs 
@outputs Entity:entity Model:string Size:string   Position:string   Mass 
@outputs Type:string Owner:string Angles:string Properties:string Double Attack2
@trigger 
interval(100)
Entity = owner():aimEntity()
Model = Entity:model()
Size = round(Entity:boxSize()):toString()
Position = round(Entity:pos()):toString()
Mass = ceil(Entity:mass())
Type = Entity:type()
Owner = Entity:owner():name()
Angles = round(Entity:angles()):toString()
Properties = "Model:  "+Model+"<br />Size:  "+Size+"<br />Position:  "+Position+"<br />Mass:  "+Mass+"<br />Type:  "+Type+"<br />Owner:  "+Owner+"<br />Angles:  "+Angles


Attack2 = (owner():keyAttack2() & owner():keyAttack1())

if (!Attack2 & $Attack2) {
Double = 1
timer("reset",1000)
}

if (clk("reset")) {Double = 0,stoptimer("reset")}

if (Double & Attack2 & $Attack2) {
hint(Properties:explode("<br />"):string(6),7)
hint(Properties:explode("<br />"):string(1),7)
hint(Properties:explode("<br />"):string(5),7)
hint(Properties:explode("<br />"):string(2),7)
hint(Properties:explode("<br />"):string(4),7)
hint(Properties:explode("<br />"):string(3),7)
hint(Properties:explode("<br />"):string(7),7)
}
