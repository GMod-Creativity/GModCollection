@name 
@inputs Screen:wirelink
@outputs 
@persist Y
@trigger all
runOnChat(1)
chatClk(owner())
if (first()) {Y = 0}
if (owner():lastSaid():explode("-"):string(1):lower() == "/rule") {
Screen:writeString(Y+1 +" : "+ owner():lastSaid():explode("-"):string(2),0,Y,911)
Y += 1
}
if (owner():lastSaid():lower() == "/reset") {
Y = -1
Screen:setNumber("Reset",1)
Screen:setNumber("Reset",0)
}
if (owner():lastSaid():explode(" "):string(1) == "/line") {Y = owner():lastSaid():explode(" "):string(2):toNumber()}
