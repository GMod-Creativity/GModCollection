@name 
@inputs On S Range
@outputs X Y Z Grab
@persist Q
@trigger all
interval(1000)
if (On & !Range) {
Q += 1
if (Q == 0) {X = 0 Y = 0 Z = 0}
if (Q == 1) {X = 250 Y = 0 Z = 0}
if (Q == 2) {X = 250 Y = 0 Z = 34}
if (Q == 3) {X = 250 Y = 50*S Z = 55}
if (Q == 4) {Grab = 1}
if (Q == 5) {X = 0 Y = 255 Z = 70}
if (Q == 6) {X = 0 Y = 225 Z = 70}
if (Q == 7) {Grab = 0}
} else {
X = 0
Y = 0
Z = 0
Q = 0
}
