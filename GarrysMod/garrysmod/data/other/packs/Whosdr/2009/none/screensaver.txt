@name 
@inputs Screen:wirelink
@outputs 
@persist X Color P
@trigger all
if (first()) {X = -1}
interval(10)
runOnChat(1)
chatClk(owner())

if (owner():lastSaid() == "Screensaver" & P == 0) {P = 1}
if (owner():lastSaid() == "Screensaver" & P == 1) {P = 0 
Screen:setNumber("Reset",1)
Screen:setNumber("Reset",0)
X = -1
}
if (P == 0) {
Screen:writeString("*",random(0,17),random(0,29),random(0,999))


}


if (P == 1) {
if (owner():lastSaid():explode(" "):string(1):lower() == "print") {
if (chatClk() & owner():lastSaid():lower() != "color") {X += 1}
if (X <= 23) {
Screen:writeString((X+1) + ": " + owner():lastSaid():sub(7,10000),0,X,Color)
}
else
{
Screen:writeString((X+1) + ":  " + owner():lastSaid():sub(7,10000),0,X,Color)
}
}




if (owner():lastSaid():length() >= 24) {X += 1}
#X 0-29
#Y 0-17
if (owner():lastSaid():lower() == "clear") {
Screen:setNumber("Reset",1)
Screen:setNumber("Reset",0)
X = -1
}



if (X >= 18) {
Screen:setNumber("Reset",1)
Screen:setNumber("Reset",0)
X = -1
}



if (owner():lastSaid():explode(" "):string(1):lower() == "color") {
Color = lastSaid():sub(7,1000):toNumber()
}
}
