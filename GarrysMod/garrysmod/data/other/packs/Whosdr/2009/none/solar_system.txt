@name Sol
@inputs On
@outputs 
@persist S:entity

interval(50)

if (first()) {
    S = entity()
    gSetGroup("solarsystem")
}

Ps = 50, Rs = 2000

gSetNum("Ps", Ps)
gSetNum("Rs", Rs)

P = S:pos(), V = S:vel()

gSetNum("Sx",P:x())
gSetNum("Sy",P:y())
gSetNum("Sz",P:z())
gSetNum("on",On)

S:applyForce(-V*50000)
S:setMass(50000)
