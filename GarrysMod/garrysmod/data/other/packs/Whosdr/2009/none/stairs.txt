@name 
@inputs On
@outputs H1 H2 H3 H4 H5 X
@persist N
@trigger all
interval(8000)
if (owner():lastSaid():toNumber() != 0) {
N = owner():lastSaid():toNumber()
} else {
N = 1}
if (On & X < 6) {X += 1}
if (!On & X > 0) {X -= 1}
if (X == 0) {H1 = 0 H2 = 0 H3 = 0 H4 = 0 H5 = 0}
if (X == 1) {H1 = N*10 H2 = N*20 H3 = N*30 H4 = N*40 H5 = N*50}
if (X == 2) {H1 = N*90 H2 = N*80 H3 = N*70 H4 = N*60 H5 = N*50}
if (X == 3) {H1 = N*90 H2 = N*100 H3 = N*110 H4 = N*120 H5 = N*130}
if (X == 4) {H1 = N*170 H2 = N*160 H3 = N*150 H4 = N*140 H5 = N*130}
if (X == 5) {H1 = N*170 H2 = N*180 H3 = N*190 H4 = N*200 H5 = N*210}
