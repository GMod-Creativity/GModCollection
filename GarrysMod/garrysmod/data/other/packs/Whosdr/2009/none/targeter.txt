@name 
@inputs 
@outputs Pos:vector Name:string
@persist Ent:entity Entity:entity
@trigger 
runOnChat(1)


if (chatClk(owner())) {
if (owner():lastSaid():explode(" "):string(1) == "!target") {    
Ent = findPlayerByName(owner():lastSaid():explode(" "):string(2))    
if (Ent) {Entity = Ent}
}
}

timer("update",50)

if (clk("update")) {
Pos = Entity:pos()
Name = Entity:name()
}
