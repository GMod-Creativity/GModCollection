@name Trafic light
@inputs On
@outputs Red Yellow Green
@persist X
interval(2000)
if (clk() & On == 1) {X += 1}
if (X == 0) {Red = 0 Yellow = 0 Green = 0}
if (X >= 1) {Red = 100 Yellow = 0 Green = 0}
if (X >= 11) {Red = 100 Yellow = 100 Green = 0}
if (X >= 13) {Red = 0 Yellow = 0 Green = 100}
if (X >= 23) {Red = 0 Yellow = 100 Green = 0}
if (X >= 25) {X = 1}
if (On == 0) {X = 0}
