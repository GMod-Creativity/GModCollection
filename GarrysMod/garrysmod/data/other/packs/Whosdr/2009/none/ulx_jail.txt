@name 
@inputs 
@outputs 
@persist Player:string
@trigger 
runOnChat(1)

if (owner():lastSaid():explode(" "):string(1) == "/jail" & chatClk(owner())) {
Player = owner():lastSaid():explode(" "):string(2)
concmd("ulx jail "+Player)
timer("unjail",owner():lastSaid():explode(" "):string(3):toNumber())
}


if (clk("unjail")) {
concmd("ulx unjail "+Player)
stoptimer("ulx jail")
}
