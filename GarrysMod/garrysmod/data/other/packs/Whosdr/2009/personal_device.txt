@name Personal Device
@inputs B1 TD TaO TaD Open Inbound
@outputs OX OY OZ DX DY DZ TE Send Retrieve 
@persist D I1 ErrorLable:string Array:array Mode:string Mode1:string Coord:vector Origin:vector Destin:vector HaveOne SaveSpot:vector Spots:array PlayersE:array PlayersN:array SpotsN:array SpotsP:array PrepareSend Sended Pos:vector Code:string CodeL:string
@trigger 
#Maded by AnKis
if( first() ){
    ErrorLable = "Error.........................."
    Bw = 52
    Bh = 42
    Bs = 10
    By = 52
    Bx = 100
    Lx = 350
    Ly = 85
    Ls = 30
    dPanel( "Personal Device - Control Panel", vec2( 400, 200 ), vec2( 600, 300 ) )
    dShow( "Personal Device - Control Panel", 0 )
    dTabHolder( "PDTH", "Personal Device - Control Panel", vec2( 10, 30 ), vec2( 580, 260 ) )
    dTab( "GDO", "PDTH" )
    dTab( "ATS", "PDTH" )
    #ATS
    dLabel( "Advanced Teleporting System", "ATS", vec2( 5, 5 ) )
    dLabel( ErrorLable, "ATS", vec2( 405, 5 ) )
    dLabel( "to", "ATS", vec2( 493, 60 ) )
    dListBox( "Players", "ATS", vec2( 5, 30 ), vec2( 100, 195 ) )
    dListBox( "Spots", "ATS", vec2( 110, 30 ), vec2( 100, 195 ) )
    dListBox( "Players1", "ATS", vec2( 215, 30 ), vec2( 100, 195 ) )
    dListBox( "Spots1", "ATS", vec2( 320, 30 ), vec2( 100, 195 ) ) 
    dTextBox( "Coordinates", "ATS", vec2( 435, 30 ), 125 )
    dDropBox( "Mode", "ATS", vec2( 435, 55 ), 50 )
    dDropBox( "Mode1", "ATS", vec2( 510, 55 ), 50 )
    dCheckBox( "Teleport Everything", "ATS", vec2( 435, 80 ) )
    dButton( "Send", "ATS", vec2( 435, 100 ), vec2( 125, 20 ) )
    dButton( "Retrieve", "ATS", vec2( 435, 125 ), vec2( 125, 20 ) )
    dTextBox( "Name", "ATS", vec2( 435, 150 ), 125 )
    dDropBox( "SaveMode", "ATS", vec2( 435, 175 ), 125 )
    dButton( "Save", "ATS", vec2( 435, 200 ), vec2( 125, 20 ) )
    dArray( "Mode", array( "Player", "Spot", "Coord." ) )
    dArray( "Mode1", array( "Player", "Spot", "Coord." ) )
    dArray( "SaveMode", array( "Current Position", "Entered Position" ) )
    dSetSval( "Mode", "Player" )
    dSetSval( "Mode1", "Player" )
    dSetSval( "SaveMode", "Current Position" )
    #GDO
    signalSetGroup( "GDO" )
    runOnSignal( "Accepted", 1, 1 )
    dLabel( "Garage Door Opener", "GDO", vec2( 5, 5 ) )
    dLabel( ErrorLable, "GDO", vec2( 405, 5 ) )
    dButton( "1", "GDO", vec2( Bx, By ), vec2( Bw, Bh ) )
    dButton( "2", "GDO", vec2( Bx + 1*( Bw + Bs ), By + 0*( Bh + Bs ) ), vec2( Bw, Bh ) )
    dButton( "3", "GDO", vec2( Bx + 2*( Bw + Bs ), By + 0*( Bh + Bs ) ), vec2( Bw, Bh ) )
    dButton( "4", "GDO", vec2( Bx + 0*( Bw + Bs ), By + 1*( Bh + Bs ) ), vec2( Bw, Bh ) )
    dButton( "5", "GDO", vec2( Bx + 1*( Bw + Bs ), By + 1*( Bh + Bs ) ), vec2( Bw, Bh ) )
    dButton( "6", "GDO", vec2( Bx + 2*( Bw + Bs ), By + 1*( Bh + Bs ) ), vec2( Bw, Bh ) )
    dButton( "7", "GDO", vec2( Bx + 0*( Bw + Bs ), By + 2*( Bh + Bs ) ), vec2( Bw, Bh ) )
    dButton( "8", "GDO", vec2( Bx + 1*( Bw + Bs ), By + 2*( Bh + Bs ) ), vec2( Bw, Bh ) )
    dButton( "9", "GDO", vec2( Bx + 2*( Bw + Bs ), By + 2*( Bh + Bs ) ), vec2( Bw, Bh ) )
    dLabel( "G.D.O.", "GDO", vec2( Lx, Ly ) )
    dLabel( "-ENTER CODE-", "GDO", vec2( Lx + 50 , Ly ) )
    dLabel( "--------STATUS--------", "GDO", vec2( Lx, Ly + Ls ) )
    dLabel( "           AUTOLOCK           ", "GDO", vec2( Lx, Ly + 2*Ls ) )
    dText( "--------STATUS--------", "IDLE                            " )
}
if( ~B1 & B1 ){
    dText( ErrorLable, " " )
    D = !D
    dShow( "Personal Device - Control Panel", D )
    enableMouse( D )
    dText( "Coordinates", "Coordinates ( x y z )" )
    dText( "Name", "Spot Name" )
    findByClass( "Player" )
    PlayersE = findToArray()
    for( I = 1, PlayersE:count() ){
        PlayersN[ I, string ] = PlayersE[ I, entity ]:name()
    }
    dArray( "Players", PlayersN )
    dArray( "Players1", PlayersN )
}
#ATS
if( Send ){
    Send = 0
}
if( clk( "Send" ) ){
    stoptimer( "Send" )
    Send = 1
}
if( Retrieve ){
    Retrieve = 0
}
if( clk( "Retrieve" ) ){
    stoptimer( "Retrieve" )
    Retrieve = 1
}
if( dClk( "Save" ) ){
    if( dSval( "SaveMode" ) == "Current Position" ){
        SaveSpot = owner():pos()

    }
    else{
        Array = dSval( "Coordinates" ):explode( " " )
        SaveSpot = vec( Array[ 1, string ]:toNumber(), Array[ 2, string ]:toNumber(), Array[ 3, string ]:toNumber() )
    }
    if( dSval( "Name" ) != "" ){
        for( I = 1, SpotsN:count() ){
            if( dSval( "Name" ) == SpotsN[ I, string ] ){
                HaveOne = 1
            }
        }
        for( I = 1, SpotsP:count() ){
            if( SaveSpot == SpotsP[ I, vector ] ){
                HaveOne = 1    
            }
        }
    } 
    if( !HaveOne ){
        SpotsN:pushString( dSval( "Name" ) )
        SpotsP:pushVector( SaveSpot )
        Array = array()
        dArray( "Spots", SpotsN )
        dArray( "Spots1", SpotsN )
    }
    else{
        timer( "Error", 2500 )
        dText( ErrorLable, "Coincidence of information" )
        HaveOne = 0
    }
}
if( clk( "Error" ) ){
    stoptimer( "Error" )
    dText( ErrorLable, "" )
}
if( dClk( "Send" ) | dClk( "Retrieve" ) ){
    Mode = dSval( "Mode" )
    Mode1 = dSval( "Mode1" )
    if( ( Mode == "Coord." | Mode1 == "Coord." ) & dSval( "Coordinates" ) == "Coordinates ( x y z )" ){
        Array = dSval( "Coordinates" ):explode( " " )
        Coord = vec( Array[ 1, string ]:toNumber(), Array[ 2, string ]:toNumber(), Array[ 3, string ]:toNumber() )
    }    
    if( Mode == "Coord." ){
        Origin = Coord
    }   
    elseif( Mode == "Spot" ){
        for( I = 1, SpotsN:count() ){
            if( SpotsN[ I, string ] == dSval( "Spots" ) ){
                break
            }
        }
        Origin = SpotsP[ I, vector ]
    }
    elseif( Mode == "Player" ){
        for( I = 1, PlayersN:count() ){
            if( PlayersN[ I, string ] == dSval( "Players" ) ){
                break
            }
        }
        Origin = PlayersE[ I, entity ]:pos()
    } 
    if( Mode1 == "Coord." ){
        Destin = Coord
  
    }
    elseif( Mode1 == "Spot" ){
        for( I = 1, SpotsN:count() ){
            if( SpotsN[ I, string ] == dSval( "Spots1" ) ){
                break
            }
        }
        Destin = SpotsP[ I, vector ]
    }   
    elseif( Mode1 == "Player" ){
        for( I = 1, PlayersN:count() ){
            if( PlayersN[ I, string ] == dSval( "Players1" ) ){
                break
            }

        }
        Destin = PlayersE[ I, entity ]:pos() + vec( 50, 50, 0 )
    }  
    if( Origin:distance( Destin ) >= 5 ){
        OX = Origin:x()
        OY = Origin:y()
        OZ = Origin:z()
        DX = Destin:x()
        DY = Destin:y()
        DZ = Destin:z()
        TE = dNval( "Teleport Everything" )
    }
    else{
        timer( "Error", 2500 )
        dText( ErrorLable, "Origin and Destination is the same" )
    }
} 
if( dClk( "Send" ) ){
    timer( "Send", 10 )
} 
if( dClk( "Retrieve" ) ){
    timer( "Retrieve", 10 )
}
#GDO
if( clk( "CodeL" ) ){
    CodeL = CodeL:sub( 1, CodeL:length() - 1 ) + "*"
    dText( "-ENTER CODE-", CodeL )
    stoptimer( "CodeL" )
}  
if( clk( "Reset" ) ){
    dText( "-ENTER CODE-", "-ENTER CODE-" )
    Code = CodeL = ""
    dText( "--------STATUS--------", "IDLE" )
    stoptimer( "Reset" )
} 
if( dClk() != "" & Code:length() < 12 ){
    timer( "Reset", 10000 )
    dText( "--------STATUS--------", "ENTERING CODE" )
    CLL = CodeL:length()
    if( CLL > 0 & CodeL:index( CLL ) != "*" ){
        CodeL = CodeL:sub( 1, CLL - 1 ) + "*"
        stoptimer( "CodeL" )
    }
    Code += dClk()
    CodeL += dClk()
    dText( "-ENTER CODE-", CodeL )
    timer( "CodeL", 1000 )
    if( CodeL:length() == 12 ){
        stoptimer( "Reset" )
        PrepareSend = 1
        dText( "--------STATUS--------", "CODE ENTERED" )
    }        
}
if( Sended & signalClk( "GDO", "Accepted", 1 ) & Open & !Inbound ){
    dText( "--------STATUS--------", "ACCEPTED" )
    Sended = 0
    entity():deleteWire( "Open" )
    entity():deleteWire( "Inbound" )
    stoptimer( "Sended" )
    stoptimer( "PrepareSend" )
    timer( "Reset", 10000 )
} 
if( clk( "Send1" ) & Open & !Inbound ){
    stoptimer( "Send1" )
    I1++
    dText( "--------STATUS--------", "SENDING CODE" )
    signalSend( Code:index( I1 ), 1 )
    if( I1 < 12 ){
        timer( "Send1", 10 )
    }  
    else{
        I = 0
        Sended = 1
        dText( "--------STATUS--------", "CODE SENDED" )
        timer( "Sended", 10000 )
    }
}      
if( PrepareSend ){
    Pos = owner():pos()
    findInSphere( Pos, 1000 )
    findClipToClass( "StarGate" )
    findSortByDistance( Pos )
    entity():createWire( find(), "Open", "Open" )
    entity():createWire( find(), "Inbound", "Inbound" )
    PrepareSend = 0
    timer( "Send1", 10 )
    timer( "PrepareSend", 10000 )   
}
if( clk( "Sended" ) | clk( "PrepareSend" ) ){
    dText( "--------STATUS--------", "NO RESPONSE" )
    timer( "Reset", 10000 )
}
