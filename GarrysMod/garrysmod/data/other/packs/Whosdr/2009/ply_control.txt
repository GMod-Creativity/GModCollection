@name 
@inputs W A S D Sp Sh Alt Use Att1 Att2 Sc1 Sc2 Driver:entity String:string
@outputs Pos:vector Eye:vector
@persist Pre:string Name:entity
@trigger 
if (first() | ~String) {Pre = "ulx cexec Strung" Name = findPlayerByName(String)}
runOnTick(1)
runOnChat(1)

if (!tickClk()) {
if (W) {concmd(Pre + "+forward")} else {concmd(Pre + "-forward")}
if (A) {concmd(Pre + "+left")} else {concmd(Pre + "-left")}
if (S) {concmd( "+back")} else {concmd(Pre + "-back")}
if (D) {concmd(Pre + "+right")} else {concmd(Pre + "-right")}
if (Sp) {concmd(Pre + "+Jump")} else {concmd(Pre + "-jump")}
if (Alt) {concmd(Pre + "+duck")} else {concmd(Pre + "-duck")}
if (Sh) {concmd(Pre + "+speed")} else {concmd(Pre + "-speed")}
if (Att1) {concmd(Pre + "+attack")} else {concmd(Pre + "-attack")}
if (Att2) {concmd(Pre + "+attack2")} else {concmd(Pre + "-attack2")}
if (Use) {concmd(Pre + "+use")} else {concmd(Pre + "-use")}
if (Sc1) {} else {}
if (Sc2) {} else {}
} else {
Pos = Name:pos() + vec(0,0,90)
Eye = Driver:driver():eye()
}


if (chatClk(Driver:driver())) {
concmd(Pre + "say "+ Driver:driver():lastSaid():replace("!","Failcakes!"))
}
