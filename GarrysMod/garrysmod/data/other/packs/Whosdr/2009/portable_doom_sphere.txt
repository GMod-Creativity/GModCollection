@name 
@inputs 
@outputs 
@persist Player:entity Pos:vector Dist:vector Entity:entity Largest
@trigger 
interval(20)

if (first()) {
holoCreate(1)
}

Entity = (entity():isWeldedTo():pos() != vec(0,0,0)) ? entity():isWeldedTo() : owner()
Pos = Entity:pos()
Largest = max(Entity:boxSize():x(),Entity:boxSize():y())
Dist = vec(Largest,Largest,Entity:boxSize():z())


holoPos(1,Pos + vec(sin(curtime()*100)*Largest,cos(curtime()*100)*Largest,45))



findExcludePlayer(owner())
findInBox(Pos,Dist)
findClipToClass("player")
Player = find()


if (Player & Player:isAlive() & Player != owner() ) {
#concmd("ulx slay "+Player:name():replace("{NN}",""))
print(Player:name())
}
