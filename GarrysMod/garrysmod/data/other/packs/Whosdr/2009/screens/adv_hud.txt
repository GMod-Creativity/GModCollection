@name 
@inputs Target:wirelink
@inputs H1:wirelink H2:wirelink H3:wirelink H4:wirelink 
@inputs H5:wirelink H6:wirelink H7:wirelink H8:wirelink 
@persist A:entity B:entity C:entity D:entity 
@persist E:entity F:entity G:entity H:entity
@persist AN:string BN:string CN:string DN:string
@persist EN:string FN:string GN:string HN:string
@trigger all
interval(100)
A = Target:entity("1_Ent")
B = Target:entity("2_Ent")
C = Target:entity("3_Ent")
D = Target:entity("4_Ent")
E = Target:entity("5_Ent")
F = Target:entity("6_Ent")
G = Target:entity("7_Ent")
H = Target:entity("8_Ent")
AN = "[" + A:name() + ":" + A:health() + "]"
BN = "[" + B:name() + ":" + B:health() + "]"
CN = "[" + C:name() + ":" + C:health() + "]"
DN = "[" + D:name() + ":" + D:health() + "]"
EN = "[" + E:name() + ":" + E:health() + "]"
FN = "[" + F:name() + ":" + F:health() + "]"
GN = "[" + G:name() + ":" + G:health() + "]"
HN = "[" + H:name() + ":" + H:health() + "]"

H1:setNumber("WorldX",A:pos():x())
H1:setNumber("WorldY",A:pos():y())
H1:setNumber("WorldZ",A:pos():z() + 150)
H1:setString("DisplayText",AN)

H2:setNumber("WorldX",B:pos():x())
H2:setNumber("WorldY",B:pos():y())
H2:setNumber("WorldZ",B:pos():z() + 150)
H2:setString("DisplayText",BN)

H3:setNumber("WorldX",C:pos():x())
H3:setNumber("WorldY",C:pos():y())
H3:setNumber("WorldZ",C:pos():z() + 150)
H3:setString("DisplayText",CN)

H4:setNumber("WorldX",D:pos():x())
H4:setNumber("WorldY",D:pos():y())
H4:setNumber("WorldZ",D:pos():z() + 150)
H4:setString("DisplayText",DN)

H5:setNumber("WorldX",E:pos():x())
H5:setNumber("WorldY",E:pos():y())
H5:setNumber("WorldZ",E:pos():z() + 150)
H5:setString("DisplayText",EN)

H6:setNumber("WorldX",F:pos():x())
H6:setNumber("WorldY",F:pos():y())
H6:setNumber("WorldZ",F:pos():z() + 150)
H6:setString("DisplayText",FN)

H7:setNumber("WorldX",G:pos():x())
H7:setNumber("WorldY",G:pos():y())
H7:setNumber("WorldZ",G:pos():z() + 150)
H7:setString("DisplayText",GN)

H8:setNumber("WorldX",H:pos():x())
H8:setNumber("WorldY",H:pos():y())
H8:setNumber("WorldZ",H:pos():z() + 150)
H8:setString("DisplayText",HN)






if (A:name() != "") {H1:setNumber("HideHUD",0)} else {H1:setNumber("HideHUD",1)}
if (B:name() != "") {H2:setNumber("HideHUD",0)} else {H2:setNumber("HideHUD",1)}
if (C:name() != "") {H3:setNumber("HideHUD",0)} else {H3:setNumber("HideHUD",1)}
if (D:name() != "") {H4:setNumber("HideHUD",0)} else {H4:setNumber("HideHUD",1)}
if (E:name() != "") {H5:setNumber("HideHUD",0)} else {H5:setNumber("HideHUD",1)}

if (G:name() != "") {H7:setNumber("HideHUD",0)} else {H7:setNumber("HideHUD",1)}
if (H:name() != "") {H8:setNumber("HideHUD",0)} else {H8:setNumber("HideHUD",1)}
