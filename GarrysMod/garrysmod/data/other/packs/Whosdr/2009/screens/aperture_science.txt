@name 
@inputs Screen:wirelink
@outputs
@persist X Text:string Y
@trigger All
if (first()) {Screen:setNumber("Reset",first())}
interval(3500)
X += 1
Y += 1
if (X == 1) {Text = "This was a triumph"}
if (X == 2) {Text = "Im making a note here"}
if (X == 3) {Text = "Huge success"}
if (X == 4) {Text = "Its hard to overstate my satisfaction"}
if (X == 5) {Y += 1 Text = "Aperture science"}
if (X == 6) {Text = "We do what we must because we can"}
if (X == 7) {Y += 1  Text = "For the good of all of us"}
if (X == 8) {Text = "Except the ones who are dead"}
if (X == 9) {Text = "But theres no sense cring over every mistake"}
if (X == 10) {Y += 1 Text = "You just keep on trying till you run out of cake"}
if (X == 11) {Y += 1  Text = "And the science gets done"}
if (X == 12) {Text = "And you make a new gun"}
if (X == 13) {Text = "For the people who are"}
if (X == 14) {
Screen:setNumber("Reset",1) Screen:setNumber("Reset",0)   
Y = -1
Screen:writeString("Still alive",12,8,911)
}
if (X >= 15 & X <= 20) {Text = ""}
if (X == 20) {Screen:setNumber("Reset",1) Screen:setNumber("Reset",0)}
Screen:writeString(Text,0,Y,911)
