@name 
@inputs Key Screen:wirelink
@outputs X Y
@persist X Y
@trigger all
if (first()) {Screen:setNumber("Reset",1) X = 0}
if (Key & ~Key & Key != 127 & Key != 13 & Key != 20 & Key != 19 & Key != 156) {Screen:writeString(toChar(Key),X,Y,999) X += 1}
if (Key & ~Key & Key == 127 & X != 0 & X > 0) {Screen:writeString(" ",X-1,Y,999), X -= 1}
if (Key & ~Key & Key == 127 & X == 0 & Y >= 1) {Screen:writeString(" ",29,Y-1,999), X = 29 Y -= 1}
if (X == 30) {X = 0 Y += 1}
if (Key == 13 & Y < 17) {X = 0 Y += 1}
if (Key == 20) {X += 1}
if (Key == 19) {X -= 1}
if (Key == 158 & Y > 1) {Y -= 1}
if (Key == 9) {Screen:setNumber("Reset",1) Screen:setNumber("Reset",0) X = 0 Y = 0}
