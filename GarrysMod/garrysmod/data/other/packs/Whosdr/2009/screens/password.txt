@name 
@inputs A B C D Clear Alert
@outputs Screen Ind
@persist Code:string CCode:string
@trigger all
if (first()) {Code = "1234"}

if (CCode:length() < 4) {

if (A & ~A) {CCode = CCode + "1"}
if (B & ~B) {CCode = CCode + "2"}
if (C & ~C) {CCode = CCode + "3"}
if (D & ~D) {CCode = CCode + "4"}

}

if (Code == CCode & Alert & ~Alert) {
Ind = 1
} else {
Ind = 0

}

Screen = CCode:toNumber()
#print(CCode)


if (Clear & ~Clear) {Ind = 0, CCode = "", Screen = 0}
