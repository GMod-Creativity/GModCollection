@name 
@inputs Screen:wirelink W A S D Active
@outputs 
@persist Dir Pos:vector2 OldPos:vector2 Target:vector2 Points Playing
@trigger 
interval(150)
if (first()) {
Dir = 1
Pos = vec2(24,24)
Screen[1048574]=1
Screen[1048569]=3
Target = vec2(randint(1,49),randint(1,49)) 
Screen[Target:x() + (Target:y()*50)] = 999
Playing = 1
}

if (W) {Dir = 1}
if (A) {Dir = 2}
if (S) {Dir = 3}
if (D) {Dir = 4}

OldPos = Pos

if (clk() & inrange(Pos,vec2(2,2),vec2(49,49)) ) {

if (Dir == 1) {
Pos += vec2(0,-1)
} elseif (Dir == 3) {
Pos += vec2(0,1)
} elseif (Dir == 2) {
Pos += vec2(-1,0)
} elseif (Dir == 4) {
Pos += vec2(1,0)
}

} elseif (!inrange(Pos,vec2(2,2),vec2(49,49)) & Playing == 1) {
print("Game Over!")
print("You scored: " + Points)
Playing = 0
}


Screen[Pos:x() + (Pos:y()*50)] = 255000000
Screen[OldPos:x() + (OldPos:y()*50)] = 0


if (Pos == Target) {
Points += 1
print("Points: "+Points)
Target = vec2(randint(1,49),randint(1,49))
Screen[Target:x() + (Target:y()*50)] = 999
}
