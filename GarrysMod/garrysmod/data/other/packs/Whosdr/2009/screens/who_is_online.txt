@name 
@inputs Screen:wirelink
@outputs 
@persist Y Num:string Num2:array
@trigger all
interval(500)
if (first()) {Screen:setNumber("Reset",first()) Y = 0}
if (Y < 16) {Y += 1} else {stoptimer("interval")}
Num = "0":repeat(1-(floor(Y/10)))
if (~Num2) {Y = 0}
findByClass("player")
Num2=findToArray()

Screen:writeString("|  |                         |",0,Y,911)
Screen:writeString("------------------------------",0,0,911)
Screen:writeString("------------------------------",0,17,911)
Screen:writeString(Num + Y,1,Y,999)
Screen:writeString(Num2:entity(Y):name(),5,Y,159)
