@name 
@inputs Screen:wirelink Up Down Left Right Mouse1
@outputs X Y
@persist On
@trigger 
if (first()) {
Screen[1048569]=3
Screen[1048574]=1
X = 24
Y = 24
On = 1
}


if (Down & Y < 48) { Y++ timer("down",200)} else {stoptimer("down")}
if (Up & Y > 0) {Y-- timer("up",200)} else {stoptimer("up")}

if (Right & X < 48) {X++ timer("right",200)} else {stoptimer("right")}
if (Left & X > 0) {X-- timer("left",200)} else {stoptimer("left")}

if (Mouse1) {On = !On}


if (clk("left")) {X--}
if (clk("right")) {X++}
if (clk("up")) {Y--}
if (clk("down")) {Y++}

if (On) {
Screen[X+(Y*50)] = 999000000 + random(999999)
}
