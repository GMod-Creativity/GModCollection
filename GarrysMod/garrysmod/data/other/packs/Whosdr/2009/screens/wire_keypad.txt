@name 
@inputs Screen:wirelink Key Clear
@outputs Active
@persist Char:string Password:string CPassword:string While Secure
@trigger 
if (first()) {
Password = "4126"
Screen[2041] = 1
Screen:writeString("      Insert Function Here    ",0,3,1,139)
Screen:writeString("                              ",0,15,0,900)
Screen:writeString("           Password           ",0,16,999)
Screen:writeString("                              ",0,17,0,900)
Secure = 0 #Change this for secure mode

}

if (Key & ~Key) {

if (CPassword:length() < Password:length() & !While) {
if (Key != 127) {
Char = Key:toString()
CPassword += Char
if (Secure) {
Screen:writeString(" ":repeat((30-CPassword:length())/2) + "*":repeat(CPassword:length()) + " ":repeat((30-CPassword:length())/2-1),0,16,999)
} else {
Screen:writeString(" ":repeat((30-CPassword:length())/2) + CPassword + " ":repeat((30-CPassword:length())/2-1),0,16,999)
}
} else {
CPassword = CPassword:sub(1,CPassword:length() -1)
if (Secure) {
Screen:writeString("*":repeat(CPassword:length())+" ",0,16,999)
} else {
Screen:writeString(CPassword+" ",0,16,999)
}
}
}

if (CPassword:length() == Password:length()) {

if (CPassword == Password) {
Active = 1
timer("Active",10000)
Screen:writeString("        Access Accepted       ",0,16,90)
While = 1
} else {
timer("Reset",10000)
While = 1
Screen:writeString("        Access Denied         ",0,16,900)
}
}
}


if (clk("Active")) {
Active = 0
Screen:writeString("           Password           ",0,16,999)
CPassword = ""
While = 0
}


if (clk("Reset")) {
Screen:writeString("           Password           ",0,16,999)
CPassword = ""
While = 0
}
 


if (Clear) {
CPassword = ""
Screen:writeString("           Password           ",0,16,999)
}
