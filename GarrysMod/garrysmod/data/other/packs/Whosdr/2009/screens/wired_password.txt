@name 
@inputs Screen:wirelink Key
@outputs Active
@persist Char:string Password:string CPassword:string
@trigger 
if (first()) {
Password = "tonyb"}

if (Key & ~Key) {

if (CPassword:length() < Password:length()) {
Char = toChar(Key)
CPassword += Char
Screen:writeString("*":repeat(CPassword:length()),0,0,999)
}

if (CPassword:length() == Password:length()) {

if (CPassword == Password) {
Active = 1
timer("Active",5000)
Screen:writeString("Access Accepted               ",0,0,90)
} else {
timer("Reset",5000)
Screen:writeString("Access Denied                 ",0,0,9)
}
}
}


if (clk("Active")) {
Active = 0
Screen:writeString("                              ",0,0,999)
CPassword = ""
}


if (clk("Reset")) {
Screen:writeString("                              ",0,0,999)
CPassword = ""
}
