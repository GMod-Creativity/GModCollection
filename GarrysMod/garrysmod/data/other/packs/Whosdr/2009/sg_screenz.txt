@name 
@inputs Chevron Inbound Active Open Fast Ent:entity
@outputs Screen
@persist 
@trigger 
if (!Open) {Ent:setSkin(6)}#static
if ((Active & !Open) | (Chevron > 0 & Chevron < 7)) {Ent:setSkin(9+Fast)}#dialing


if (Open) {
if (Inbound) {Ent:setSkin(8)} else {Ent:setSkin(11)}#dialed
}

Screen = Ent:getSkin()
