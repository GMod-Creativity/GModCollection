@name 
@inputs Key
@outputs R G B 
@persist Vec4:vector Table:table X String:string
@trigger 

if (first()) {
findByClass("player")
for (X = 1,findToArray():count()) {
Table[findResult(X):name(),vector] = vec(random(0,255),random(0,255),random(0,255))
}
String = ""
}

if (Key != 13) {String += toChar(Key) print(String)} else {
Vec4 = Table[findPlayerByName(String):name(),vector]
String = ""
R = Vec4:x()
G = Vec4:y()
B = Vec4:z()

entity():setColor(R,G,B)
}
