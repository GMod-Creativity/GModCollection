@name 
@inputs On Habit Press Temp Grav
@outputs 
@persist S:string X G
@trigger all
interval(1000)
runOnChat(1)
chatClk()
if (On) {S = "On"} else {S = "Off"}
if (owner():lastSaid():lower() == "status") {
G = 1
}
if (clk() & G) {
X += 1
if (X == 1) {print("Status: "+ S)}
if (X == 2 & On) {print("Habitable: "+Habit)}
if (X == 3 & On) {print("Pressure: "+Press)}
if (X == 4 & On) {print("Temp: "+Temp)}
if (X == 5 & On) {print("Gravity: "+Grav)}
if (X == 6) {concmd("say End.") G = 0 X = 0}
}
