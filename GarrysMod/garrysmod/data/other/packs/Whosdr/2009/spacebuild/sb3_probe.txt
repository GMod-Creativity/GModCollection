@name 
@inputs O2 Co2 Nitrogen Hydrogen Empty Pressure Tempereture Gravity On
@outputs 
@persist S:string X G
@trigger all
interval(1000)
runOnChat(1)
chatClk()
if (On) {Status = "On"} else {Status = "Off"}
if (owner():lastSaid():lower() == "status") {
G = 1
}
if (clk() & G) {
X += 1
if (X == 1) {print("Status: "+ S)}
if (X == 2) {print("O2: "+ O2)}
if (X == 3) {print("Co2: "+Co2)}
if (X == 4) {print("Nitro: "+Nitrogen)}
if (X == 5) {print("Hydro: "+Hydrogen)}
if (X == 6) {print("Empty Air: "+ Empty)}
if (X == 7) {print("Pressure: "+Pressure)}
if (X == 8) {print("Temp: "+Tempereture)}
if (X == 9) {print("Gravity: "+Gravity)}
if (X == 10) {concmd("say End.") G = 0 X = 0}

}
