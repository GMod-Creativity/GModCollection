@name 
@inputs Active
@outputs Dial Close
@persist X Y
@trigger all
runOnChat(1)
chatClk(owner())
interval(10)
if(lastSaid():explode(" "):string(1) == "Dial" & Y == 1) {
Lastsaid=owner():lastSaid()
S1 = toByte(Lastsaid,6)
S2 = toByte(Lastsaid,7)
S3 = toByte(Lastsaid,8)
S4 = toByte(Lastsaid,9)
S5 = toByte(Lastsaid,10)
S6 = toByte(Lastsaid,11)

if (X <= 7) {X += 1}
}


if (X == 1) {Dial = S1}
if (X == 2) {Dial = S2}
if (X == 3) {Dial = S3}
if (X == 4) {Dial = S4}
if (X == 5) {Dial = S5}
if (X == 6) {Dial = S6}
if (X == 7) {Dial = 13}
if (X == 8) {X = -1 Y = 0}

if (chatClk()) {Y = 1 Close = 0}

if (Active & owner():lastSaid() == "Close") {Close = 1}
