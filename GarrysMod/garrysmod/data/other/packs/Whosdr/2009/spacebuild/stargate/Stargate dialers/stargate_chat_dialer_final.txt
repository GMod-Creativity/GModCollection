@name 
@inputs 
@outputs Dial
@persist X Y
@trigger all
runOnChat(1)
chatClk(owner())
interval(10)
if(lastSaid():explode(" "):string(1) == "Dial" & Y == 1) {
S1 = toByte(lastSaid():explode(""):string(6))
S2 = toByte(lastSaid():explode(""):string(7))
S3 = toByte(lastSaid():explode(""):string(8))
S4 = toByte(lastSaid():explode(""):string(9))
S5 = toByte(lastSaid():explode(""):string(10))
S6 = toByte(lastSaid():explode(""):string(11))
if (X <= 7) {X += 1}
}


if (X == 1) {Dial = S1}
if (X == 2) {Dial = S2}
if (X == 3) {Dial = S3}
if (X == 4) {Dial = S4}
if (X == 5) {Dial = S5}
if (X == 6) {Dial = S6}
if (X == 7) {Dial = 13}
if (X == 8) {X = -1 Y = 0}

if (chatClk()) {Y = 1}
