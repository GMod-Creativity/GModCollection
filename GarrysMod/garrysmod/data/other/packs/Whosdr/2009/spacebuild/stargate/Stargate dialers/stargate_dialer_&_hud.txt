@name 
@inputs Stargate:wirelink Iris:wirelink
@outputs String:string
@persist X Y T I:string SG:string IN:string Add:string Address:string
@trigger all
runOnChat(1)
chatClk(owner())
interval(200)
if(owner():lastSaid():explode(" "):string(1):lower() == "!dial" & Y == 1 & owner():lastSaid():length() == 12) {
Lastsaid=owner():lastSaid()
S1 = toByte(Lastsaid,7)
S2 = toByte(Lastsaid,8)
S3 = toByte(Lastsaid,9)
S4 = toByte(Lastsaid,10)
S5 = toByte(Lastsaid,11)
S6 = toByte(Lastsaid,12)

if (X <= 7) {X += 1}
}


if(owner():lastSaid():explode(" "):string(1):lower() == "!dial" & Y == 1 & owner():lastSaid():length() != 12) {
timer("error",1000)
}
if (clk("error")) {
stoptimer("error")
print("You have entered an invalid code.")
}

if (X == 1) {Stargate:setNumber("Dial Address",S1)}
if (X == 2) {Stargate:setNumber("Dial Address",S2)}
if (X == 3) {Stargate:setNumber("Dial Address",S3)}
if (X == 4) {Stargate:setNumber("Dial Address",S4)}
if (X == 5) {Stargate:setNumber("Dial Address",S5)}
if (X == 6) {Stargate:setNumber("Dial Address",S6)}
if (X == 7) {Stargate:setNumber("Dial Address",13)}
if (X == 8) {X = -1 Y = 0}

if (chatClk(owner())) {Y = 1}

if (Stargate:number("Active") == 1 & owner():lastSaid():upper() == "!CLOSE") {Stargate:setNumber("Close",1)}

if (owner():lastSaid():lower() == "!disable") {T = 1}
elseif (owner():lastSaid() == "!enable")
{T = 0}
else {T = T}

if (T == 1 & clk()) {Stargate:setNumber("Close",1) Stargate:setNumber("Close",0)}


if (owner():lastSaid():lower() == "!slow") {Stargate:setNumber("Dial Mode",0)}
elseif (owner():lastSaid():lower() == "!fast") {Stargate:setNumber("Dial Mode",1)}

if (owner():lastSaid():lower() == "!iris" & !clk() & lastSpoke() == owner()) {Iris:setNumber("Activate",!Iris:number("Activated"))}


Address = toChar(S1) + toChar(S2) + toChar(S3) + toChar(S4) + toChar(S5) + toChar(S6)



if (!clk()) {
if (first()) {Add = "[N/A]"}
if (Iris:number("Activated")) {I = "[Iris:Closed]"} else {I = "[Iris:Open]"}
if (!Stargate:number("Chevron")) {SG = "[Stargate:Static]"}
if (Stargate:number("Chevron")) {SG = "[Stargate:Dialing]"}
if (Stargate:number("Chevron") == 7) {SG = "[Stargate:Active]"}
if (Stargate:number("Inbound") & Stargate:number("Active")) {IN="[Inbound:Dialed in]"}
if (!Stargate:number("Inbound") & Stargate:number("Active")) {IN="[Inbound:Dialed out]"}
if (!Stargate:number("Active")) {IN="[Inbound:N/A]"}
if (Address != "") {
Add = "[Address:"+ Address + "]"
} else {
Add = "[Address:N/A]"
}
String = I + "    " + SG + "    " + IN + "    " + Add
}


if (owner():lastSaid():index(1) == "!") {hideChat(1)} else {hideChat(0)}
