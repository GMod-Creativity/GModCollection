@name Stargate status
@inputs Iris Chevron Inbound Active Address:string
@outputs String:string
@persist I:string SG:string IN:string Add:string
@trigger all
if (first()) {Add = "[N/A]"}
if (Iris == 1) {I = "Closed"} else {I = "Open"}
if (Chevron == 0) {SG = "Static"}
if (Chevron) {SG = "Dialing"}
if (Chevron == 7) {SG = "Active"}
if (Inbound & Active) {IN="Dialed in"}
if (!Inbound & Active) {IN="Dialed out"}
if (!Active) {IN="N/A"}
String = "[SG:"+SG+"]    [Inbound:" + IN + "]    [Iris:"+I+"]    " + Add

 if (Active & Address != "") {Add = "[Address:"+ Address + "]"} 
elseif (!Active) {Add = "[Address:N/A]"}
