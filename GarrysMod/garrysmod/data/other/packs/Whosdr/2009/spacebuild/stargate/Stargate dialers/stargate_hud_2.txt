@name Stargate status
@inputs Iris Chevron Inbound Active Address:string
@outputs String:string
@persist I:string SG:string IN:string Add:string
@trigger all
if (first()) {Add = "[N/A]"}
if (Iris == 1) {I = "[Iris:Closed]"} else {I = "[Iris:Open]"}
if (Chevron == 0) {SG = "[Stargate:Static]"}
if (Chevron) {SG = "[Stargate:Dialing]"}
if (Chevron == 7) {SG = "[Stargate:Active]"}
if (Inbound & Active) {IN="[Inbound:Dialed in]"}
if (!Inbound & Active) {IN="[Inbound:Dialed out]"}
if (!Active) {IN="[Inbound:N/A]"}
if (Active & Address != "") {Add = "[Address:"+ Address + "]"} 
elseif (!Active) {Add = "[Address:N/A]"}

String = I + "    " + SG + "    " + IN + "    " + Add
