@name 
@inputs Active On
@outputs 
@persist Props:array Prop:entity Vec:vector
@trigger 
if (first()) {
findByModel("models/Zup/sg_rings/ring.mdl")
findClipToClass("prop_physics")
Props = findToArray()
}

runOnTick(1)

for (X = 0,4) {
Prop = Props[X,entity]
Vec = -Prop:pos() - Prop:vel()           + entity():pos() + vec(0,0,200+(!Active*300)) - vec(0,0,X*50*Active*!On) - vec(0,0,X*20*!Active)
Prop:applyForce(Vec*Prop:mass())
Prop:applyAngForce((-Prop:angles() - Prop:angVel())*Prop:mass())
}
