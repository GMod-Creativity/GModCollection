@name 
@inputs Next Prev Active M1 M2 Ent:entity
@outputs Pos:vector Eye:vector Name:string Toggle
@persist X Players:array Player:entity ET
@trigger 
if (changed(Active)) {
findByClass("player")
findExcludePlayer(Ent:driver())
Players = findToArray()
X = 1
}
runOnTick(1)
if (Next & ~Next) {if (X < Players:count()) {X++}}
if (Prev & ~Prev) {if (X > 1) {X--}}
if (~Next | ~Prev) {
Player = Players[X,entity]
Name = Player:name()
}

if (M1 & M1) {Toggle = !Toggle}
if (!Active) {Toggle = 0}


Pos = Player:pos() + vec(0,0,90)
if (M2 & ~M2) {ET = !ET}
if (ET) {Eye = Player:eye()} else {Eye = Ent:driver():eye()}
