@name 
@inputs Active Iris
@outputs Close Recharge IOpen Set

if (first()) {
Recharge = 10 #start energy
} else {
interval(1000) #speed
}

if (Active & Recharge) {Recharge--}
if (Active & !Recharge) {Close = 1}
if (!Active & Recharge < 100) {Recharge++}
if (Iris*Recharge & Recharge >= 1) {Recharge-- Set = Iris} else {Set = 0}
IOpen = Iris*Recharge
