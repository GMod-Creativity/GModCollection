@name 
@inputs 
@outputs Pos:vector
@persist LS:array CPlayer Player:string Class:string Model:string Ent:entity Aim
@trigger 
runOnChat(1)

if (chatClk(owner()) & !clk("pos")) {

LS = owner():lastSaid():explode(" ")
if (LS[1,string]:lower() == "!class") {
Class = LS[2,string]
if (LS[2,string]:lower() != "player") {CPlayer = 0} else {CPlayer = 1}
}


if (LS[1,string]:lower() == "!player") {
if (CPlayer == 0) {print("Class is not set to player")} else {
Player = findPlayerByName(LS[2,string]):name()
}
}


if (LS[1,string]:lower() == "!model") {
Model = LS[2,string]
}

if (owner():lastSaid() == "!reset") {
CPlayer = 0
Player = ""
Class = ""
Model = ""
}

if (owner():lastSaid() == "!search") {
if (CPlayer == 1) {findIncludePlayer(Player)} else {findExcludePlayer("")}
findByClass(Class)
findClipToModel(Model)
Ent = find()
Aim = 1
}

if (owner():lastSaid():lower() == "!aim") {Aim = !Aim}

}


timer("pos",20)
if (!Aim) {
Pos = Ent:pos() + vec(0,0,80)
} else {
Pos = owner():aimEntity():pos() + vec(0,0,80)
}
