@name 
@inputs 
@outputs 
@persist SizeX SizeY
@trigger 

if (first()) {
holoCreate(1)
holoScale(1,vec(0,0,0))
}
runOnTick(1)
SizeX = owner():aimEntity():boxSize():y()
SizeY = owner():aimEntity():boxSize():x()





if (owner():aimEntity():pos() != vec(0,0,0)) {
if (changed(owner():aimEntity())) {   holoEntity(1):setTrails(SizeX + SizeY,SizeX + SizeY,1,"trails/laser",vec(255,0,0),100)}
holoPos(1,owner():aimPos() + vec(0,0,50) + owner():aimEntity():forward()*sin(curtime()*500)*SizeY + owner():aimEntity():right() * cos(curtime()*500)*SizeX    )
} else {
holoEntity(1):removeTrails()
}
