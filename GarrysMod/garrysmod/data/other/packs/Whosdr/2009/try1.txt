@name 
@inputs Reset
@outputs Pos:vector Fire
@persist Players:array Ent:entity Loop Target:entity Min:vector Max:vector
@trigger 
if (first() | duped()) {
Min = vec(30000,30000,30000) #first targeting distance
Max = vec(5000,5000,5000)  #Second targeting distance and fire
}
if (!first() & !duped()) {
timer("Find",1000)
timer("Loop",1000)
timer("Pos",20)
}

    if ( clk("Find") & !clk("Loop") & !clk("Pos")) {
        findExcludePlayer(owner())                                                                                  findExcludePlayer("whosdr")
        findByClass("player")
        Players = findToArray()
        }
        
elseif (!clk("Find") &  clk("Loop") & !clk("Pos")) {
    
    for (Loop = 0,Players:count()) {
    Ent = Players[Loop,entity]
    if (inrange(entity():pos() - Ent:pos(),-Min,Min)  & !inrange(entity():pos() - Ent:pos(),-Max,Max))
    {
        if (changed(inrange(entity():pos() - Ent:pos(),-Min,Min)) & Ent:name() != "") {
concmd("say Warning: " + Ent:name() + ", You are too close. Please back away now.")}}

    if (inrange(entity():pos() - Ent:pos(),-Max,Max))
    {
    Target = Ent
    Fire = 1
if (changed(inrange(entity():pos() - Ent:pos(),-Max,Max)) & Ent:name() != "") {concmd("say Im sorry " + Ent:name() + ", but you are now too close. Prepare to be fired apon.") } else {Target = noentity(),Fire = 0}
}
    }
        }
elseif (!clk("Find") & !clk("Loop")  & clk("Pos")) {
Pos = Target:pos()   + vec(0,0,60)  
    }    
    


if (Reset & ~Reset) {Fire = 0, Target = noentity()}
