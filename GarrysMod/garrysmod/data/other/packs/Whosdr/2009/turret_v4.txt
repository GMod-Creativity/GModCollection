@name Chat-Turret
@outputs Name:string Pos:vector Fire Active
@persist Mode:string Owner:entity Target:entity Player:entity 
if (first()) {
Owner = owner()
Player = Owner
Mode = "target"
#If weapon is automatic, wire to Active. If not, and it requires
#A pulse, wire to Fire.
}


timer("update",20)
runOnChat(1)

if (clk("update")) {

if (Mode == "target") {
Pos = Target:pos() + vec(0,0,30-(Target:isCrouch()*15))
} elseif (Mode == "aim") {
Pos = Player:aimPos()
if (Player:keyAttack2()) {Active = 1} else {Active = 0}
}
if (Active) {Fire = !Fire} else {Fire = 0}


}



if (chatClk(owner())) {

if (owner():lastSaid():explode(" ")[1,string] == "-owner") {
Player = findPlayerByName(owner():lastSaid():explode(" ")[2,string])
}

if (owner():lastSaid() == "-me") {Player = Owner}

}


if (chatClk(Player)) {
if (Player:lastSaid():explode(" ")[1,string] == "-t") {
Target = findPlayerByName(Player:lastSaid():explode(" ")[2,string])
if (!Target) {Target = noentity(), print("No target.")} else {print("Target = "+Target:name()) Name = Target:name()}
Mode = "target"
}
if (Player:lastSaid() == "!" | Owner:lastSaid() == "!") {Active = !Active print("Firing: "+ (Active) ? "On" : "Off")}

if (Player:lastSaid() == "-aim") {Mode = "aim"}

}


if (chatClk(owner())) {
if (owner():lastSaid():index(1) == "!" | owner():lastSaid():index(1) == "-") {
hideChat(1)
print(owner():lastSaid())
} else {
hideChat(0)
}
}
