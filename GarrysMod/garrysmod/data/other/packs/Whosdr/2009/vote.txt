@name 
@inputs 
@outputs 
@persist Players:table Yes No Text:string Player:array
@trigger 
runOnChat(1)
if (owner():lastSaid():sub(1,5) == "!vote" & chatClk(owner())) {
Text = owner():lastSaid():sub(6,1000)
concmd("say [vote]"+Text)
Players = table()
Yes = 0, No = 0
timer("end",60000)
findByClass("player")
Player = findToArray()
}


if (lastSaid() == "!yes" & !Players[lastSpoke():name(),number]) {
Yes += 1
Players[lastSpoke():name(),number] = 1
}

if (lastSaid() == "!no" & !Players[lastSpoke():name(),number]) {
No += 1
Players[lastSpoke():name(),number] = 1
}


if (clk("end") | (Yes+No == Player:count())   & !first()) {
stoptimer("end")
print("[vote]Voting over. The results are Yes:"+Yes+" No:"+No)
}
