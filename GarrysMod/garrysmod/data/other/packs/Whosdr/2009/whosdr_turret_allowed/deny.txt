@name 
@inputs 
@outputs Pos:vector Fire Name:string
@persist Array:array Find:entity Player:entity X
@trigger 

interval(100)

if (Player == noentity()) {
findByClass("player")
findClipToSphere(entity():pos(),5000)
Array = findToArray()
if (X > Array:count()) {X = -1}
Find = Array[X,entity]
if (fileRead("/allow/"+Find:name()+".txt") == "False") {Player = Find} else {X++ Player = noentity()}
}

timer("Pos",10)
if (clk("Pos")) {
Pos = Player:pos()
if (Player:isAlive()) {Fire = 1} else {Player = noentity() Fire = 0}
Name = Player:name()
}
