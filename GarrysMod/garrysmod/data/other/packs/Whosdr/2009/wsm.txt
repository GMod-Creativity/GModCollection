@name 
@inputs Screen:wirelink W A S D
@outputs Points
@persist Pos:vector2 Old:vector2 X Dir
@persist [BoxU BoxD BoxL BoxR]:vector2 Jump
@persist PU PD PL PR

@trigger none

interval(100)

if (first()) {
Screen[1048574] = 1 #reset
Screen[1048569] = 3 #color mode
    for (X = 0,32) {
    Screen[X+(15*32)] = 111999000 #draws green floor
    
    }
    Screen[7+(13*32)] = 999999999 #draws white box
    Screen[8+(13*32)] = 999999999 #draws white box
    Screen[9+(13*32)] = 999999999 #draws white box
    Screen[10+(13*32)] = 999999999 #draws white box

for (Y = 1,12,+2) {
for (X = 0,6) {
Screen[random(32)+(Y*32)] = 999999999
}
}

    Dir = 0
    Pos = vec2(0,14)
}
Dir = 3
if (changed(W)) {Jump = 1}
if (W) {Dir = 1}
if (A) {Dir = 2}
if (S) {Dir = 3}
if (D) {Dir = 4}


BoxU = Pos + vec2(0,-1)
BoxD = Pos + vec2(0, 1)
BoxL = Pos + vec2(-1,0)
BoxR = Pos + vec2(1,0)
PU = Screen[BoxU:x()+BoxU:y()*32]
PD = Screen[BoxD:x()+BoxD:y()*32]
PL = Screen[BoxL:x()+BoxL:y()*32]
PR = Screen[BoxR:x()+BoxR:y()*32]

Old = Pos

if (Dir == 1 & Jump & Pos:y()+vec2(0,-2):y() > 0) {
if (PD != 0) {
if (PU == 0) {
Pos += vec2(0,-2)
}


if (PU == 999999999) {
Points += 10
Screen[BoxU:x()+(BoxU:y()*32)] = 555555555
}



} else {Pos += vec2(0,2) Jump = 0}
}

elseif (Dir == 3 & PD == 0) {Pos += vec2(0,2)}
elseif (Dir == 2 & PL == 0) {Pos += vec2(-1,0)}
elseif (Dir == 4 & PR == 0) {Pos += vec2( 1,0)}


Screen[Pos:x()+(Pos:y()*32)] = 222000000
if (changed(Pos) & !first()) {
Screen[Old:x()+(Old:y()*32)] = 0
}


if (clk("Jump")) {Jump = 0}
