@name 2D Projector
@inputs Entities:array
@outputs 
@persist SignificantFigures PropIndex PropCount [Prefix PrintCoordinatesKey]:string DimensionMul:vector2
@trigger none
@model models/sprops/misc/origin.mdl

if(first() | dupefinished()) {
    
    # ~ Unit Conversion ~ #
    
    DistanceFrom = 192      #GMOD unit (inches) reference
    
    DistanceTo = 80         #Reference unit to convert to
    
    # ~ Special ~ #
    
    Prefix = "LineInd++, LineVertices[LineInd,vector2] = "
    #Prefix = "PolyInd++, PolyVertices[PolyInd,vector2] = "
    
    SignificantFigures = 3      #Which decimal place to round to
    
    DimensionMul = vec2(1, -1)      #Multiplier of the corrdinate that get's printed out
    
    # ~ Keybinds ~ #
    
    PrintCoordinatesKey = "G"   #Press this to start printing coordinates
    
    # # # # # # # # # # # # # # # # # # # # # # # # # # # 
    
    DimensionMul = DimensionMul * (DistanceTo / DistanceFrom)
    
    PropIndex = 0
    PropCount = 0
    
    runOnKeys(owner(), 1)
}

#Start printing coordinates
if(changed(owner():keyPressed(PrintCoordinatesKey)) & owner():keyPressed(PrintCoordinatesKey)) {
    
    PropIndex = 0
    PropCount = Entities:count()
    
    print("Printing...")
    
    timer("PrintDelay", 210)
}

#Actually print the coordinates
if(clk("PrintDelay")) {
    
    PropIndex++
    
    local LocalV2 = round(vec2( entity():toLocal(Entities[PropIndex,entity]:pos()) ) * DimensionMul, SignificantFigures)
    local Suffix = "vec2(" + LocalV2[1] + ", " + LocalV2[2] + ")"
    
    print(Prefix + Suffix)
    
    #Keep going
    if(PropIndex < PropCount) {timer("PrintDelay", 210)}
    
    #Done
    else {print("Done!")}
}
