@name Axis Arm Maker r7
@model models/sprops/misc/origin.mdl
@inputs Marker:array
@persist [User]:entity SelfDestruct [Constrain]:string WheelCount [HingeOffset]:vector
@trigger Button

if(first() | dupefinished()) {
    
    SelfDestruct = 0            #Destroy the expression after hitting the button?
    
    Constrain = "T"         #Press this to make the constraint.
    
    HingeOffset = vec(19.031, 0, 11.233)
    
    # # # # # # # # # # # # # # # # # # # # #
    
    #User and runOnKeys to update the function
    User = owner()
    runOnKeys(User,1)
}

if(changed(User:keyPressed(Constrain)) & User:keyPressed(Constrain)) {
    
    local E2Pos = entity():pos()
    local OffsetRelE2 = entity():toWorld(HingeOffset)
    local Cnt = 1
    
    for(I=2, Marker:count(), 1) {
        
        local Base = Marker[1,entity]
        local Hub = Marker[I,entity]
        
        Hub:constraintBreak("axis")
        
        #Suspension hinge
        local WorldOffset = OffsetRelE2 + (Hub:pos() - E2Pos)
        axis(Hub, Hub:toLocal(WorldOffset), Base, Base:toLocal(WorldOffset), 0, Hub:toLocal(WorldOffset + entity():right()))
        
        Cnt++
    }
    
    #Confirmation
    hint((Cnt-1) + " axi created!", 5)
    
    if(SelfDestruct) {selfDestruct()}
}
