@name Hydraulic Pivot
@inputs Angle_Desired
@outputs Length1 Length2 Length3 Length4 Length5
@persist HomePoint:vector2 FireOrder:array
@trigger none

#[
    
    Instructions:
    
        The orientation of the E2 does not matter!
    
        You may need to add Length outputs to @outputs. If you do, remember to set the Length output to a FireOrder slot.
    I.E.
    
    Length1 = FireOutput[1, number]
    Length2 = FireOutput[2, number]
    Length3 = FireOutput[3, number]
    
        It is helpful to picture your rotater and the points where the hydros connect to it from above and imagine it were on
    a X, Y coordinate plane!
    
]#

if(first() | dupefinished()) {
    
    # Settings
    
    HydraulicCount = 5          #How many hydros do you have
    
    HomePoint = vec2(0, -1000)  #The point where the hydros push/pull from
    
    RadiusLength = 500          #Radius where hydros connect to the Rotater
    
    AngOffset = 0               #Rotates the frame of reference of the system
    
    # # # # # # # # # # # # # # # # # # # # #
    
    #Organizing into array
    local AngDiv = 360 / HydraulicCount
    for(I=1, 3, 1) {FireOrder[I,vector2] = vec2(0, RadiusLength):rotate(AngOffset + (I-1) * AngDiv)}
}


interval(75)


#Actual fire order
local FireOutput = array()
for(I=1, 3, 1) {FireOutput[I,number] = HomePoint:distance( FireOrder[I,vector2]:rotate(Angle_Desired) )}

#Hydraulic lengths
Length1 = FireOutput[1, number]
Length2 = FireOutput[2, number]
Length3 = FireOutput[3, number]
Length4 = FireOutput[4, number]
Length5 = FireOutput[5, number]
