function string getAnimation(){
    if(Noclip == 0){
        if(User:keyPressed("F")){
            return "idle_magic"
        }
        else{
            if(User:keyDuck()){
                if(User:keyForward() | User:keyBack()){
                    return "cidle_all"
                }
                else{
                    return "cidle_all"
                }
            }
            else{
                if(User:keySprint()){
                    return "menu_walk"
                }
                else{
                    if(User:keyForward() | User:keyBack()){
                        return "menu_walk"
                    }
                    else{
                        return "idle_all_01"
                    }
                }
            }
        }
    }
    else{
        return "jump_fist"
    }
}
