@name Hop
#include "propcoreTeleport"
#include "Destinations"

if(first()){
    runOnChat(1)    
}

if(chatClk(owner())){
    CMD = owner():lastSaid():explode(" ")
    if(CMD[1,string] == ".hop"){
        hideChat(1)
        if(CMD[2,string] == "thru"){
            Seat = seatSpawn("models/Gibs/HGIBS.mdl",owner():aimPos() + owner():eyeAngles():forward() * 20,ang(vec(0)),1)
            Seat:use()
            Seat:propDelete()
        }
        else{
            Seat = seatSpawn("models/Gibs/HGIBS.mdl",owner():aimPos(),ang(vec(0)),1)
            Seat:use()
            Seat:propDelete()
        }
    }
    elseif(CMD[1,string] == ".port" | CMD[1,string] == ".go"){
        hideChat(1)
        
        if(CMD[2,string] == "#random" | CMD[2,string] == "#r"){
            Pos = players()[randint(1,players():count()),entity]:pos()
        }
        elseif(getDestination(CMD[2,string]) != vec(0)){
            Pos = getDestination(CMD[2,string])
        }
        else{
            Pos = findPlayerByName(CMD[2,string]):pos()
        }
        
        
        Seat = seatSpawn("models/Gibs/HGIBS.mdl",Pos,ang(vec(0)),1)
        Seat:use()
        Seat:propDelete()
    }
    
}
