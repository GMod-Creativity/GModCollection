@name Arcane Imbuer
@persist Angle:angle Angle2:angle Ply:entity Gravity Flight Cooldown Iterations
@outputs OnGround IsCrouch Shockwave
interval(20)
runOnLast(1)
OnGround = owner():isOnGround()
IsCrouch = owner():isCrouch()
#Setup#
if(first()){
Shockwave = 30
holoCreate(1)
holoCreate(2)
holoCreate(3)
holoCreate(4)
holoCreate(5)
Angle2 = Angle2 + ang(vec(0,180,0))
}
#Setup#


#First Holo#
Angle = Angle + ang(vec(0,1,0))

holoPos(1,owner():pos())
holoAng(1,Angle)
holoScale(1,vec(10,10,1))
holoMaterial(1,"models/props_combine/tprings_globe")
holoModel(1,"tetra")
#First Holo#


#Second Holo#
Angle2 = Angle2 + ang(vec(0,1,0))

holoPos(2,owner():pos())
holoAng(2,Angle2)
holoScale(2,vec(10,10,1))
holoMaterial(2,"models/props_combine/tprings_globe")
holoModel(2,"tetra")
#Second Holo#


#Circle#
holoModel(3,"hq_torus")
holoScale(3,vec(4,4,3))
holoPos(3,owner():pos())
holoColor(3,vec(256,100,0))
holoMaterial(3,"models/props_combine/stasisshield_sheet")
#Circle#



#Beam#


if(toString(owner():weapon())=="Weapon [393][gmod_camera]"){
holoModel(4,"cylinder")
holoScale(4,vec(0.5,500,0.5))
holoPos(4,owner():pos()+vec(0,0,58))
holoMaterial(4,"models/props_combine/tprings_globe")
holoAng(4,owner():eyeAngles()+ ang(vec(0,90,0)))}

else{holoScale(4,vec(0,0,0))}

#Beam#


#Shockwave#
holoModel(5,"cylinder")
holoPos(5,owner():pos())
holoColor(5,vec(256,0,0))

if(owner():isCrouch()){Shockwave = 0}

if(Shockwave <= 15){
    if(owner():isOnGround()){
    holoScale(5,vec(Shockwave, Shockwave, 0.5))
    Shockwave = Shockwave + 0.5}}

else{holoScale(5,vec(0,0,0))}
#Shockwave#

if(owner():keyPressed("LSHIFT")){
    if(owner():keyPressed("SPACE")){
        if(Cooldown == 0){
            Cooldown = 1
            Iterations = 20
        }
    }
if(owner():keyPressed("W")){owner():teleport(owner():pos()+owner():angles():forward() * 50)}

if(Iterations > 0){
    owner():teleport(owner():pos() + vec(0,0,20))
    Iterations -= 1
}}
if(owner():isOnGround()){Cooldown = 0}
if(owner():inNoclip()){}
