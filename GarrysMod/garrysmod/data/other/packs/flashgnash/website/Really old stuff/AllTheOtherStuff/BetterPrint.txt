function void printCore(Str:string,Prefix:vector,Text:vector,Highlight:vector,PrefixText:string){
    
    ExplodedMessage = Str:explode("")
    FinishedString = array(Prefix,PrefixText,Text)
    for( I=1,ExplodedMessage:count() ){
        Char = ExplodedMessage:shiftString()
        
        
        if(Char == "["){
            FinishedString:pushString(Char)
            FinishedString:pushVector(Highlight)
        }
        elseif(Char == "]"){
            FinishedString:pushVector(Text)
            FinishedString:pushString(Char)
        }
        else{
            FinishedString:pushString(Char)
        }
        
        
        
    }
    printColor(FinishedString)
}


function void print(Num){
    printCore(toString(Num),PrefixColour,TextColour,HighlightColour,PrefixText)
}
 
function void print(Str:string){
    printCore(Str,PrefixColour,TextColour,HighlightColour,PrefixText)
}




function void setPrefixColour(Vec:vector){
    PrefixColour = Vec
}

function void setTextColour(Vec:vector){
    TextColour = Vec
}

function void setHighlightColour(Vec:vector){
    HighlightColour = Vec
}

function void setPrefix(Text:string){
    PrefixText = Text
}


function void setColourScheme(Vec1:vector,Vec2:vector,Vec3:vector,Prefix:string){
    PrefixColour = Vec1
    TextColour = Vec2
    HighlightColour  = Vec3 
    PrefixText = Prefix
}
