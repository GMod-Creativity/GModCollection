@name Telekinesis V2
@persist Ent:entity Cooldown Toggle PermEnt:entity
if(first()){
    On = 0
    Cooldown = 0
}
Ply = owner()
interval(20)

if(dupefinished()){reset()}

if(clk("Cooldown")){Cooldown = 0}

if(clk("Parent")){PermEnt:parentTo(Ply)}

if(Ply:keyPressed("L")){
    if(Cooldown == 0){
    timer("Cooldown",500)
    Cooldown = 1
    Ent = Ply:aimEntity()

    if(Toggle){
        Toggle = 0
        PermEnt:deparent()
    }
    else{
        Toggle = 1
        PermEnt = Ent
        Ent:setPos(Ent:pos() + vec(0,0,Ent:height() / 2))
        timer("Parent",10)
        }
}}
