function array formatName(Player:entity){
    StrippedName = ""
    FirstCol = 1
    ExplodedName = Player:name():explode(">")
    ColouredName = array()
    FirstColVector = vec(255)
    if(Player:name():find("color=") | Player:name():find("c=")){
        for (I=1,ExplodedName:count()) {
            if(ExplodedName[I,string]:find("color=") | ExplodedName[I,string]:find("c=")){
    
                ColorArray = ExplodedName[I,string]:replace("<color=",""):replace("<c=",""):replace(">",""):explode(",")
                Col = vec(ColorArray[1,string]:toNumber(),ColorArray[2,string]:toNumber(),ColorArray[3,string]:toNumber())
                ColouredName:pushVector(Col)
                if(FirstCol){
                    FirstCol = 0
                    FirstColVector = Col
                }
            }
            else{
                ColouredName:pushString(ExplodedName[I,string])
                StrippedName += ExplodedName[I,string]
            }
        }
    }
    else{
        FirstColVector = teamColor(Player:team())
        ColouredName:pushString( Player:name() )
        StrippedName = Player:name()
    }
    ColouredName:pushVector(vec(255))
    return array(StrippedName,FirstColVector,ColouredName)
}
