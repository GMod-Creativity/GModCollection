@name CPU
@inputs I1 I2 I3 I4 I5 RESET
@outputs O1 O2 O3 O4 O5
@inputs IN:wirelink
@outputs ON Elight
@persist OUT:array INPUT:array REG:array EL RUN_PROGRAM LINE R
@persist A B
@trigger
if(RESET==1 & IN[0]==0)
{
    REG[1,number]=0, REG[2,number]=0, REG[3,number]=0, REG[4,number]=0, REG[5,number]=0
}
interval(250)

if(IN[0]==1)
{
    ON=1
    #OUTPUT
    if(IN[1]==1)
    {
		      if(IN[2]==1 | IN[2]==2)
		      {
			         if(IN[3] >= 1 & IN[3] <= 5)
			         {
				            if(IN[4] >= 1 & IN[4] <= 3)
				            {
					               if(IN[4]==1)
					               {
						                  if(IN[2]==1){OUT[IN[3],number] = IN[5]}
						                  if(IN[2]==2){REG[IN[3],number] = IN[5]}
					               }
					               if(IN[4]==2)
					               {
						                  if(IN[5] >= 1 & IN[5] <= 5)
						                  {
							                     if(IN[2]==1){OUT[IN[3],number] = INPUT[IN[5],number]}
							                     if(IN[2]==2){REG[IN[3],number] = INPUT[IN[5],number]}
						                  }
						                  else{IN[0]=0,IN[10]=2,IN[11]=4,EL=1}
					               }
					               if(IN[4]==3)
					               {
                    				if(IN[5] >= 1 & IN[5] <= 5)
						                  {
							                     if(IN[2]==1){OUT[IN[3],number] = REG[IN[5],number]}
							                     if(IN[2]==2){REG[IN[3],number] = REG[IN[5],number]}
						                  }
						                  else{IN[0]=0,IN[10]=1,IN[11]=4,EL=1}
					               }
				            }else{IN[0]=0,IN[10]=1,IN[11]=3,EL=1}
			         }else{IN[0]=0,IN[10]=1,IN[11]=2,EL=1}
		      }else{IN[10]=1,IN[11]=1,EL=1}
        R=1
    }
    #ADDITION
    if(IN[1]==2)
    {
        if(IN[2]==1 | IN[2]==2)
        {
            if(IN[3] >= 1 & IN[3] <= 5)
            {
                if(IN[4] >= 1 & IN[4] <= 3)
                {
                    if(IN[4]==1)
                    {
                        A = IN[5]
                    }
                    if(IN[4]==2)
                    {
                        if(IN[5] >= 1 & IN[5] <= 5)
                        {
                            A = INPUT[IN[5],number]
                        }else{IN[10]=2, IN[11]=4, EL=1}
                    }
                    if(IN[4]==3)
                    {
                        if(IN[5] >= 1 & IN[5] <= 5)
                        {
                            A = REG[IN[5],number]
                        }else{IN[10]=2, IN[11]=4, EL=1}
                    }
                }else{IN[10]=2, IN[11]=3, EL=1}
                if(IN[6] >= 1 & IN[6] <= 3)
                {
                    if(IN[6]==1)
                    {
                        B=IN[7]
                    }
                    if(IN[6]==2)
                    {
                        if(IN[7] >= 1 & IN[7] <= 5)
                        {
                            B = INPUT[IN[7],number]
                        }else{IN[10]=2, IN[11]=6, EL=1}
                    }
                    if(IN[6]==3)
                    {
                        if(IN[7] >= 1 & IN[7] <= 5)
                        {
                            B = REG[IN[7],number]
                        }else{IN[10]=2, IN[11]=6, EL=1}
                    }
                }else{IN[10]=2, IN[11]=5, EL=1}
                if(IN[2]==1)
                {
                    OUT[IN[3],number] = A+B
                }
                if(IN[2]==2)
                {
                    REG[IN[3],number] = A+B
                }
            }else{IN[10]=2, IN[11]=2, EL=1}
        }else{IN[10]=2, IN[11]=1, EL=1}
        R=1
    }
    #SUBTRACTION
    if(IN[1]==3)
    {
        if(IN[2]==1 | IN[2]==2)
        {
            if(IN[3] >= 1 & IN[3] <= 5)
            {
                if(IN[4] >= 1 & IN[4] <= 3)
                {
                    if(IN[4]==1)
                    {
                        A = IN[5]
                    }
                    if(IN[4]==2)
                    {
                        if(IN[5] >= 1 & IN[5] <= 5)
                        {
                            A = INPUT[IN[5],number]
                        }else{IN[10]=3, IN[11]=4, EL=1}
                    }
                    if(IN[4]==3)
                    {
                        if(IN[5] >= 1 & IN[5] <= 5)
                        {
                            A = REG[IN[5],number]
                        }else{IN[10]=3, IN[11]=4, EL=1}
                    }
                }else{IN[10]=3, IN[11]=3, EL=1}
                if(IN[6] >= 1 & IN[6] <= 3)
                {
                    if(IN[6]==1)
                    {
                        B=IN[7]
                    }
                    if(IN[6]==2)
                    {
                        if(IN[7] >= 1 & IN[7] <= 5)
                        {
                            B = INPUT[IN[7],number]
                        }else{IN[10]=3, IN[11]=6, EL=1}
                    }
                    if(IN[6]==3)
                    {
                        if(IN[7] >= 1 & IN[7] <= 5)
                        {
                            B = REG[IN[7],number]
                        }else{IN[10]=3, IN[11]=6, EL=1}
                    }
                }else{IN[10]=3, IN[11]=5, EL=1}
                if(IN[2]==1)
                {
                    OUT[IN[3],number] = A-B
                }
                if(IN[2]==2)
                {
                    REG[IN[3],number] = A-B
                }
            }else{IN[10]=3, IN[11]=2, EL=1}
        }else{IN[10]=3, IN[11]=1, EL=1}
        R=1
    }
    #MULTIPLICATION
    if(IN[1]==4)
    {
        if(IN[2]==1 | IN[2]==2)
        {
            if(IN[3] >= 1 & IN[3] <= 5)
            {
                if(IN[4] >= 1 & IN[4] <= 3)
                {
                    if(IN[4]==1)
                    {
                        A = IN[5]
                    }
                    if(IN[4]==2)
                    {
                        if(IN[5] >= 1 & IN[5] <= 5)
                        {
                            A = INPUT[IN[5],number]
                        }else{IN[10]=4, IN[11]=4, EL=1}
                    }
                    if(IN[4]==3)
                    {
                        if(IN[5] >= 1 & IN[5] <= 5)
                        {
                            A = REG[IN[5],number]
                        }else{IN[10]=4, IN[11]=4, EL=1}
                    }
                }else{IN[10]=4, IN[11]=3, EL=1}
                if(IN[6] >= 1 & IN[6] <= 3)
                {
                    if(IN[6]==1)
                    {
                        B=IN[7]
                    }
                    if(IN[6]==2)
                    {
                        if(IN[7] >= 1 & IN[7] <= 5)
                        {
                            B = INPUT[IN[7],number]
                        }else{IN[10]=4, IN[11]=6, EL=1}
                    }
                    if(IN[6]==3)
                    {
                        if(IN[7] >= 1 & IN[7] <= 5)
                        {
                            B = REG[IN[7],number]
                        }else{IN[10]=4, IN[11]=6, EL=1}
                    }
                }else{IN[10]=4, IN[11]=5, EL=1}
                if(IN[2]==1)
                {
                    OUT[IN[3],number] = A*B
                }
                if(IN[2]==2)
                {
                    REG[IN[3],number] = A*B
                }
            }else{IN[10]=4, IN[11]=2, EL=1}
        }else{IN[10]=4, IN[11]=1, EL=1}
        R=1
    }
    #DIVISION
    if(IN[1]==4)
    {
        if(IN[2]==1 | IN[2]==2)
        {
            if(IN[3] >= 1 & IN[3] <= 5)
            {
                if(IN[4] >= 1 & IN[4] <= 3)
                {
                    if(IN[4]==1)
                    {
                        A = IN[5]
                    }
                    if(IN[4]==2)
                    {
                        if(IN[5] >= 1 & IN[5] <= 5)
                        {
                            A = INPUT[IN[5],number]
                        }else{IN[10]=4, IN[11]=4, EL=1}
                    }
                    if(IN[4]==3)
                    {
                        if(IN[5] >= 1 & IN[5] <= 5)
                        {
                            A = REG[IN[5],number]
                        }else{IN[10]=4, IN[11]=4, EL=1}
                    }
                }else{IN[10]=5, IN[11]=3, EL=1}
                if(IN[6] >= 1 & IN[6] <= 3)
                {
                    if(IN[6]==1)
                    {
                        B=IN[7]
                    }
                    if(IN[6]==2)
                    {
                        if(IN[7] >= 1 & IN[7] <= 5)
                        {
                            B = INPUT[IN[7],number]
                        }else{IN[10]=4, IN[11]=6, EL=1}
                    }
                    if(IN[6]==3)
                    {
                        if(IN[7] >= 1 & IN[7] <= 5)
                        {
                            B = REG[IN[7],number]
                        }else{IN[10]=4, IN[11]=6, EL=1}
                    }
                }else{IN[10]=5, IN[11]=5, EL=1}
                if(IN[2]==1)
                {
                    OUT[IN[3],number] = A/B
                }
                if(IN[2]==2)
                {
                    REG[IN[3],number] = A/B
                }
            }else{IN[10]=5, IN[11]=2, EL=1}
        }else{IN[10]=5, IN[11]=1, EL=1}
        R=1
    }
}            
if(IN[0]==0)
{
    ON=0
}
if(EL==1)
{
    R=1
    IN[0]=0
    if(Elight==0){Elight=1}
    else{Elight=0}
    if(RESET==1){EL=0, Elight=0, ON=0, IN[10]=0, IN[11]=0} 
} 

if(R==1)
{
    IN[1]=0, IN[2]=0, IN[3]=0, IN[4]=0, IN[5]=0, IN[6]=0, IN[7]=0, IN[8]=0, IN[9]=0
	   R=0
}

O1 = OUT[1,number]
O2 = OUT[2,number]
O3 = OUT[3,number]
O4 = OUT[4,number]
O5 = OUT[5,number]

INPUT[1,number] = I1
INPUT[2,number] = I2
INPUT[3,number] = I3
INPUT[4,number] = I4
INPUT[5,number] = I5
