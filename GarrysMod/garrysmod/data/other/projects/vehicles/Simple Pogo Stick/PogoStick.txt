@name Pogo
@outputs Ang:angle EA:angle
@inputs B:entity P:entity
@persist Se:entity M1 Dr:entity
#@trigger none
interval(10)
Se=entity():isConstrainedTo()
Roll=B:angles():roll()
Ang=angnorm(-B:angles())
B:applyAngForce((Ang+$Ang*5)*B:mass()*25)
#B:applyAngForce(-B:angVel()*B:mass())
EA=P:angles()
Dr=Se:driver()
if(Se==noentity()){
    selfDestruct()
}
M1=Dr:keyAttack1()
if(changed(M1)&M1){
    timer("f",20)
}
if(clk("f")){
    P:applyForce(P:toWorld(vec(0,0,-2000)*P:mass()*2))
    B:applyForce(B:toWorld(vec(0,0,-2000)*B:mass()*2))
}
